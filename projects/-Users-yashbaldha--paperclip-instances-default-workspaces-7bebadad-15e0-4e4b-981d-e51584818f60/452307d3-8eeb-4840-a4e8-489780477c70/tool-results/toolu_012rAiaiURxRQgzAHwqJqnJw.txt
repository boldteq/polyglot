   357→  // ── Scope custom CSS: auto-prefix selectors with widget container ID ──────
   358→  function scopeCustomCss(rawCss, wid) {
   359→    if (!rawCss) return "";
   360→    // Strip </style> tags to prevent injection
   361→    var css = rawCss.replace(/<\/style>/gi, "");
   362→    var result = "";
   363→    var i = 0;
   364→    while (i < css.length) {
   365→      // Skip whitespace
   366→      while (i < css.length && /\s/.test(css[i])) { result += css[i]; i++; }
   367→      if (i >= css.length) break;
   368→      // Pass through @keyframes and @font-face unchanged
   369→      if (css[i] === '@') {
   370→        var atRule = "";
   371→        var j = i;
   372→        while (j < css.length && css[j] !== '{' && css[j] !== ';') { atRule += css[j]; j++; }
   373→        var atName = atRule.trim().toLowerCase();
   374→        if (atName.indexOf("@keyframes") === 0 || atName.indexOf("@font-face") === 0) {
   375→          result += atRule;
   376→          if (j < css.length && css[j] === '{') {
   377→            var depth = 1; result += css[j]; j++;
   378→            while (j < css.length && depth > 0) {
   379→              if (css[j] === '{') depth++;
   380→              if (css[j] === '}') depth--;
   381→              result += css[j]; j++;
   382→            }
   383→          }
   384→          i = j; continue;
   385→        }
   386→        // @media: scope inner rules
   387→        if (atName.indexOf("@media") === 0) {
   388→          result += atRule;
   389→          if (j < css.length && css[j] === '{') {
   390→            result += '{'; j++;
   391→            // Collect inner content
   392→            var depth = 1; var inner = "";
   393→            while (j < css.length && depth > 0) {
   394→              if (css[j] === '{') depth++;
   395→              if (css[j] === '}') depth--;
   396→              if (depth > 0) inner += css[j];
   397→              j++;
   398→            }
   399→            result += scopeCustomCss(inner, wid);
   400→            result += '}';
   401→          }
   402→          i = j; continue;
   403→        }
   404→      }
   405→      // Regular rule: grab selector, prefix with #wid
   406→      var sel = ""; var body = "";
   407→      while (i < css.length && css[i] !== '{') { sel += css[i]; i++; }
   408→      if (i < css.length) { i++; } // skip {
   409→      var depth = 1;
   410→      while (i < css.length && depth > 0) {
   411→        if (css[i] === '{') depth++;
   412→        if (css[i] === '}') depth--;
   413→        if (depth > 0) body += css[i];
   414→        i++;
   415→      }
   416→      if (sel.trim()) {
   417→        // Split multiple selectors by comma and prefix each
   418→        var selectors = sel.split(',').map(function(s) {
   419→          s = s.trim();
   420→          if (!s) return '';
   421→          return '#' + wid + ' ' + s;
   422→        }).filter(Boolean).join(', ');
   423→        result += selectors + ' {' + body + '}';
   424→      }
   425→    }
   426→    return result;
   427→  }

   430→  function formatDeliveryDate(isoDate) {
   431→    if (!isoDate) return null;
   432→    try {
   433→      var parts = isoDate.split('-');
   434→      var d = new Date(parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2]));
   435→      var days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
   436→      var months = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
   437→      return days[d.getDay()] + ', ' + months[d.getMonth()] + ' ' + d.getDate();
   438→    } catch(e) { return null; }
   439→  }

   441→  // ── Countdown timer functions ─────────────────────────────────────────────────
   442→  function getCountdownData(cutoffTime) {
   443→    if (!cutoffTime) return null;
   444→    var parts = cutoffTime.split(':');
   445→    var cutoffHr = parseInt(parts[0], 10);
   446→    var cutoffMin = parseInt(parts[1] || '0', 10);
   447→    var now = new Date();
   448→    var cutoff = new Date();
   449→    cutoff.setHours(cutoffHr, cutoffMin, 0, 0);
   450→    var diff = cutoff - now;
   451→    if (diff <= 0) return { passed: true, hours: 0, minutes: 0 };
   452→    var hours = Math.floor(diff / 3600000);
   453→    var minutes = Math.floor((diff % 3600000) / 60000);
   454→    return { passed: false, hours: hours, minutes: minutes };
   455→  }

   457→  function buildCountdownHtml(cutoffTime, widgetId) {
   458→    var cd = getCountdownData(cutoffTime);
   459→    if (!cd) return '';
   460→    var colorClass, text;
   461→    if (cd.passed) {
   462→      colorClass = 'zcc-countdown--passed';
   463→      text = 'Order now for next business day dispatch';
   464→    } else {
   465→      var totalMinutes = cd.hours * 60 + cd.minutes;
   466→      if (totalMinutes > 180) colorClass = 'zcc-countdown--green';
   467→      else if (totalMinutes > 60) colorClass = 'zcc-countdown--amber';
   468→      else colorClass = 'zcc-countdown--red';

   677→    function normalizeZip(raw) {
   678→      var z = raw.trim().toUpperCase();
   679→      z = z.replace(/\s+/g, '');        // collapse all spaces (UK: SW1A 2AA → SW1A2AA)
   680→      z = z.replace(/-\d{4}$/, '');     // strip US ZIP+4 suffix (90210-1234 → 90210)
   681→      return z;

   683→
   684→    // Store the original button HTML so we can restore it exactly after loading
   685→    var btnOriginalHtml = btn.innerHTML;

   687→    function setBtnLoading(loading) {

   690→        btn.classList.add('zcc-btn--loading');
   691→        // Replace entire button content with a lone spinner — no text overlap
   692→        btn.innerHTML = '<span class="zcc-spinner" aria-hidden="true"></span>';
   693→      } else {
   694→        btn.classList.remove('zcc-btn--loading');
   695→        // Restore original icon + label markup
   696→        btn.innerHTML = btnOriginalHtml;
   697→        // Keep button text in sync with config (merchant may have customised it)
   698→        var labelEl = btn.querySelector('.zcc-btn-label');
   699→        if (labelEl) labelEl.textContent = cfg.buttonText;

   703→    function checkZip() {

   708→      setBtnLoading(true);

   717→            // Non-2xx: try to parse JSON error body, fall back to status text
   718→            return r.json().catch(function() {
   719→              return { error: 'Server error (' + r.status + ')' };

   930→          var lastValid = sessionStorage.getItem('zcc:valid') === 'true';
   931→          // If the last check was successful, re-block as soon as the input
   932→          // changes (any change away from the verified value is unverified).
   933→          if (lastValid && lastValidZip) {
   934→            var currentVal = normalizeZip(inp.value);
   935→            if (currentVal !== lastValidZip) {
   936→              // Mark as unverified immediately
   937→              sessionStorage.setItem('zcc:valid', 'false');
   938→              sessionStorage.removeItem('zcc:zip');
   939→              // Re-block ATC buttons until the new ZIP is verified
   940→              var ATC_SELECTORS = [
   941→                'button[name="add"]',
   942→                '.product-form__submit',
   943→                '[data-product-add]',
   944→                '.btn--add-to-cart',
   945→                '.product-form button[type="submit"]',
   946→                'button.shopify-payment-button__button',
   947→                '.shopify-payment-button button',
   948→                'button.product-form__cart-submit'
   949→              ].join(',');
   950→              var atcBtns = document.querySelectorAll(ATC_SELECTORS);
   951→              atcBtns.forEach(function(b) {
   952→                if (!b.closest('[data-zcc-widget]')) {

   959→                  b.setAttribute('data-zcc-blocked', 'true');
   960→                }
   961→              });
   962→            }
   963→          }
   964→          // If the last check was NOT successful (buttons are already blocked),
   965→          // do nothing — clearing or changing the input must never unblock buttons.
   966→        } catch(e) {}
   967→      });
   968→    }

   971→  function joinWaitlist(zip, email, name, wBtn) {
   972→    // Store original button HTML so we can restore it exactly (same pattern as setBtnLoading)
   973→    var wBtnOriginalHtml = wBtn.innerHTML;
   974→
   975→    wBtn.disabled = true;
   976→    wBtn.classList.add('zcc-btn--loading');
   977→    wBtn.innerHTML = '<span class="zcc-spinner" aria-hidden="true"></span>';
   978→
   979→    var payload = { shop: SHOP, zip: zip, email: email };
   980→    if (name) payload.name = name;
   981→
   982→    function restoreBtn() {
   983→      wBtn.disabled = false;
   984→      wBtn.classList.remove('zcc-btn--loading');
   985→      wBtn.innerHTML = wBtnOriginalHtml;
   986→    }
   987→
   988→    fetch(API_BASE + "/api/waitlist", {
   989→      method: "POST",
   990→      headers: { "Content-Type": "application/json" },
   991→      body: JSON.stringify(payload),
   992→    })
   993→      .then(function (r) { return r.json(); })
   994→      .then(function (data) {
   995→        var wlEl = wBtn.closest(".zcc-wl");

   997→          if (wlEl) wlEl.innerHTML =

  1001→            '</div>';
  1002→        } else {
  1003→          restoreBtn();
  1004→        }
  1005→      })
  1006→      .catch(function () {
  1007→        restoreBtn();
  1008→      });
  1009→  }
