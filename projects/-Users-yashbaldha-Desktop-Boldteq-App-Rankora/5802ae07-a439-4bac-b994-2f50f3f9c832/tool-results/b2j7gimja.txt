import { useState } from "react";
import { toast } from "sonner";
import { format } from "date-fns";
import {
  Users,
  Crown,
  Shield,
  UserPlus,
  Trash2,
  Loader2,
  Mail,
  RotateCw,
  Briefcase,
  FileText,
  CheckSquare,
  ArrowRight,
  AlertCircle,
  BarChart3,
  Settings as SettingsIcon,
  LogOut,
  Activity as ActivityIcon,
} from "lucide-react";
import { useNavigate, useSearchParams } from "react-router-dom";
import SidebarLayout from "@/components/SidebarLayout";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Badge } from "@/components/ui/badge";
import { Skeleton } from "@/components/ui/skeleton";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import {
  Table,
  TableBody,
  TableCell,
  TableHead,
  TableHeader,
  TableRow,
} from "@/components/ui/table";
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
} from "@/components/ui/alert-dialog";
import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";
import { useTeam, type TeamMember, type TeamInvitation } from "@/hooks/useTeam";
import PipelineStageManager from "@/components/PipelineStageManager";
import ScorecardTemplateManager from "@/components/ScorecardTemplateManager";
import StageAutomationManager from "@/components/StageAutomationManager";
import { useTeamDashboard } from "@/hooks/useTeamDashboard";
import ActivityItem from "@/components/ActivityItem";
import { supabase } from "@/integrations/supabase/client";
import { useAuth } from "@/hooks/useAuth";
import { useActivityFeed } from "@/hooks/useActivityFeed";

type TeamRole = "owner" | "admin" | "member";

const ROLE_LABELS: Record<TeamRole, string> = {
  owner: "Owner",
  admin: "Admin",
  member: "Member",
};

const ROLE_VARIANTS: Record<TeamRole, "default" | "secondary" | "outline"> = {
  owner: "default",
  admin: "secondary",
  member: "outline",
};

function RoleBadge({ role }: { role: TeamRole }) {
  const Icon = role === "owner" ? Crown : role === "admin" ? Shield : Users;
  return (
    <Badge variant={ROLE_VARIANTS[role]} className="gap-1 text-xs font-medium">
      <Icon className="h-3 w-3" />
      {ROLE_LABELS[role]}
    </Badge>
  );
}

function MembersSkeleton() {
  return (
    <div className="space-y-3">
      {[1, 2, 3].map((i) => (
        <div key={i} className="flex items-center gap-4 px-4 py-3">
          <Skeleton className="h-8 w-8 rounded-full shrink-0" />
          <div className="flex-1 space-y-1.5">
            <Skeleton className="h-4 w-48" />
            <Skeleton className="h-3 w-32" />
          </div>
          <Skeleton className="h-6 w-16 rounded-full" />
          <Skeleton className="h-8 w-8 rounded" />
        </div>
      ))}
    </div>
  );
}

const TAB_TRIGGER_CLASS =
  "rounded-none border-b-2 border-transparent data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:shadow-none px-4 pb-2.5 text-sm font-medium";

// ─── Activity filter constants ────────────────────────────────────────────────

const ACTIVITY_FILTERS = ["All", "Tasks", "Notes", "Stages", "Applications", "Members"] as const;
type ActivityFilter = (typeof ACTIVITY_FILTERS)[number];

// ─── Valid tab values ─────────────────────────────────────────────────────────

const VALID_TABS = ["overview", "members", "activity", "settings"] as const;
type TabValue = (typeof VALID_TABS)[number];

function isValidTab(v: string | null): v is TabValue {
  return VALID_TABS.includes(v as TabValue);
}

// ─── Activity skeleton ────────────────────────────────────────────────────────

function ActivitySkeleton() {
  return (
    <div className="space-y-3 p-4">
      {[1, 2, 3, 4].map((i) => (
        <div key={i} className="flex items-center gap-3">
          <Skeleton className="h-8 w-8 rounded-full shrink-0" />
          <Skeleton className="h-12 flex-1 rounded-lg" />
        </div>
      ))}
    </div>
  );
}

const Team = () => {
  const { user } = useAuth();
  const navigate = useNavigate();
  const [searchParams, setSearchParams] = useSearchParams();

  // URL-synced tab — defaults to "overview"
  const rawTab = searchParams.get("tab");
  const activeTab: TabValue = isValidTab(rawTab) ? rawTab : "overview";

  function setTab(tab: TabValue) {
    setSearchParams((prev) => {
      const next = new URLSearchParams(prev);
      next.set("tab", tab);
      return next;
    }, { replace: true });
  }

  const { team, members, invitations, myRole, isLoading, error, refetch } = useTeam();

  const [inviteEmail, setInviteEmail] = useState("");
  const [inviteRole, setInviteRole] = useState<"admin" | "member">("member");
  const [isSendingInvite, setIsSendingInvite] = useState(false);
  const [revokingId, setRevokingId] = useState<string | null>(null);
  const [removingId, setRemovingId] = useState<string | null>(null);
  const [updatingRoleId, setUpdatingRoleId] = useState<string | null>(null);

  const [leaveDialogOpen, setLeaveDialogOpen] = useState(false);
  const [revokeDialogTarget, setRevokeDialogTarget] = useState<TeamInvitation | null>(null);
  const [removeDialogTarget, setRemoveDialogTarget] = useState<TeamMember | null>(null);
  const [isLeavingTeam, setIsLeavingTeam] = useState(false);
  const [teamNameInput, setTeamNameInput] = useState("");
  const [isSavingTeamName, setIsSavingTeamName] = useState(false);

  // Activity tab state
  const [activityFilter, setActivityFilter] = useState<ActivityFilter>("All");
  const { activities, isLoading: actLoading, isFetching: actFetching, hasMore: actHasMore, loadMore: actLoadMore, error: actError } =
    useActivityFeed(team?.id ?? null, activityFilter);

  const canManage = myRole === "owner" || myRole === "admin";

  async function handleInvite(e: React.FormEvent) {
    e.preventDefault();
    if (!inviteEmail.trim() || !team?.id) return;

    setIsSendingInvite(true);
    try {
      const { data: { session } } = await supabase.auth.getSession();
      if (!session?.access_token) throw new Error("Not authenticated");

      const res = await fetch(
        `${import.meta.env.VITE_SUPABASE_URL}/functions/v1/send-team-invite`,
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${session.access_token}`,
            "apikey": import.meta.env.VITE_SUPABASE_ANON_KEY,
          },
          body: JSON.stringify({
            team_id: team.id,
            email: inviteEmail.trim().toLowerCase(),
            role: inviteRole,
          }),
        }
      );

      const json = await res.json();
      if (!res.ok) throw new Error(json.error || "Failed to send invitation");

      toast.success(`Invitation sent to ${inviteEmail.trim()}`);
      setInviteEmail("");
      refetch();
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : "Failed to send invitation";
      toast.error(msg);
    } finally {
      setIsSendingInvite(false);
    }
  }

  async function handleConfirmRevoke() {
    if (!revokeDialogTarget) return;
    const { id, email } = revokeDialogTarget;
    setRevokingId(id);
    setRevokeDialogTarget(null);
    try {
      const { error } = await (supabase as any)
        .from("team_invitations")
        .update({ status: "revoked" })
        .eq("id", id);

      if (error) throw error;

      toast.success(`Invitation to ${email} revoked`);
      refetch();
    } catch (err: unknown) {
      toast.error("Failed to revoke invitation");
    } finally {
      setRevokingId(null);
    }
  }

  async function handleChangeRole(member: TeamMember, newRole: TeamRole) {
    if (newRole === member.role) return;
    setUpdatingRoleId(member.id);
    try {
      const { error } = await (supabase as any)
        .from("team_members")
        .update({ role: newRole })
        .eq("id", member.id);

      if (error) throw error;

      toast.success(`Role updated to ${ROLE_LABELS[newRole]}`);
      refetch();
    } catch (err: unknown) {
      toast.error("Failed to update role");
    } finally {
      setUpdatingRoleId(null);
    }
  }

  async function handleRemoveMember(member: TeamMember) {
    setRemovingId(member.id);
    try {
      const { error } = await (supabase as any)
        .from("team_members")
        .delete()
        .eq("id", member.id);

      if (error) throw error;

      toast.success("Member removed from team");
      refetch();
    } catch (err: unknown) {
      toast.error("Failed to remove member");
    } finally {
      setRemovingId(null);
    }
  }

  async function handleLeaveTeam() {
    if (!user?.id || !team?.id) return;
    setIsLeavingTeam(true);
    try {
      const myMember = members.find((m) => m.user_id === user.id);
      if (!myMember) throw new Error("You are not a member of this team");

      const { error } = await (supabase as any)
        .from("team_members")
        .delete()
        .eq("id", myMember.id);

      if (error) throw error;

      toast.success("You have left the team");
      navigate("/");
    } catch (err: unknown) {
      const msg = err instanceof Error ? err.message : "Failed to leave team";
      toast.error(msg);
      setIsLeavingTeam(false);
    }
  }

  async function handleSaveTeamName() {
    const trimmed = teamNameInput.trim();
    if (!trimmed || !team?.id) return;
    setIsSavingTeamName(true);
    try {
      const { error } = await (supabase as any)
        .from("teams")
        .update({ name: trimmed })
        .eq("id", team.id);

      if (error) throw error;

      toast.success("Team name updated");
      refetch();
      setTeamNameInput("");
    } catch (err: unknown) {
      toast.error("Failed to update team name");
    } finally {
      setIsSavingTeamName(false);
    }
  }

  const pendingInvitations = invitations.filter((inv) => inv.status === "pending");

  const { data: dashboard, isLoading: dashLoading } = useTeamDashboard(team?.id ?? null);

  const statCards = [
    {
      icon: Briefcase,
      label: "Total Jobs",
      value: dashboard?.totalJobs ?? 0,
    },
    {
      icon: FileText,
      label: "Active Candidates",
      value: dashboard?.activeCandidates ?? 0,
    },
    {
      icon: CheckSquare,
      label: "Pending Tasks",
      value: dashboard?.pendingTasks ?? 0,
    },
    {
      icon: Users,
      label: "Team Members",
      value: dashboard?.memberCount ?? 0,
    },
  ];

  return (
    <SidebarLayout>
      <main className="w-full px-6 sm:px-8 lg:px-10 py-8">
        {/* Page header */}
        <div className="flex items-center justify-between mb-1">
          <div>
            <h1 className="text-2xl font-bold text-foreground">Team Hub</h1>
            {team && (
              <p className="text-sm text-muted-foreground mt-0.5">{team.name}</p>
            )}
          </div>
          {myRole !== "owner" && myRole !== null && (
            <Button
              variant="outline"
              size="sm"
              className="gap-1.5 text-destructive border-destructive/40 hover:bg-destructive/10 hover:text-destructive"
              onClick={() => setLeaveDialogOpen(true)}
              disabled={isLeavingTeam}
            >
              {isLeavingTeam ? (
                <Loader2 className="h-3.5 w-3.5 animate-spin" />
              ) : (
                <LogOut className="h-3.5 w-3.5" />
              )}
              Leave team
            </Button>
          )}
        </div>
        <p className="text-sm text-muted-foreground mb-6">
          Manage your team members, activity, and settings.
        </p>

        {/* Error state */}
        {!isLoading && error && (
          <Card className="border-destructive/50 mb-6">
            <CardContent className="py-8 text-center">
              <AlertCircle className="h-8 w-8 text-destructive mx-auto mb-2" />
              <p className="text-sm text-muted-foreground">Failed to load data. Please refresh the page.</p>
            </CardContent>
          </Card>
        )}

        {/* ── Tabs ── */}
        <Tabs value={activeTab} onValueChange={(v) => setTab(v as TabValue)} className="w-full">
          <TabsList className="w-full justify-start border-b border-border bg-transparent px-0 rounded-none h-auto mb-0">
            <TabsTrigger value="overview" className={TAB_TRIGGER_CLASS}>
              <BarChart3 className="h-3.5 w-3.5 mr-1.5" />
              Overview
            </TabsTrigger>
            <TabsTrigger value="members" className={TAB_TRIGGER_CLASS}>
              <Users className="h-3.5 w-3.5 mr-1.5" />
              Members
            </TabsTrigger>
            <TabsTrigger value="activity" className={TAB_TRIGGER_CLASS}>
              <ActivityIcon className="h-3.5 w-3.5 mr-1.5" />
              Activity
            </TabsTrigger>
            {canManage && (
              <TabsTrigger value="settings" className={TAB_TRIGGER_CLASS}>
                <SettingsIcon className="h-3.5 w-3.5 mr-1.5" />
                Settings
              </TabsTrigger>
            )}
          </TabsList>

          {/* ---------------------------------------------------------------- */}
          {/* Tab 1: Overview                                                   */}
          {/* ---------------------------------------------------------------- */}
          <TabsContent value="overview" className="mt-6 space-y-4">
            {!team ? (
              <Card className="border border-border/60 shadow-sm">
                <CardContent className="py-12 text-center">
                  <div className="rounded-xl bg-muted/50 p-4 inline-flex mb-3">
                    <Users className="h-7 w-7 text-muted-foreground/50" />
                  </div>
                  <p className="text-sm font-medium text-foreground mb-1">No team yet</p>
                  <p className="text-sm text-muted-foreground max-w-xs mx-auto">
                    You are not part of a team. Ask your team owner to invite you.
                  </p>
                </CardContent>
              </Card>
            ) : (
              <>
                {/* Stats row */}
                <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
                  {statCards.map(({ icon: Icon, label, value }) => (
                    <Card key={label} className="border border-border/60 shadow-sm">
                      <CardContent className="p-4">
                        <div className="flex items-center gap-2.5 mb-2">
                          <div className="h-8 w-8 rounded-lg bg-primary/10 flex items-center justify-center shrink-0">
                            <Icon className="h-4 w-4 text-primary" />
                          </div>
                        </div>
                        {dashLoading ? (
                          <Skeleton className="h-7 w-12 mb-1" />
                        ) : (
                          <p className="text-2xl font-bold text-foreground">{value}</p>
                        )}
                        <p className="text-xs text-muted-foreground">{label}</p>
                      </CardContent>
                    </Card>
                  ))}
                </div>

                {/* Recent Activity */}
                <Card className="border border-border/60 shadow-sm">
                  <CardHeader className="pb-2 pt-4">
                    <div className="flex items-center justify-between">
                      <CardTitle className="text-sm font-semibold">Recent Activity</CardTitle>
                      <button
                        onClick={() => setTab("activity")}
                        className="inline-flex items-center gap-1 text-xs text-primary hover:text-primary/80 transition-colors font-medium"
                      >
                        View all
                        <ArrowRight className="h-3 w-3" />
                      </button>
                    </div>
                  </CardHeader>
                  <CardContent className="pb-3">
                    {dashLoading ? (
                      <div className="space-y-2">
                        {[1, 2, 3].map((i) => (
                          <Skeleton key={i} className="h-9 w-full rounded-lg" />
                        ))}
                      </div>
                    ) : !dashboard?.recentActivity.length ? (
                      <p className="text-sm text-muted-foreground py-2">No activity yet</p>
                    ) : (
                      <div className="divide-y divide-border/40">
                        {dashboard.recentActivity.map((activity) => (
                          <ActivityItem key={activity.id} activity={activity} compact />
                        ))}
                      </div>
                    )}
                  </CardContent>
                </Card>

                {/* Task summary by assignee */}
                {!dashLoading && dashboard?.tasksByAssignee && dashboard.tasksByAssignee.length > 0 && (
                  <Card className="border border-border/60 shadow-sm">
                    <CardHeader className="pb-2 pt-4">
                      <CardTitle className="text-sm font-semibold">Task Summary by Member</CardTitle>
                    </CardHeader>
                    <CardContent className="p-0">
                      <Table>
                        <TableHeader>
                          <TableRow>
                            <TableHead className="text-xs">Member</TableHead>
                            <TableHead className="text-xs text-right">Pending</TableHead>
                            <TableHead className="text-xs text-right">Completed</TableHead>
                          </TableRow>
                        </TableHeader>
                        <TableBody>
                          {dashboard.tasksByAssignee.map((row) => {
                            const member = members.find((m) => m.user_id === row.user_id);
                            const label = member?.email ?? row.user_id.slice(0, 8) + "…";
                            return (
                              <TableRow key={row.user_id}>
                                <TableCell className="text-sm font-medium">{label}</TableCell>
                                <TableCell className="text-sm text-right text-amber-600 dark:text-amber-400 font-medium">
                                  {row.pending}
                                </TableCell>
                                <TableCell className="text-sm text-right text-emerald-600 dark:text-emerald-400 font-medium">
                                  {row.completed}
                                </TableCell>
                              </TableRow>
                            );
                          })}
                        </TableBody>
                      </Table>
                    </CardContent>
                  </Card>
                )}
              </>
            )}
          </TabsContent>

          {/* ---------------------------------------------------------------- */}
          {/* Tab 2: Members                                                    */}
          {/* ---------------------------------------------------------------- */}
          <TabsContent value="members" className="mt-6 space-y-6">
            {/* Invite card — owners/admins only */}
            {canManage && (
              <Card className="border border-border/60 shadow-sm">
                <CardHeader className="pb-3">
                  <CardTitle className="text-sm font-semibold flex items-center gap-2">
                    <UserPlus className="h-4 w-4 text-primary" />
                    Invite a teammate
                  </CardTitle>
                </CardHeader>
                <CardContent>
                  <form onSubmit={handleInvite} className="flex flex-col sm:flex-row gap-2">
                    <div className="relative flex-1">
                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground pointer-events-none" />
                      <Input
                        type="email"
                        placeholder="colleague@company.com"
                        value={inviteEmail}
                        onChange={(e) => setInviteEmail(e.target.value)}
                        className="pl-9"
                        required
                        disabled={isSendingInvite}
                      />
                    </div>
                    <Select
                      value={inviteRole}
                      onValueChange={(v) => setInviteRole(v as "admin" | "member")}
                      disabled={isSendingInvite}
                    >
                      <SelectTrigger className="w-full sm:w-32">
                        <SelectValue />
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value="member">Member</SelectItem>
                        <SelectItem value="admin">Admin</SelectItem>
                      </SelectContent>
                    </Select>
                    <Button
                      type="submit"
                      disabled={isSendingInvite || !inviteEmail.trim()}
                      className="gap-1.5 shrink-0"
                    >
                      {isSendingInvite ? (
                        <Loader2 className="h-4 w-4 animate-spin" />
                      ) : (
                        <UserPlus className="h-4 w-4" />
                      )}
                      Send invite
                    </Button>
                  </form>
                </CardContent>
              </Card>
            )}

            {/* Pending invitations */}
            {(pendingInvitations.length > 0 || (isLoading && canManage)) && (
              <Card className="border border-border/60 shadow-sm">
                <CardHeader className="pb-3">
                  <div className="flex items-center justify-between">
                    <CardTitle className="text-sm font-semibold flex items-center gap-2">
                      <Mail className="h-4 w-4 text-primary" />
                      Pending invitations
                      {!isLoading && (
                        <Badge variant="secondary" className="ml-1 text-xs">
                          {pendingInvitations.length}
                        </Badge>
                      )}
                    </CardTitle>
                    <Button
                      variant="ghost"
                      size="icon"
                      className="h-7 w-7"
                      onClick={() => refetch()}
                      disabled={isLoading}
                      aria-label="Refresh invitations"
                    >
                      <RotateCw className={`h-3.5 w-3.5 ${isLoading ? "animate-spin" : ""}`} />
                    </Button>
                  </div>
                </CardHeader>
                <CardContent className="p-0">
                  {isLoading ? (
                    <div className="space-y-2 p-4">
                      {[1, 2].map((i) => (
                        <Skeleton key={i} className="h-10 w-full rounded-md" />
                      ))}
                    </div>
                  ) : (
                    <Table>
                      <TableHeader>
                        <TableRow>
                          <TableHead className="text-xs">Email</TableHead>
                          <TableHead className="text-xs">Role</TableHead>
                          <TableHead className="text-xs">Expires</TableHead>
                          {canManage && <TableHead className="text-xs w-16" />}
                        </TableRow>
                      </TableHeader>
                      <TableBody>
                        {pendingInvitations.map((inv: TeamInvitation) => (
                          <TableRow key={inv.id}>
                            <TableCell className="text-sm font-medium">{inv.email}</TableCell>
                            <TableCell>
                              <RoleBadge role={inv.role} />
                            </TableCell>
                            <TableCell className="text-xs text-muted-foreground">
                              {inv.expires_at
                                ? format(new Date(inv.expires_at), "MMM d, yyyy")
                                : "—"}
                            </TableCell>
                            {canManage && (
                              <TableCell>
                                <Button
                                  variant="ghost"
                                  size="icon"
                                  className="h-7 w-7 text-muted-foreground hover:text-destructive"
                                  onClick={() => setRevokeDialogTarget(inv)}
                                  disabled={revokingId === inv.id}
                                  aria-label={`Revoke invite for ${inv.email}`}
                                >
                                  {revokingId === inv.id ? (
                                    <Loader2 className="h-3.5 w-3.5 animate-spin" />
                                  ) : (
                                    <Trash2 className="h-3.5 w-3.5" />
                                  )}
                                </Button>
                              </TableCell>
                            )}
                          </TableRow>
                        ))}
                      </TableBody>
                    </Table>
                  )}
                </CardContent>
              </Card>
            )}

            {/* Members table */}
            <Card className="border border-border/60 shadow-sm">
              <CardHeader className="pb-3">
                <div className="flex items-center justify-between">
                  <CardTitle className="text-sm font-semibold flex items-center gap-2">
                    <Users className="h-4 w-4 text-primary" />
                    Members
                    {!isLoading && (
                      <Badge variant="secondary" className="ml-1 text-xs">
                        {members.length}
                      </Badge>
                    )}
                  </CardTitle>
                  <Button
                    variant="ghost"
                    size="icon"
                    className="h-7 w-7"
                    onClick={() => refetch()}
                    disabled={isLoading}
                    aria-label="Refresh members"
                  >
                    <RotateCw className={`h-3.5 w-3.5 ${isLoading ? "animate-spin" : ""}`} />
                  </Button>
                </div>
              </CardHeader>
              <CardContent className="p-0">
                {isLoading ? (
                  <MembersSkeleton />
                ) : members.length === 0 ? (
                  <div className="flex flex-col items-center justify-center py-12 space-y-3 text-center px-4">
                    <div className="rounded-xl bg-muted/50 p-4">
                      <Users className="h-7 w-7 text-muted-foreground/50" />
                    </div>
                    <div className="space-y-1">
                      <p className="text-sm font-medium text-foreground">No team yet</p>
                      <p className="text-sm text-muted-foreground max-w-xs">
                        Invite a colleague above to start collaborating on job rankings.
                      </p>
                    </div>
                  </div>
                ) : (
                  <Table>
                    <TableHeader>
                      <TableRow>
                        <TableHead className="text-xs">Member</TableHead>
                        <TableHead className="text-xs">Role</TableHead>
                        <TableHead className="text-xs">Joined</TableHead>
                        {canManage && <TableHead className="text-xs w-12" />}
                      </TableRow>
                    </TableHeader>
                    <TableBody>
                      {members.map((member: TeamMember) => {
                        const isCurrentUser = member.user_id === user?.id;
                        const isOwner = member.role === "owner";
                        return (
                          <TableRow key={member.id}>
                            <TableCell>
                              <div className="flex flex-col">
                                <span className="text-sm font-medium">
                                  {member.email ?? member.user_id.slice(0, 8) + "…"}
                                  {isCurrentUser && (
                                    <span className="ml-1.5 text-xs text-muted-foreground font-normal">(you)</span>
                                  )}
                                </span>
                              </div>
                            </TableCell>
                            <TableCell>
                              {canManage && !isOwner && !isCurrentUser ? (
                                <Select
                                  value={member.role}
                                  onValueChange={(v) => handleChangeRole(member, v as TeamRole)}
                                  disabled={updatingRoleId === member.id}
                                >
                                  <SelectTrigger className="h-7 w-28 text-xs border-0 bg-transparent px-0 focus:ring-0 focus:ring-offset-0">
                                    <SelectValue />
                                  </SelectTrigger>
                                  <SelectContent>
                                    <SelectItem value="admin">Admin</SelectItem>
                                    <SelectItem value="member">Member</SelectItem>
                                  </SelectContent>
                                </Select>
                              ) : (
                                <RoleBadge role={member.role} />
                              )}
                            </TableCell>
                            <TableCell className="text-xs text-muted-foreground">
                              {member.joined_at
                                ? format(new Date(member.joined_at), "MMM d, yyyy")
                                : "—"}
                            </TableCell>
                            {canManage && (
                              <TableCell>
                                {!isOwner && !isCurrentUser && (
                                  <Button
                                    variant="ghost"
                                    size="icon"
                                    className="h-7 w-7 text-muted-foreground hover:text-destructive"
                                    disabled={removingId === member.id}
                                    aria-label={`Remove ${member.email ?? "member"}`}
                                    onClick={() => setRemoveDialogTarget(member)}
                                  >
                                    {removingId === member.id ? (
                                      <Loader2 className="h-3.5 w-3.5 animate-spin" />
                                    ) : (
                                      <Trash2 className="h-3.5 w-3.5" />
                                    )}
                                  </Button>
                                )}
                              </TableCell>
                            )}
                          </TableRow>
                        );
                      })}
                    </TableBody>
                  </Table>
                )}
              </CardContent>
            </Card>
          </TabsContent>

          {/* ---------------------------------------------------------------- */}
          {/* Tab 3: Activity                                                    */}
          {/* ---------------------------------------------------------------- */}
          <TabsContent value="activity" className="mt-6">
            {/* Filter pills */}
            <div className="flex flex-wrap gap-1.5 mb-5">
              {ACTIVITY_FILTERS.map((f) => (
                <button
                  key={f}
                  onClick={() => setActivityFilter(f)}
                  className={
                    activityFilter === f
                      ? "px-3 h-7 rounded-full text-xs font-semibold bg-primary text-primary-foreground transition-colors"
                      : "px-3 h-7 rounded-full text-xs font-medium text-muted-foreground hover:text-foreground hover:bg-muted/60 transition-colors"
                  }
                >
                  {f}
                </button>
              ))}
            </div>

            {/* No team */}
            {!team && (
              <div className="flex flex-col items-center justify-center py-16 space-y-4 text-center">
                <div className="rounded-xl bg-muted/50 p-4">
                  <ActivityIcon className="h-7 w-7 text-muted-foreground/50" />
                </div>
                <div className="space-y-1">
                  <p className="text-sm font-medium">No team yet</p>
                  <p className="text-sm text-muted-foreground max-w-xs">
                    Create or join a team to see activity here.
                  </p>
                </div>
              </div>
            )}

            {/* Loading */}
            {team && actLoading && (
              <Card className="border border-border/60 shadow-sm">
                <ActivitySkeleton />
              </Card>
            )}

            {/* Error */}
            {team && !actLoading && actError && (
              <Card className="border-destructive/50">
                <CardContent className="py-8 text-center">
                  <AlertCircle className="h-8 w-8 text-destructive mx-auto mb-2" />
                  <p className="text-sm text-muted-foreground">Failed to load activity. Please refresh the page.</p>
                </CardContent>
              </Card>
            )}

            {/* Empty */}
            {team && !actLoading && !actError && activities.length === 0 && (
              <div className="flex flex-col items-center justify-center py-16 space-y-4 text-center">
                <div className="rounded-xl bg-muted/50 p-4">
                  <ActivityIcon className="h-7 w-7 text-muted-foreground/50" />
                </div>
                <div className="space-y-1">
                  <p className="text-sm font-medium">No activity yet</p>
                  <p className="text-sm text-muted-foreground max-w-xs">
                    Activity from your team will appear here as things happen.
                  </p>
                </div>
              </div>
            )}

            {/* List */}
            {team && !actLoading && activities.length > 0 && (
              <Card className="border border-border/60 shadow-sm">
                <CardContent className="p-0">
                  <div className="divide-y divide-border/50">
                    {activities.map((activity) => (
                      <div key={activity.id} className="px-4">
                        <ActivityItem activity={activity} compact={false} />
                      </div>
                    ))}
                  </div>
                  {actHasMore && (
                    <div className="flex justify-center px-4 py-3 border-t border-border/50">
                      <Button
                        variant="ghost"
                        size="sm"
                        onClick={actLoadMore}
                        disabled={actFetching}
                        className="text-xs text-muted-foreground"
                      >
                        {actFetching ? (
                          <Loader2 className="h-3.5 w-3.5 animate-spin mr-1.5" />
                        ) : null}
                        Load more
                      </Button>
                    </div>
                  )}
                </CardContent>
              </Card>
            )}
          </TabsContent>

          {/* ---------------------------------------------------------------- */}
          {/* Tab 4: Settings (owners/admins only)                              */}
          {/* ---------------------------------------------------------------- */}
          {canManage && (
            <TabsContent value="settings" className="mt-6 space-y-6">
              {/* Team name edit */}
              {myRole === "owner" && team && (
                <Card className="border border-border/60 shadow-sm">
                  <CardHeader className="pb-3">
                    <CardTitle className="text-sm font-semibold flex items-center gap-2">
                      <SettingsIcon className="h-4 w-4 text-primary" />
                      Team Name
                    </CardTitle>
                  </CardHeader>
                  <CardContent>
                    <p className="text-xs text-muted-foreground mb-3">
                      Current name: <span className="font-medium text-foreground">{team.name}</span>
                    </p>
                    <div className="flex gap-2">
                      <Input
                        placeholder="New team name…"
                        value={teamNameInput}
                        onChange={(e) => setTeamNameInput(e.target.value)}
                        disabled={isSavingTeamName}
                        className="max-w-xs"
                      />
                      <Button
                        onClick={handleSaveTeamName}
                        disabled={isSavingTeamName || !teamNameInput.trim()}
                        size="sm"
                        className="gap-1.5"
                      >
                        {isSavingTeamName ? (
                          <Loader2 className="h-4 w-4 animate-spin" />
                        ) : null}
                        Save
                      </Button>
                    </div>
                  </CardContent>
                </Card>
              )}

              {/* Pipeline + Scorecard + Automations */}
              <PipelineStageManager teamId={team?.id ?? ""} />
              <ScorecardTemplateManager teamId={team?.id ?? ""} />
              <StageAutomationManager teamId={team?.id ?? ""} />
            </TabsContent>
          )}
        </Tabs>

        {/* ── Controlled dialogs (outside Tabs to avoid unmount issues) ── */}

        {/* Leave team confirmation */}
        <AlertDialog open={leaveDialogOpen} onOpenChange={setLeaveDialogOpen}>
          <AlertDialogContent>
            <AlertDialogHeader>
              <AlertDialogTitle>Leave team?</AlertDialogTitle>
              <AlertDialogDescription>
                You will immediately lose access to this team and all its data. This action cannot be undone.
              </AlertDialogDescription>
            </AlertDialogHeader>
            <AlertDialogFooter>
              <AlertDialogCancel disabled={isLeavingTeam}>Cancel</AlertDialogCancel>
              <AlertDialogAction
                onClick={handleLeaveTeam}
                disabled={isLeavingTeam}
                className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
              >
                {isLeavingTeam ? <Loader2 className="h-4 w-4 animate-spin mr-1.5" /> : null}
                Leave team
              </AlertDialogAction>
            </AlertDialogFooter>
          </AlertDialogContent>
        </AlertDialog>

        {/* Revoke invitation confirmation */}
        <AlertDialog
          open={!!revokeDialogTarget}
          onOpenChange={(open) => { if (!open) setRevokeDialogTarget(null); }}
        >
          <AlertDialogContent>
            <AlertDialogHeader>
              <AlertDialogTitle>Revoke invitation?</AlertDialogTitle>
              <AlertDialogDescription>
                The invitation for{" "}
                <span className="font-medium text-foreground">{revokeDialogTarget?.email}</span>{" "}
                will be cancelled and they will no longer be able to join the team.
              </AlertDialogDescription>
            </AlertDialogHeader>
            <AlertDialogFooter>
              <AlertDialogCancel>Cancel</AlertDialogCancel>
              <AlertDialogAction
                onClick={handleConfirmRevoke}
                className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
              >
                Revoke
              </AlertDialogAction>
            </AlertDialogFooter>
          </AlertDialogContent>
        </AlertDialog>

        {/* Remove member confirmation */}
        <AlertDialog
          open={!!removeDialogTarget}
          onOpenChange={(open) => { if (!open) setRemoveDialogTarget(null); }}
        >
          <AlertDialogContent>
            <AlertDialogHeader>
              <AlertDialogTitle>Remove member?</AlertDialogTitle>
              <AlertDialogDescription>
                <span className="font-medium text-foreground">
                  {removeDialogTarget?.email ?? "This member"}
                </span>{" "}
                will lose access to the team immediately. This action cannot be undone.
              </AlertDialogDescription>
            </AlertDialogHeader>
            <AlertDialogFooter>
              <AlertDialogCancel>Cancel</AlertDialogCancel>
              <AlertDialogAction
                onClick={() => {
                  if (removeDialogTarget) {
                    handleRemoveMember(removeDialogTarget);
                    setRemoveDialogTarget(null);
                  }
                }}
                className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
              >
                Remove
              </AlertDialogAction>
            </AlertDialogFooter>
          </AlertDialogContent>
        </AlertDialog>
      </main>
    </SidebarLayout>
  );
};

export default Team;
