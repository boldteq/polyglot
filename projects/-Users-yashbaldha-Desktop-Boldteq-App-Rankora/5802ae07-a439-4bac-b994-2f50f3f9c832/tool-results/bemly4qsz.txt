// ── Main page ────────────────────────────────────────────────────────────────

const Billing = () => {
  const { user, profile, refreshProfile } = useAuth();

  const [annualToggle, setAnnualToggle] = useState(false);
  const [buying, setBuying] = useState<string | null>(null);
  const [cancelling, setCancelling] = useState(false);
  const [downgrading, setDowngrading] = useState(false);
  const [confirmAction, setConfirmAction] = useState<{
    type: "upgrade" | "downgrade" | "cancel_subscription";
    planId?: string;
    planName?: string;
    price?: number;
  } | null>(null);

  const currentPlan = profile?.plan ?? "free";
  const isPaid = currentPlan !== "free";
  const subscriptionEndsAt = (profile as any)?.subscription_ends_at
    ? new Date((profile as any).subscription_ends_at)
    : null;
  const isExpired = subscriptionEndsAt ? subscriptionEndsAt < new Date() : false;
  const hasActiveSubscription = !!(profile as any)?.dodo_subscription_id;

  // Credits
  const subCredits = profile?.subscription_credits ?? 0;
  const bonusCredits = profile?.bonus_credits ?? 0;
  const legacyCredits = profile?.credits ?? 0;
  const splitTotal = subCredits + bonusCredits;
  const totalCredits = splitTotal > 0 ? splitTotal : legacyCredits;

  const { data: plans } = useQuery({
    queryKey: ["subscription_plans_active"],
    queryFn: async () => {
      const { data, error } = await supabase
        .from("plans")
        .select("*")
        .eq("active", true)
        .order("price");
      if (error) throw error;
      return data;
    },
  });

  // Filter by billing type when toggle changes
  const billingType = annualToggle ? "yearly" : "monthly";
  const allBillingTypes = plans?.map((p: any) => p.billing_type) ?? [];
  const hasAnnual = allBillingTypes.includes("yearly");
  const filteredPlans = hasAnnual
    ? plans?.filter((p: any) => p.billing_type === billingType)
    : plans;

  const popularIdx =
    filteredPlans && filteredPlans.length > 1
      ? Math.floor((filteredPlans.length - 1) / 2)
      : -1;

  const handleSubscribe = async (planId: string) => {
    if (!user) return;
    setBuying(planId);
    try {
      const { data, error } = await supabase.functions.invoke("create-dodo-checkout", {
        body: {
          plan_id: planId,
          return_url: `${window.location.origin}/payment/callback`,
        },
      });
      if (error) throw error;
      if (!data?.checkout_url) {
        toast.error(data?.error || "Failed to create checkout session");
        return;
      }
      window.location.href = data.checkout_url;
    } catch (err: unknown) {
      toast.error(err instanceof Error ? err.message : "Checkout failed");
    } finally {
      setBuying(null);
    }
  };

  const handleCancelSubscription = async () => {
    setCancelling(true);
    try {
      const { error } = await supabase.functions.invoke("cancel-dodo-subscription", {});
      if (error) throw error;
      toast.success("Auto-renewal cancelled. Your plan stays active until the billing period ends.");
      await refreshProfile();
      setConfirmAction(null);
    } catch (err: unknown) {
      toast.error("Failed to cancel: " + (err instanceof Error ? err.message : "Unknown error"));
    } finally {
      setCancelling(false);
    }
  };

  const handleDowngradeToFree = async () => {
    setDowngrading(true);
    try {
      if (hasActiveSubscription) {
        const { data, error: cancelError } = await supabase.functions.invoke(
          "cancel-dodo-subscription",
          {}
        );
        if (cancelError || data?.error) {
          toast.error(
            "Could not cancel subscription with payment provider. Please try again or contact support."
          );
          setDowngrading(false);
          return;
        }
      }
      const { error } = await supabase
        .from("profiles")
        .update({
          plan: "free",
          subscription_ends_at: null,
          dodo_subscription_id: null,
        } as any)
        .eq("user_id", user?.id);
      if (error) throw error;
      toast.success("Switched to Free plan. Your remaining credits are preserved.");
      await refreshProfile();
      setConfirmAction(null);
    } catch (err: unknown) {
      toast.error(
        "Failed to downgrade: " + (err instanceof Error ? err.message : "Unknown error")
      );
    } finally {
      setDowngrading(false);
    }
  };

  const handleConfirmedAction = () => {
    if (!confirmAction) return;
    if (confirmAction.type === "upgrade" && confirmAction.planId) {
      handleSubscribe(confirmAction.planId);
      setConfirmAction(null);
    } else if (confirmAction.type === "downgrade") {
      handleDowngradeToFree();
    } else if (confirmAction.type === "cancel_subscription") {
      handleCancelSubscription();
    }
  };

  return (
    <SidebarLayout>
      <main className="w-full px-6 sm:px-8 lg:px-10 py-8">
        {/* Page header */}
        <div className="mb-8">
          <h1 className="text-2xl font-semibold text-foreground tracking-tight">Billing & Plan</h1>
          <p className="text-sm text-muted-foreground mt-1">
            Manage your subscription and view usage history.
          </p>
        </div>

        <div className="space-y-8">
          {/* ── Current plan summary card ─────────────────────────────── */}
          <Card>
            <div className="px-6 pt-5 pb-5 flex items-start justify-between gap-4 flex-wrap">
              <div className="flex items-center gap-3">
                <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-primary/10 shrink-0">
                  {isPaid ? (
                    <Crown className="h-5 w-5 text-primary" />
                  ) : (
                    <Zap className="h-5 w-5 text-muted-foreground" />
                  )}
                </div>
                <div>
                  <div className="flex items-center gap-2 flex-wrap">
                    <Badge
                      variant={isPaid ? "default" : "secondary"}
                      className="gap-1.5 capitalize text-sm px-2.5 py-0.5"
                    >
                      {isPaid ? <Crown className="h-3.5 w-3.5" /> : <Zap className="h-3.5 w-3.5" />}
                      {isPaid ? currentPlan : "Free"}
                    </Badge>
                    {subscriptionEndsAt && !isExpired && (
                      <Badge variant="secondary" className="gap-1 text-xs">
                        <CalendarClock className="h-3 w-3" />
                        {hasActiveSubscription
                          ? `Renews ${format(subscriptionEndsAt, "MMM d, yyyy")}`
                          : `Active until ${format(subscriptionEndsAt, "MMM d, yyyy")}`}
                      </Badge>
                    )}
                    {isExpired && (
                      <Badge variant="destructive" className="text-xs">Expired</Badge>
                    )}
                  </div>
                  <p className="text-sm text-muted-foreground mt-1">
                    {totalCredits.toLocaleString()} credits remaining
                    {subCredits > 0 && bonusCredits > 0 && (
                      <span className="text-xs ml-1">
                        ({subCredits.toLocaleString()} monthly + {bonusCredits.toLocaleString()} bonus)
                      </span>
                    )}
                  </p>
                </div>
              </div>

              <div className="flex items-center gap-2 shrink-0">
                {isPaid && hasActiveSubscription && !isExpired && (
                  <Button
                    variant="outline"
                    size="sm"
                    className="h-8 text-xs gap-1.5 border-destructive/30 text-destructive hover:bg-destructive/10 hover:text-destructive"
                    onClick={() => setConfirmAction({ type: "cancel_subscription" })}
                    disabled={cancelling}
                  >
                    {cancelling ? (
                      <Loader2 className="h-3 w-3 animate-spin" />
                    ) : (
                      <XCircle className="h-3 w-3" />
                    )}
                    Cancel renewal
                  </Button>
                )}
                {isPaid && (
                  <Button
                    variant="outline"
                    size="sm"
                    className="h-8 text-xs gap-1.5 border-destructive/30 text-destructive hover:bg-destructive/10 hover:text-destructive"
                    onClick={() => setConfirmAction({ type: "downgrade" })}
                    disabled={downgrading}
                  >
                    {downgrading ? (
                      <Loader2 className="h-3 w-3 animate-spin" />
                    ) : (
                      <RefreshCw className="h-3 w-3" />
                    )}
                    Downgrade to Free
                  </Button>
                )}
              </div>
            </div>
          </Card>

          {/* ── Plan cards section ────────────────────────────────────── */}
          <section>
            <div className="flex items-center justify-between mb-5 flex-wrap gap-3">
              <div>
                <h2 className="text-lg font-semibold text-foreground">Choose Your Plan</h2>
                <p className="text-sm text-muted-foreground">
                  {isPaid
                    ? "Switch plans at any time. Upgrades take effect immediately."
                    : "Unlock more features and credits with a subscription."}
                </p>
              </div>
              {hasAnnual && (
                <div className="flex items-center gap-2">
                  <Label htmlFor="annual-toggle" className="text-sm text-muted-foreground">Monthly</Label>
                  <Switch
                    id="annual-toggle"
                    checked={annualToggle}
                    onCheckedChange={setAnnualToggle}
                  />
                  <Label htmlFor="annual-toggle" className="text-sm text-muted-foreground">
                    Annual
                    <Badge variant="secondary" className="ml-1.5 text-[10px] text-primary bg-primary/10 px-1 h-4">
                      Save 20%
                    </Badge>
                  </Label>
                </div>
              )}
            </div>

            <div
              className={`grid gap-4 ${
                filteredPlans && filteredPlans.length >= 3
                  ? "sm:grid-cols-2 lg:grid-cols-4"
                  : filteredPlans && filteredPlans.length === 2
                  ? "sm:grid-cols-3"
                  : "sm:grid-cols-2"
              }`}
            >
              {/* Free card */}
              <div
                className={`relative flex flex-col rounded-xl border p-5 transition-all ${
                  !isPaid
                    ? "border-primary/40 bg-primary/5 ring-1 ring-primary/20"
                    : "border-border bg-card hover:border-border/80"
                }`}
              >
                {!isPaid && (
                  <span className="absolute -top-2.5 left-3">
                    <Badge className="text-[10px] px-1.5 h-5 bg-primary text-primary-foreground gap-1">
                      <Check className="h-2.5 w-2.5" />
                      Current
                    </Badge>
                  </span>
                )}
                <div className="mb-3">
                  <p className="text-sm font-semibold text-foreground">Free</p>
                  <p className="text-[11px] text-muted-foreground mt-0.5">No commitment</p>
                </div>
                <div className="mb-4">
                  <div className="flex items-baseline gap-1">
                    <span className="text-2xl font-bold text-foreground">$0</span>
                    <span className="text-xs text-muted-foreground">/mo</span>
                  </div>
                </div>
                <ul className="space-y-1.5 flex-1 mb-5">
                  {["1 active job", "50 CVs per month", "Core ranking features", "Credits never expire"].map((f) => (
                    <li key={f} className="flex items-start gap-1.5 text-[11px] text-muted-foreground">
                      <Check className="h-3 w-3 text-primary mt-0.5 shrink-0" />
                      {f}
                    </li>
                  ))}
                </ul>
                {isPaid ? (
                  <Button
                    variant="outline"
                    size="sm"
                    className="w-full h-8 text-xs gap-1 border-destructive/30 text-destructive hover:bg-destructive/10 hover:text-destructive"
                    onClick={() => setConfirmAction({ type: "downgrade" })}
                    disabled={downgrading}
                  >
                    {downgrading ? <Loader2 className="h-3 w-3 animate-spin" /> : <RefreshCw className="h-3 w-3" />}
                    Switch to Free
                  </Button>
                ) : (
                  <div className="h-8 flex items-center justify-center">
                    <span className="text-[11px] text-muted-foreground">Your current plan</span>
                  </div>
                )}
              </div>

              {/* Subscription plan cards */}
              {filteredPlans?.map((plan: any, idx: number) => {
                const isCurrentPlan = currentPlan === plan.name;
                const isPopular = idx === popularIdx;
                const features: string[] = PLAN_FEATURES[plan.name] || PLAN_FEATURES["default"];

                return (
                  <div
                    key={plan.id}
                    className={`relative flex flex-col rounded-xl border p-5 transition-all ${
                      isCurrentPlan
                        ? "border-primary/40 bg-primary/5 ring-1 ring-primary/20"
                        : isPopular
                        ? "border-primary/25 bg-card shadow-sm"
                        : "border-border bg-card hover:border-border/80"
                    }`}
                  >
                    {isPopular && !isCurrentPlan && (
                      <span className="absolute -top-2.5 left-3">
                        <Badge className="text-[10px] px-1.5 h-5 bg-primary text-primary-foreground gap-1">
                          <Sparkles className="h-2.5 w-2.5" />
                          Popular
                        </Badge>
                      </span>
                    )}
                    {isCurrentPlan && (
                      <span className="absolute -top-2.5 left-3">
                        <Badge className="text-[10px] px-1.5 h-5 bg-primary text-primary-foreground gap-1">
                          <Check className="h-2.5 w-2.5" />
                          Current
                        </Badge>
                      </span>
                    )}

                    <div className="mb-3">
                      <p className="text-sm font-semibold text-foreground capitalize">{plan.name}</p>
                      <p className="text-[11px] text-muted-foreground mt-0.5 capitalize">
                        {plan.billing_type} • {plan.credits} credits
                      </p>
                    </div>

                    <div className="mb-4">
                      <div className="flex items-baseline gap-1">
                        <span className="text-2xl font-bold text-foreground">${plan.price}</span>
                        <span className="text-xs text-muted-foreground">
                          /{plan.billing_type === "yearly" ? "yr" : "mo"}
                        </span>
                      </div>
                    </div>

                    <ul className="space-y-1.5 flex-1 mb-5">
                      {features.map((f) => (
                        <li key={f} className="flex items-start gap-1.5 text-[11px] text-muted-foreground">
                          <Check className="h-3 w-3 text-primary mt-0.5 shrink-0" />
                          {f}
                        </li>
                      ))}
                    </ul>

                    {isCurrentPlan ? (
                      <div className="h-8 flex items-center justify-center">
                        <span className="text-[11px] text-muted-foreground">Your current plan</span>
                      </div>
                    ) : (
                      <Button
                        size="sm"
                        className="w-full h-8 text-xs gap-1"
                        onClick={() =>
                          setConfirmAction({
                            type: "upgrade",
                            planId: plan.id,
                            planName: plan.name,
                            price: plan.price,
                          })
                        }
                        disabled={buying === plan.id}
                      >
                        {buying === plan.id ? (
                          <Loader2 className="h-3.5 w-3.5 animate-spin" />
                        ) : (
                          <>
                            Subscribe
                            <ArrowRight className="h-3 w-3" />
                          </>
                        )}
                      </Button>
                    )}
                  </div>
                );
              })}

              {(!plans || plans.length === 0) && (
                <div className="col-span-full text-center py-8 text-sm text-muted-foreground">
                  No subscription plans available right now.
                </div>
              )}
            </div>
          </section>

          {/* ── Feature comparison table ──────────────────────────────── */}
          <section>
            <h2 className="text-lg font-semibold text-foreground mb-1">Compare All Features</h2>
            <p className="text-sm text-muted-foreground mb-4">Everything included in each plan.</p>

            <Card>
              <CardContent className="p-0 overflow-x-auto">
                <table className="w-full text-left">
                  <thead>
                    <tr className="border-b border-border">
                      <th className="px-6 py-3 text-xs font-medium text-muted-foreground w-[45%]">Feature</th>
                      <th className="px-4 py-3 text-xs font-medium text-muted-foreground text-center">Free</th>
                      <th className="px-4 py-3 text-xs font-medium text-muted-foreground text-center">Starter</th>
                      <th className="px-4 py-3 text-xs font-medium text-primary text-center">Growth</th>
                      <th className="px-4 py-3 text-xs font-medium text-muted-foreground text-center">Agency</th>
                    </tr>
                  </thead>
                  <tbody>
                    {FEATURE_COMPARISON.map((cat, ci) => (
                      <React.Fragment key={ci}>
                        <tr>
                          <td
                            colSpan={5}
                            className="px-6 pt-5 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/60"
                          >
                            {cat.category}
                          </td>
                        </tr>
                        {cat.features.map((f, fi) => (
                          <tr key={fi} className="border-b border-border/40 last:border-0 hover:bg-muted/30 transition-colors">
                            <td className="px-6 py-2.5 text-sm text-foreground">{f.name}</td>
                            {([f.free, f.starter, f.growth, f.agency] as FVal[]).map((v, vi) => (
                              <td key={vi} className="px-4 py-2.5 text-center">
                                {v === true ? (
                                  <CheckCircle2 className="h-4 w-4 text-emerald-500 mx-auto" />
                                ) : v === false ? (
                                  <span className="text-muted-foreground/30 text-lg leading-none">—</span>
                                ) : (
                                  <span className="text-xs font-semibold text-foreground">{v}</span>
                                )}
                              </td>
                            ))}
                          </tr>
                        ))}
                      </React.Fragment>
                    ))}
                  </tbody>
                </table>
              </CardContent>
            </Card>

            <p className="text-[11px] text-muted-foreground text-center mt-3">
              All plans include unlimited users. Subscriptions auto-renew. Cancel anytime.
            </p>
          </section>

          {/* ── Payment + Usage history ───────────────────────────────── */}
          <PaymentHistoryCard />
          <UsageHistoryCard />
        </div>
      </main>

      {/* ── Confirm dialog ───────────────────────────────────────────── */}
      <AlertDialog
        open={confirmAction !== null}
        onOpenChange={(o) => !o && setConfirmAction(null)}
      >
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>
              {confirmAction?.type === "upgrade"
                ? `Subscribe to ${confirmAction.planName}?`
                : confirmAction?.type === "cancel_subscription"
                ? "Cancel Auto-Renewal?"
                : "Switch to Free?"}
            </AlertDialogTitle>
            <AlertDialogDescription asChild>
              <div>
                {confirmAction?.type === "upgrade" ? (
                  <span>
                    You'll be redirected to our secure checkout to complete payment of{" "}
                    <strong className="text-foreground font-semibold">${confirmAction.price}</strong>.
                    Your subscription will auto-renew each billing period.
                  </span>
                ) : confirmAction?.type === "cancel_subscription" ? (
                  <span>
                    Your plan stays active until the end of your current billing period. After
                    that you'll move to Free. Your remaining credits are preserved.
                  </span>
                ) : (
                  <span>
                    You'll switch to the Free plan. Your remaining credits are preserved.
                    {hasActiveSubscription && " Your active subscription will also be cancelled."}
                  </span>
                )}
              </div>
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Keep Current Plan</AlertDialogCancel>
            <AlertDialogAction
              onClick={handleConfirmedAction}
              disabled={downgrading || cancelling}
              className={
                confirmAction?.type !== "upgrade"
                  ? "bg-destructive text-destructive-foreground hover:bg-destructive/90"
                  : ""
              }
            >
              {(downgrading || cancelling) && (
                <Loader2 className="h-4 w-4 animate-spin mr-1.5" />
              )}
              {confirmAction?.type === "upgrade"
                ? "Proceed to Payment"
                : confirmAction?.type === "cancel_subscription"
                ? "Cancel Auto-Renewal"
                : "Confirm Downgrade"}
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </SidebarLayout>
  );
};

export default Billing;
