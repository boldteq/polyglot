function getStatusBadge(status: string) {
  if (status === "completed") {
    return (
      <Badge
        variant="outline"
        className="border-emerald-200/80 bg-emerald-50/50 text-emerald-700 text-xs font-medium"
      >
        Completed
      </Badge>
    );

      <Badge
        variant="outline"
        className="border-amber-200/80 bg-amber-50/50 text-amber-700 text-xs font-medium animate-pulse"

      </Badge>
    );

  return (
    <Badge variant="outline" className="text-xs">

/** Generate a stable teaser score 45–72 for an unauthenticated guest scan */
function generateTeaserScore(url: string): number {
  let hash = 0;
  for (let i = 0; i < url.length; i++) {
    hash = (hash * 31 + url.charCodeAt(i)) & 0xffffffff;
  }
  return 45 + (Math.abs(hash) % 28);
}

/** Basic URL validation — checks structure, not reachability */
function isValidUrl(str: string): boolean {
  try {
    const url = new URL(str.startsWith("http") ? str : `https://${str}`);
    return url.protocol === "https:" || url.protocol === "http:";
  } catch {
    return false;
  }
}

/** Normalize URL: prepend https:// if missing */
function normalizeUrl(str: string): string {
  const trimmed = str.trim();
  if (!trimmed) return trimmed;
  if (/^https?:\/\//i.test(trimmed)) return trimmed;
  return `https://${trimmed}`;
}

// ── Map scan status to step index ───────────────────────────────

function statusToStep(status: string): number {
  switch (status) {
    case "pending": return 0;
    case "analyzing": return 3;
    case "completed": return SCAN_STEPS.length;
    case "failed": return -1;
    default: return 0;
  }
}

  const scansUsed = profile?.scans_used ?? 0;
  const scanLimit = profile?.scan_limit ?? 3;
  const scansRemaining = scanLimit - scansUsed;
  const atLimit = scansRemaining <= 0;

  // Poll the scan row while it's pending/analyzing
  const { data: pollingScan } = useScan(activeScanId ?? undefined, {
    refetchInterval: activeScanId ? 2000 : false,
  });

  const isAnalyzing = activeScanId !== null || guestAnimating;
  const normalizedUrl = normalizeUrl(rawUrl);

  const totalSteps = SCAN_STEPS.length;
  const currentStep = isAnalyzing
    ? pollingScan
      ? statusToStep(pollingScan.status)
      : simulatedStep
    : -1;
  const progressPercent = currentStep < 0 ? 0 : Math.round(((currentStep + 1) / totalSteps) * 100);

  // When polling finds a completed/failed scan, navigate or show error
  useEffect(() => {
    if (!pollingScan) return;
    if (pollingScan.status === "completed") {
      setTimeout(() => navigate(`/reports/${pollingScan.id}`), 600);
    } else if (pollingScan.status === "failed") {
      setActiveScanId(null);
      setSimulatedStep(-1);
      setScanFailed(true);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [pollingScan?.status]);

  // Simulate step progression while waiting for the edge function (authenticated)
  useEffect(() => {
    if (!isAnalyzing || !activeScanId) return;
    if (pollingScan?.status === "completed" || pollingScan?.status === "failed") return;
    if (simulatedStep >= totalSteps - 1) return;

    const durations = [900, 1400, 1200, 1100, 1300, 1000, 1600];
    const timer = setTimeout(
      () => setSimulatedStep((prev) => prev + 1),
      durations[simulatedStep] ?? 1200
    );
    return () => clearTimeout(timer);
    // totalSteps is derived from a constant — safe to omit
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [isAnalyzing, activeScanId, simulatedStep, pollingScan?.status]);

  // Guest simulated step progression — no API call, just animation
  useEffect(() => {
    if (!guestAnimating || guestAnimationDone) return;
    if (simulatedStep >= totalSteps - 1) {
      // Animation finished — show gate wall
      const timer = setTimeout(() => {
        setGuestAnimationDone(true);
        setGuestAnimating(false);
      }, 800);
      return () => clearTimeout(timer);
    }

    const durations = [900, 1400, 1200, 1100, 1300, 1000, 1600];
    const timer = setTimeout(
      () => setSimulatedStep((prev) => prev + 1),
      durations[simulatedStep] ?? 1200
    );
    return () => clearTimeout(timer);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [guestAnimating, guestAnimationDone, simulatedStep]);

  const startAnalysis = useCallback(() => {
    setUrlError(null);
    if (!rawUrl.trim()) {
      setUrlError("Please enter a URL.");
      return;
    }
    if (!isValidUrl(normalizedUrl)) {
      setUrlError("Enter a valid URL like https://example.com/page");
      return;
    }

    // Unauthenticated users: if free scan disabled, redirect to signup immediately
    if (!isAuthenticated) {
      if (!freeScanEnabled) {
        navigate("/signup");
        return;
      }
      setGuestAnimating(true);
      setGuestAnimationDone(false);
      setGuestTeaserScore(generateTeaserScore(normalizedUrl));
      setSimulatedStep(0);
      return;
    }

    if (atLimit) return;

    setSimulatedStep(0);

    createScan.mutate(
      { url: normalizedUrl, page_type: pageType },
      {
        onSuccess: (scan) => {
          setActiveScanId(scan.id);
        },
        onError: (err: unknown) => {
          setSimulatedStep(-1);
          const message = err instanceof Error ? err.message : "Failed to start scan";
          toast.error(message);
        },
      }
    );
  }, [rawUrl, normalizedUrl, pageType, isAuthenticated, atLimit, createScan]);

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    startAnalysis();
  };

  // ── Guest Gate Wall ─────────────────────────────────────────
  if (guestAnimationDone) {
    return (
      <PageTransition>
        <ScanGateWall teaserScore={guestTeaserScore} url={normalizedUrl} />
      </PageTransition>
    );
  }

  // ── Analysis Progress View ──────────────────────────────────
  if (isAnalyzing) {
    let hostname = rawUrl;
    try { hostname = new URL(normalizedUrl).hostname; } catch { /* keep raw */ }

    const displayStep = Math.max(0, currentStep);

    return (
      <PageTransition>
        <div className="mx-auto max-w-2xl space-y-6">
          <div>
            <h2 className="text-2xl font-bold tracking-tight text-foreground">Analyzing your page</h2>
            <p className="mt-1 text-sm text-muted-foreground">
              Scanning <span className="font-medium text-foreground">{hostname}</span> — this usually takes 20-40 seconds
            </p>
          </div>

          {/* Progress bar */}
          <div className="space-y-1.5">
            <div className="flex items-center justify-between text-sm">
              <span className="text-muted-foreground">
                Step {Math.min(displayStep + 1, totalSteps)} of {totalSteps}
              </span>
              <span className="tabular-nums font-medium text-foreground">{Math.min(progressPercent, 100)}%</span>
            </div>
            <Progress value={Math.min(progressPercent, 100)} className="h-2" />
          </div>

          {/* Browser frame */}
          <BrowserFrame url={normalizedUrl}>
            <div className="relative flex h-52 items-center justify-center bg-gradient-to-br from-slate-50 via-white to-slate-50">
              {/* Scanning animation */}
              <div className="absolute inset-x-0 top-0 h-1 overflow-hidden">
                <div className="h-full w-1/3 animate-[shimmer_1.5s_ease-in-out_infinite] bg-gradient-to-r from-transparent via-primary/30 to-transparent" />
              </div>
              <div className="text-center">
                <Loader2 className="mx-auto mb-3 h-8 w-8 animate-spin text-primary" />
                <p className="text-sm font-medium text-muted-foreground">
                  {displayStep < totalSteps ? SCAN_STEPS[displayStep]?.label : "Finalizing report"}...
                </p>
              </div>
            </div>
          </BrowserFrame>

          {/* Timeline */}
          <Card className="border-border/50 shadow-soft">
            <CardContent className="p-6">
              <ScanProgressTimeline steps={SCAN_STEPS} currentStep={displayStep} />
            </CardContent>
          </Card>
        </div>
      </PageTransition>
    );
  }

  // ── URL Input Form ──────────────────────────────────────────
  return (
    <PageTransition>
      <div className="mx-auto max-w-2xl space-y-8">
        <div className="text-center">
          <h2 className="text-2xl font-bold tracking-tight text-foreground">Analyze your page</h2>
          <p className="mt-2 text-muted-foreground">
            Enter any URL to get a detailed CRO audit with actionable recommendations
          </p>
        </div>

        {/* URL Input Card */}
        <Card className="border-border/50 shadow-soft">
          <CardContent className="p-8">
            <form onSubmit={handleSubmit} className="space-y-4">
              <div>
                <label htmlFor={inputId} className="mb-1.5 block text-sm font-medium text-foreground">
                  Page URL
                </label>
                <div className="relative">
                  <Globe className="absolute left-4 top-1/2 h-4.5 w-4.5 -translate-y-1/2 text-muted-foreground" />
                  <Input
                    id={inputId}
                    type="text"
                    placeholder="example.com/landing-page"
                    value={rawUrl}
                    onChange={(e) => { setRawUrl(e.target.value); setUrlError(null); }}
                    aria-invalid={urlError ? true : undefined}
                    aria-describedby={urlError ? urlErrorId : undefined}
                    className={cn(
                      "h-14 rounded-xl border-border/50 bg-card pl-11 pr-5 text-base placeholder:text-muted-foreground focus-visible:border-primary focus-visible:ring-primary/20",
                      urlError && "border-red-300 focus-visible:border-red-400 focus-visible:ring-red-400/20"
                    )}
                  />
                </div>
                {urlError && (
                  <p id={urlErrorId} role="alert" className="mt-1.5 flex items-center gap-1 text-sm text-red-600">
                    <AlertCircle className="h-3.5 w-3.5" aria-hidden="true" />
                    {urlError}
                  </p>
                )}
              </div>

              <div>
                <label className="mb-1.5 block text-sm font-medium text-foreground">Page type</label>
                <Select value={pageType} onValueChange={(v) => setPageType(v as PageType)}>
                  <SelectTrigger className="h-12 rounded-xl border-border/50 bg-card">
                    <SelectValue placeholder="Select page type" />
                  </SelectTrigger>
                  <SelectContent>
                    {PAGE_TYPES.map((type) => (
                      <SelectItem key={type.value} value={type.value}>
                        {type.label}
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
                <p className="mt-1 text-xs text-muted-foreground">
                  Page type affects how we weight each CRO pillar in the overall score
                </p>
              </div>

              <Button
                type="submit"
                className="h-14 w-full rounded-xl text-base font-semibold"
                disabled={!rawUrl.trim() || (isAuthenticated && (atLimit || createScan.isPending)) || guestAnimating}
              >
                {isAuthenticated && atLimit ? (
                  "Scan limit reached — Upgrade to continue"
                ) : isAuthenticated && createScan.isPending ? (
                  <>
                    <Loader2 className="mr-2 h-5 w-5 animate-spin" />
                    Starting scan...
                  </>
                ) : !isAuthenticated ? (
                  <>
                    <ScanLine className="mr-2 h-5 w-5" />
                    Analyze Free — No Sign Up Needed
                  </>
                ) : (
