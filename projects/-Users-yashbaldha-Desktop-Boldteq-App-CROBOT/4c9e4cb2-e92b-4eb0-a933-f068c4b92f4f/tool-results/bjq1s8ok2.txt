export const analyticsKeys = {
  all: ["analytics"] as const,
  byUrl: (url: string) => ["analytics", url] as const,
};

// ── Edge function response shapes ─────────────────────────────

interface AnalyticsNotConnected {
  connected: false;
}

interface AnalyticsSetupRequired {
  connected: true;
  setup_required: true;
}

interface AnalyticsError {
  connected: true;
  error: string;
}

interface AnalyticsSuccess {
  connected: true;
  metrics: AnalyticsMetrics;
}

type FetchAnalyticsResponse =
  | AnalyticsNotConnected
  | AnalyticsSetupRequired
  | AnalyticsError
  | AnalyticsSuccess;

// ── useGa4Connection ───────────────────────────────────────────
// Returns whether the current user has an active GA4 integration.

export function useGa4Connection(): {
  isConnected: boolean;
  isLoading: boolean;
} {
  const { data: integrations = [], isLoading } = useIntegrations();

  const isConnected = integrations.some(
    (i) => i.provider === "google_analytics" && i.is_active,
  );

  return { isConnected, isLoading };
}

// ── useAnalyticsData ───────────────────────────────────────────
// Fetches GA4 analytics for the given URL.
// Returns null when GA4 is not connected or when there is no data.

export function useAnalyticsData(url: string): {
  data: AnalyticsMetrics | null;
  isLoading: boolean;
  isError: boolean;
  setupRequired: boolean;
  errorMessage: string | null;
  notConnected: boolean;
} {
  const { isConnected } = useGa4Connection();

  const query = useQuery({
    queryKey: analyticsKeys.byUrl(url),
    queryFn: async (): Promise<FetchAnalyticsResponse> => {
      const { data, error } = await supabase.functions.invoke<FetchAnalyticsResponse>(
        "fetch-analytics",
        { body: { url } },
      );

      if (error) throw new Error(error.message);
      if (!data) throw new Error("Empty response from fetch-analytics");
      return data;
    },
    enabled: isConnected && url.length > 0,
    staleTime: 1000 * 60 * 5, // 5 minutes — GA4 data doesn't change per second
    retry: 1,
  });

  const response = query.data;

  if (!response) {
    return {
      data: null,
      isLoading: query.isLoading,
      isError: query.isError,
      setupRequired: false,
      errorMessage: null,
      notConnected: !isConnected,
    };
  }

  if (!response.connected) {
    return {
      data: null,
      isLoading: false,
      isError: false,
      setupRequired: false,
      errorMessage: null,
      notConnected: true,
    };
  }

  if ("setup_required" in response && response.setup_required) {
    return {
      data: null,
      isLoading: false,
      isError: false,
      setupRequired: true,
      errorMessage: null,
      notConnected: false,
    };
  }

  if ("error" in response && response.error) {
    return {
      data: null,
      isLoading: false,
      isError: true,
      setupRequired: false,
      errorMessage: response.error,
      notConnected: false,
    };
  }

  if ("metrics" in response && response.metrics) {
    return {
      data: response.metrics,
      isLoading: false,
      isError: false,
      setupRequired: false,
      errorMessage: null,
      notConnected: false,
    };
  }

  return {
    data: null,
    isLoading: false,
    isError: false,
    setupRequired: false,
    errorMessage: null,
    notConnected: false,
  };
}

// ── SHA-256 helper (same algorithm as edge function) ──────────

async function hashApiKey(key: string): Promise<string> {
  const encoded = new TextEncoder().encode(key);
  const hashBuffer = await crypto.subtle.digest("SHA-256", encoded);
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  return hashArray.map((b) => b.toString(16).padStart(2, "0")).join("");
}

// ── useApiKeys ────────────────────────────────────────────────
// Fetches active (non-revoked) API keys for the current user.

export function useApiKeys() {
  const { user } = useAuth();

  return useQuery<ApiKey[]>({
    queryKey: ["api-keys", user?.id],
    queryFn: async () => {
      if (!user) return [];

      const { data, error } = await supabase
        .from("api_keys")
        .select(
          "id, user_id, team_id, name, key_prefix, permissions, rate_limit, last_used_at, created_at, revoked_at",
        )
        .eq("user_id", user.id)
        .is("revoked_at", null)
        .order("created_at", { ascending: false });

      if (error) throw new Error(error.message);
      return (data ?? []) as ApiKey[];
    },
    enabled: !!user,
  });
}

// ── useCreateApiKey ───────────────────────────────────────────
// Generates a new API key, hashes it, stores the hash, and returns
// the FULL key (shown only once — never stored in plaintext).

export interface CreatedApiKey {
  fullKey: string;
  row: ApiKey;
}

export function useCreateApiKey() {
  const { user } = useAuth();
  const queryClient = useQueryClient();

  return useMutation<CreatedApiKey, Error, { name: string }>({
    mutationFn: async ({ name }) => {
      if (!user) throw new Error("Not authenticated");

      // Generate key: cs_live_ + 32 hex chars from UUID (no dashes = 32 chars)
      const rawUuid = crypto.randomUUID().replace(/-/g, "");
      const fullKey = `cs_live_${rawUuid}`;
      const keyPrefix = fullKey.slice(0, 12);

      const keyHash = await hashApiKey(fullKey);

      const { data, error } = await supabase
        .from("api_keys")
        .insert({
          user_id: user.id,
          name,
          key_hash: keyHash,
          key_prefix: keyPrefix,
          permissions: ["read", "write"],
        })
        .select(
          "id, user_id, team_id, name, key_prefix, permissions, rate_limit, last_used_at, created_at, revoked_at",
        )
        .single();

      if (error) throw new Error(error.message);
      if (!data) throw new Error("Failed to create API key.");

      return { fullKey, row: data as ApiKey };
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["api-keys"] });
    },
    onError: (error) => {
      toast.error(error.message || "Failed to create API key.");
    },
  });
}

// Sets revoked_at = NOW() on the given key row.

export function useRevokeApiKey() {

    },
  });
}

// ── URL validation helpers ───────────────────────────────────────

/** Normalize a single URL string — prepend https:// if missing */
export function normalizeUrl(str: string): string {
  const trimmed = str.trim();
  if (!trimmed) return trimmed;
  if (/^https?:\/\//i.test(trimmed)) return trimmed;
  return `https://${trimmed}`;
}

/** Returns true if the normalized string is a valid http/https URL */
export function isValidUrl(str: string): boolean {
  try {
    const url = new URL(str);
    return url.protocol === "https:" || url.protocol === "http:";
  } catch {
    return false;
  }
}

export interface ParsedUrlLine {
  lineNumber: number;
  raw: string;
  normalized: string;
  valid: boolean;
}

/** Parse a textarea value into per-line URL results */
export function parseUrlLines(raw: string): ParsedUrlLine[] {
  return raw
    .split("\n")
    .map((line, idx) => {
      const normalized = normalizeUrl(line);
      return {
        lineNumber: idx + 1,
        raw: line,
        normalized,
        valid: normalized.length > 0 && isValidUrl(normalized),
      };
    })
    .filter((l) => l.raw.trim().length > 0);
}

      return job.status === "pending" || job.status === "processing" ? 2000 : false;

    staleTime: 0,
  });
}

// ── useCreateBulkScan ────────────────────────────────────────────

interface CreateBulkScanInput {
  urls: string[];
  page_type: string;
  team_id?: string | null;
}

export function useCreateBulkScan() {
  const { user } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ urls, page_type, team_id }: CreateBulkScanInput): Promise<BulkScanJob> => {
      if (!user?.id) throw new Error("Not authenticated");

      // Validate all URLs before inserting
      const invalid = urls.filter((u) => !isValidUrl(normalizeUrl(u)));
      if (invalid.length > 0) {
        throw new Error(`${invalid.length} invalid URL(s) detected. Please fix them before submitting.`);
      }

      const normalizedUrls = urls.map(normalizeUrl);

      // Insert the job row
      const { data, error } = await supabase
        .from("bulk_scan_jobs")
        .insert({
          user_id: user.id,
          team_id: team_id ?? null,
          status: "pending" as const,
          urls: normalizedUrls,
          page_type,
          total: normalizedUrls.length,
          completed: 0,
          failed: 0,
        })
        .select("*")
        .single();

      if (error) throw new Error(error.message);

      const job = rowToJob(data as BulkScanJobRow);

      // Fire the edge function — non-blocking
      supabase.functions
        .invoke("bulk-scan", {
          body: { job_id: job.id },
        })
        .catch((err: unknown) => {
          console.error("Bulk scan edge function invocation failed:", err);
          toast.error("Bulk scan engine failed to start. Please try again.", { duration: 6000 });
        });

      return job;
    },
    onSuccess: (data) => {
      if (!user?.id) return;
      // Prepend to list cache
      queryClient.setQueryData<BulkScanJob[]>(bulkScanKeys.list(user.id), (prev) =>
        prev ? [data, ...prev] : [data]
      );
      // Prime detail cache
      queryClient.setQueryData(bulkScanKeys.detail(data.id), data);
    },
  });
}

        .update({ status: "failed" })

        .eq("user_id", user.id);

      // Update detail cache

      );
      // Update in list cache
      queryClient.setQueryData<BulkScanJob[]>(bulkScanKeys.list(user.id), (prev) =>
        prev ? prev.map((j) => (j.id === jobId ? { ...j, status: "failed" } : j)) : prev
      );
      toast.success("Bulk scan cancelled.");
    },
    onError: (err: unknown) => {
      const message = err instanceof Error ? err.message : "Failed to cancel job";
      toast.error(message);
    },
  });
}

interface CreateGroupInput {
  name: string;
  primary_url: string;
  competitor_urls: string[];
}

export function useCreateCompetitorGroup() {
  const { user } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async (input: CreateGroupInput): Promise<CompetitorGroupWithUrls> => {

      // 1. Create the group
      const groupInsert: CompetitorGroupInsert = {

        .select("*");

        competitor_urls: urlData as CompetitorUrlRow[],

      queryClient.setQueryData<CompetitorGroupWithUrls[]>(

// ── useScanCompetitor ───────────────────────────────────────────

interface ScanCompetitorInput {
  competitorUrlId: string;
  url: string;
  groupId: string;
}

export function useScanCompetitor() {
  const { user } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async (input: ScanCompetitorInput): Promise<void> => {
      if (!user?.id) throw new Error("Not authenticated");

      // Invoke the same edge function used for regular scans
      const { data, error } = await supabase.functions.invoke("analyze-url", {
        body: { url: input.url, page_type: "landing" },
      });

      if (error) throw new Error(error.message);

      const result = data as AnalyzeUrlResponse;
      const overallScore = result?.overall_score ?? null;
      const scanId = result?.scan_id ?? null;

      // Update the competitor_urls row with scan_id + latest_score
      const { error: updateError } = await supabase
        .from("competitor_urls")
        .update({
          scan_id: scanId,
          latest_score: overallScore,
          scanned_at: new Date().toISOString(),
        })
        .eq("id", input.competitorUrlId);

      if (updateError) throw new Error(updateError.message);
    },
    onSuccess: (_data, input) => {
      if (!user?.id) return;
      // Refetch the group list to reflect updated scores
      void queryClient.invalidateQueries({
        queryKey: competitorKeys.list(user.id),
      });
      toast.success(`Scan complete for ${tryHostname(input.url)}`);
    },
    onError: (error: unknown) => {
      const msg = error instanceof Error ? error.message : "Scan failed";
      toast.error(msg);
    },
  });
}

=== use-count-up ===
import { useState, useEffect, useRef } from "react";

/**
 * Animates a number from 0 to target value.
 * Used for KPI card numbers on the dashboard.
 */
export function useCountUp(target: number, duration: number = 1200): number {
  const [value, setValue] = useState(0);
  const startTime = useRef<number | null>(null);
  const rafId = useRef<number>(0);

  useEffect(() => {
    startTime.current = null;

    const animate = (timestamp: number) => {
      if (startTime.current === null) startTime.current = timestamp;
      const elapsed = timestamp - startTime.current;
      const progress = Math.min(elapsed / duration, 1);

      // Ease-out cubic for smooth deceleration
      const eased = 1 - Math.pow(1 - progress, 3);
      setValue(Math.round(eased * target));

      if (progress < 1) {
        rafId.current = requestAnimationFrame(animate);
      }
    };

    rafId.current = requestAnimationFrame(animate);

    return () => cancelAnimationFrame(rafId.current);
  }, [target, duration]);

  return value;
}

// ── Helpers ─────────────────────────────────────────────────────

function getWeekLabel(date: Date): string {
  // Show the Monday of each week as the label (e.g. "Jun 2")
  const day = date.getDay();
  const monday = new Date(date);
  monday.setDate(date.getDate() - (day === 0 ? 6 : day - 1));
  return monday.toLocaleDateString("en-US", { month: "short", day: "numeric" });
}

function getWeekLabelForIndex(weeksAgo: number): string {
  const d = new Date(Date.now() - weeksAgo * 7 * 24 * 60 * 60 * 1000);
  return getWeekLabel(d);
}

function getBucketFill(range: string): string {
  const fills: Record<string, string> = {
    "0-20": "#EF4444",
    "21-40": "#F87171",
    "41-60": "#F59E0B",
    "61-80": "#22C55E",
    "81-100": "#16A34A",
  };
  return fills[range] ?? "#94A3B8";
}

function scoreToRange(score: number): string {
  if (score <= 20) return "0-20";
  if (score <= 40) return "21-40";
  if (score <= 60) return "41-60";
  if (score <= 80) return "61-80";
  return "81-100";
}

function rowToScan(row: ScanRow): Scan {
  return {
    id: row.id,
    user_id: row.user_id,
    url: row.url,
    favicon_url: row.favicon_url,
    page_title: row.page_title,
    page_type: row.page_type,
    status: row.status,
    overall_score: row.overall_score,
    issues_found: row.issues_found,
    critical_issues: row.critical_issues,
    warnings: row.warnings,
    quick_wins: row.quick_wins,
    screenshot_url: row.screenshot_url,
    device: row.device,
    created_at: row.created_at,
    completed_at: row.completed_at,
  };
}

// ── useDashboardStats ───────────────────────────────────────────

export function useDashboardStats() {
  const { user } = useAuth();

  return useQuery({
    queryKey: ["dashboard-stats", user?.id ?? ""],
    queryFn: async (): Promise<DashboardData> => {
      if (!user?.id) return EMPTY_DASHBOARD;

      // Fetch all completed scans for the user
      const { data: scansData, error: scansError } = await supabase
        .from("scans")
        .select("*")
        .eq("user_id", user.id)
        .order("created_at", { ascending: false });

      if (scansError) throw new Error(scansError.message);

      const scans = (scansData as ScanRow[]).map(rowToScan);

      if (scans.length === 0) return EMPTY_DASHBOARD;

      const completedScans = scans.filter((s) => s.status === "completed" && s.overall_score !== null);
      const recentScans = scans.slice(0, 5);
      const latestScanId = completedScans[0]?.id ?? null;

      // ── Stats aggregation ─────────────────────────────────────

      const now = new Date();
      const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
      const lastMonthStart = new Date(now.getFullYear(), now.getMonth() - 1, 1);

      const scansThisMonth = completedScans.filter(
        (s) => new Date(s.created_at) >= monthStart
      );
      const scansLastMonth = completedScans.filter(
        (s) =>
          new Date(s.created_at) >= lastMonthStart &&
          new Date(s.created_at) < monthStart
      );

      const avgScore =
        completedScans.length > 0
          ? Math.round(
              completedScans.reduce((sum, s) => sum + (s.overall_score ?? 0), 0) /
                completedScans.length
            )
          : 0;

      const avgScoreThisMonth =
        scansThisMonth.length > 0
          ? Math.round(
              scansThisMonth.reduce((sum, s) => sum + (s.overall_score ?? 0), 0) /
                scansThisMonth.length
            )
          : 0;

      const avgScoreLastMonth =
        scansLastMonth.length > 0
          ? Math.round(
              scansLastMonth.reduce((sum, s) => sum + (s.overall_score ?? 0), 0) /
                scansLastMonth.length
            )
          : 0;

      const totalIssues = completedScans.reduce((sum, s) => sum + s.issues_found, 0);
      const totalCritical = completedScans.reduce((sum, s) => sum + s.critical_issues, 0);
      const totalQuickWins = completedScans.reduce((sum, s) => sum + s.quick_wins, 0);

      const scoreTrendDelta =
        avgScoreLastMonth > 0 ? avgScoreThisMonth - avgScoreLastMonth : 0;
      const scansTrendDelta =
        scansLastMonth.length > 0
          ? Math.round(
              ((scansThisMonth.length - scansLastMonth.length) /
                scansLastMonth.length) *
                100
            )
          : scansThisMonth.length > 0
          ? 100
          : 0;

      const stats: DashboardStats = {
        total_scans: scansThisMonth.length,
        avg_score: avgScore,
        total_issues: totalIssues,
        critical_issues: totalCritical,
        estimated_revenue_impact: Math.round(totalQuickWins * 125),
        score_trend: scoreTrendDelta,
        scans_trend: scansTrendDelta,
        quick_wins: totalQuickWins,
      };

      // ── Score Trend (last 4 weeks) ────────────────────────────

      const fourWeeksAgo = new Date(now.getTime() - 28 * 24 * 60 * 60 * 1000);
      const recentForTrend = completedScans.filter(
        (s) => new Date(s.created_at) >= fourWeeksAgo
      );

      const weekBuckets = new Map<number, number[]>(); // weekIndex → scores[]
      for (const scan of recentForTrend) {
        const scanDate = new Date(scan.created_at);
        const diffMs = now.getTime() - scanDate.getTime();
        const weekIndex = 3 - Math.min(3, Math.floor(diffMs / (1000 * 60 * 60 * 24 * 7)));
        const existing = weekBuckets.get(weekIndex) ?? [];
        if (scan.overall_score !== null) {
          existing.push(scan.overall_score);
          weekBuckets.set(weekIndex, existing);
        }
      }

      const scoreTrend: ScoreTrendPoint[] = [3, 2, 1, 0].map((weeksAgo, idx) => {
        const label = getWeekLabelForIndex(weeksAgo);
        const scores = weekBuckets.get(idx) ?? [];
        const avg =
          scores.length > 0
            ? Math.round(scores.reduce((s, v) => s + v, 0) / scores.length)
            : 0;
        return { week: label, score: avg };
      }).filter((p) => p.score > 0);

      // ── Score Distribution ────────────────────────────────────

      const RANGES = ["0-20", "21-40", "41-60", "61-80", "81-100"];
      const distributionMap = new Map<string, number>(RANGES.map((r) => [r, 0]));

      for (const scan of completedScans) {
        if (scan.overall_score !== null) {
          const bucket = scoreToRange(scan.overall_score);
          distributionMap.set(bucket, (distributionMap.get(bucket) ?? 0) + 1);
        }
      }

      const scoreDistribution: ScoreDistributionBucket[] = RANGES.map((range) => ({
        range,
        count: distributionMap.get(range) ?? 0,
        fill: getBucketFill(range),
      }));

      // ── Top Quick Wins from latest completed scan ─────────────

      let topQuickWins: (Finding & { pillar: string })[] = [];

      if (latestScanId) {
        const { data: resultsData } = await supabase
          .from("scan_results")
          .select("*")
          .eq("scan_id", latestScanId);

        if (resultsData && resultsData.length > 0) {
          const allFindings: (Finding & { pillar: string })[] = (
            resultsData as ScanResultRow[]
          ).flatMap((r) =>
            (r.findings ?? [])
              .filter((f: DbFinding) => f.severity !== "passed" && f.effort === "quick")
              .map((f: DbFinding) => ({
                id: f.id,
                title: f.title,
                severity: f.severity,
                current_state: f.current_state,
                recommendation: f.recommendation,
                ai_suggestion: f.ai_suggestion,
                is_premium: f.is_premium,
                effort: f.effort,
                estimated_lift: f.estimated_lift,
                category: f.category,
                pillar: r.pillar,
              }))
          );

          topQuickWins = allFindings
            .sort((a, b) => {
              const liftA = parseFloat(a.estimated_lift.match(/[\d.]+/)?.[0] ?? "0");
              const liftB = parseFloat(b.estimated_lift.match(/[\d.]+/)?.[0] ?? "0");
              return liftB - liftA;
            })
            .slice(0, 3);
        }
      }

      // ── New fields: best_score, scans_this_week ───────────────

      const bestScore =
        completedScans.length > 0
          ? Math.max(...completedScans.map((s) => s.overall_score ?? 0))
          : 0;

      const oneWeekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
      const scansThisWeek = scans.filter(
        (s) => new Date(s.created_at) >= oneWeekAgo
      ).length;

      // ── score_trend_data: weekly averages for last 8 weeks ────

      const eightWeeksAgo = new Date(now.getTime() - 56 * 24 * 60 * 60 * 1000);
      const scansForTrend = completedScans.filter(
        (s) => new Date(s.created_at) >= eightWeeksAgo
      );

      const weekBuckets8 = new Map<number, number[]>();
      for (const scan of scansForTrend) {
        const diffMs = now.getTime() - new Date(scan.created_at).getTime();
        const weekIdx = 7 - Math.min(7, Math.floor(diffMs / (7 * 24 * 60 * 60 * 1000)));
        const existing = weekBuckets8.get(weekIdx) ?? [];
        if (scan.overall_score !== null) {
          existing.push(scan.overall_score);
          weekBuckets8.set(weekIdx, existing);
        }
      }

      const scoreTrendData: Array<{ date: string; score: number }> = Array.from(
        { length: 8 },
        (_, i) => {
          const d = new Date(now.getTime() - (7 - i) * 7 * 24 * 60 * 60 * 1000);
          const label = d.toLocaleDateString("en-US", { month: "short", day: "numeric" });
          const scores = weekBuckets8.get(i) ?? [];
          const avg =
            scores.length > 0
              ? Math.round(scores.reduce((s, v) => s + v, 0) / scores.length)
              : 0;
          return { date: label, score: avg };
        }
      ).filter((p) => p.score > 0);

      // ── critical_action_items: top 3 from recent completed scans

      let criticalActionItems: CriticalActionItem[] = [];

      const recentCompletedIds = completedScans.slice(0, 3).map((s) => s.id);
      if (recentCompletedIds.length > 0) {
        const { data: actionResultsData } = await supabase
          .from("scan_results")
          .select("scan_id, findings")
          .in("scan_id", recentCompletedIds);

        if (actionResultsData && actionResultsData.length > 0) {
          const allCritical: CriticalActionItem[] = (
            actionResultsData as Array<{ scan_id: string; findings: DbFinding[] | null }>
          ).flatMap((r) =>
            (r.findings ?? [])
              .filter(
                (f: DbFinding) =>
                  f.severity === "critical" || f.severity === "warning"
              )
              .map((f: DbFinding) => ({
                id: f.id,
                title: f.title,
                severity: f.severity,
                scanId: r.scan_id,
              }))
          );

          // Sort criticals first, take top 3
          criticalActionItems = allCritical
            .sort((a, b) => {
              if (a.severity === "critical" && b.severity !== "critical") return -1;
              if (b.severity === "critical" && a.severity !== "critical") return 1;
              return 0;
            })
            .slice(0, 3);
        }
      }

      return {
        stats,
        recentScans,
        scoreTrend,
        scoreDistribution,
        topQuickWins,
        latestScanId,
        best_score: bestScore,
        scans_this_week: scansThisWeek,
        score_trend_data: scoreTrendData,
        critical_action_items: criticalActionItems,
      };
    },
    enabled: !!user?.id,
    staleTime: 1000 * 60 * 2, // 2 minutes
  });
}

import { useState, useEffect } from "react";

export function useDebounce<T>(value: T, delay = 300): T {
  const [debounced, setDebounced] = useState<T>(value);
  useEffect(() => {
    const t = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);
  return debounced;
}

/**
 * Global feature flag hook — used throughout the app (not admin-only).
 * Reads from the same system_config table as the admin feature flags.
 * Has a longer staleTime (5 min) since feature flags change rarely.
 */

import { useQuery } from "@tanstack/react-query";
import supabase from "@/lib/supabase";
import { FLAG_DEFINITIONS } from "@/hooks/admin/use-feature-flags";

const DEFAULT_FLAGS: Record<string, boolean> = Object.fromEntries(
  FLAG_DEFINITIONS.map((f) => [f.key, f.defaultValue])
);

async function fetchAllFeatureFlags(): Promise<Record<string, boolean>> {
  const { data, error } = await supabase
    .from("system_config")
    .select("value")
    .eq("key", "feature_flags")
    .single();

  if (error && error.code !== "PGRST116") {
    // On error, return defaults silently — never break app for missing config
    return DEFAULT_FLAGS;
  }

  const stored = (data?.value ?? {}) as Record<string, boolean>;
  return { ...DEFAULT_FLAGS, ...stored };
}

/**
 * Returns the boolean value of a single feature flag.
 * Falls back to `defaultValue` if the flag is not set or loading.
 */
export function useFeatureFlag(key: string, defaultValue = true): boolean {
  const { data } = useQuery({
    queryKey: ["feature-flags"],
    queryFn: fetchAllFeatureFlags,
    staleTime: 5 * 60 * 1000, // 5 minutes
    // Never throw on error — silently use defaults
    retry: 1,
  });

  if (data === undefined) {
    // Still loading — return the flag's defined default or the passed default
    const flagDef = FLAG_DEFINITIONS.find((f) => f.key === key);
    return flagDef?.defaultValue ?? defaultValue;
  }

  return data[key] ?? defaultValue;
}

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import supabase from "@/lib/supabase";

interface FixedFindingsResult {
  fixedSet: Set<string>;
  pillarFixedCount: Record<string, number>;
  rawData: { id: string; pillar: string; fixed_findings: string[] }[];
}

export function useFixedFindings(scanId: string | undefined) {
  return useQuery({
    queryKey: ["fixed-findings", scanId],
    queryFn: async (): Promise<FixedFindingsResult> => {
      const { data, error } = await supabase
        .from("scan_results")
        .select("id, pillar, fixed_findings")
        .eq("scan_id", scanId as string);
      if (error) throw error;

      const allFixed = new Set<string>();
      const pillarFixedCount: Record<string, number> = {};

      for (const row of data ?? []) {
        const fixed = (row.fixed_findings as string[] | null) ?? [];
        fixed.forEach((fid) => allFixed.add(fid));
        pillarFixedCount[row.pillar as string] = fixed.length;
      }

      return {
        fixedSet: allFixed,
        pillarFixedCount,
        rawData: (data ?? []).map((row) => ({
          id: row.id as string,
          pillar: row.pillar as string,
          fixed_findings: (row.fixed_findings as string[] | null) ?? [],
        })),
      };
    },
    enabled: !!scanId,
  });
}

interface ToggleFindingFixedVariables {
  scanResultId: string;
  findingId: string;
  currentlyFixed: boolean;
}

export function useToggleFindingFixed() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({
      scanResultId,
      findingId,
      currentlyFixed,
    }: ToggleFindingFixedVariables): Promise<void> => {
      const { data: current, error: fetchError } = await supabase
        .from("scan_results")
        .select("fixed_findings")
        .eq("id", scanResultId)
        .single();
      if (fetchError) throw fetchError;

      const existing = (current?.fixed_findings as string[] | null) ?? [];
      const updated = currentlyFixed
        ? existing.filter((id) => id !== findingId)
        : [...existing, findingId];

      const { error } = await supabase
        .from("scan_results")
        .update({ fixed_findings: updated })
        .eq("id", scanResultId);
      if (error) throw error;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ["fixed-findings"] });
    },
  });
}

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import supabase from "@/lib/supabase";

// ── useIntegrations ────────────────────────────────────────────
// Fetches all integrations for the current user.

export function useIntegrations() {
  const { profile } = useAuth();
  const userId = profile?.id;

  return useQuery({
    queryKey: integrationKeys.all,
    queryFn: async (): Promise<IntegrationRow[]> => {
      if (!userId) return [];

      const { data, error } = await supabase
        .from("integrations")
        .select("*")
        .eq("user_id", userId)
        .order("connected_at", { ascending: false });

      if (error) throw new Error(error.message);
      return data as IntegrationRow[];
    },
    enabled: !!userId,
    staleTime: 1000 * 60, // 1 minute
  });
}

// ── useConnectIntegration ──────────────────────────────────────
// Upserts an integration row. Creates on first connect, updates on reconnect.

export function useConnectIntegration() {
  const { profile } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ provider, trackingId }: ConnectIntegrationInput): Promise<IntegrationRow> => {
      const userId = profile?.id;
      if (!userId) throw new Error("Not authenticated");

      const { data, error } = await supabase
        .from("integrations")
        .upsert(
          {
            user_id: userId,
            provider,
            access_token: trackingId,
            is_active: true,
          },
          { onConflict: "user_id,provider" }
        )
        .select("*")
        .single();

      if (error) throw new Error(error.message);
      return data as IntegrationRow;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: integrationKeys.all });
    },
  });
}

// ── useDisconnectIntegration ───────────────────────────────────
// Marks an integration as inactive and clears the stored token.

export function useDisconnectIntegration() {
  const { profile } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ provider }: DisconnectIntegrationInput): Promise<void> => {
      const userId = profile?.id;
      if (!userId) throw new Error("Not authenticated");

      const { error } = await supabase
        .from("integrations")
        .update({ is_active: false, access_token: null })
        .eq("user_id", userId)
        .eq("provider", provider);

      if (error) throw new Error(error.message);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: integrationKeys.all });
    },
  });
}

=== use-mobile ===
import * as React from "react";

const MOBILE_BREAKPOINT = 768;

export function useIsMobile() {
  const [isMobile, setIsMobile] = React.useState<boolean | undefined>(undefined);

  React.useEffect(() => {
    const mql = window.matchMedia(`(max-width: ${MOBILE_BREAKPOINT - 1}px)`);
    const onChange = () => {
      setIsMobile(window.innerWidth < MOBILE_BREAKPOINT);
    };
    mql.addEventListener("change", onChange);
    setIsMobile(window.innerWidth < MOBILE_BREAKPOINT);
    return () => mql.removeEventListener("change", onChange);
  }, []);

  return !!isMobile;
}

import { useQuery } from "@tanstack/react-query";
import supabase from "@/lib/supabase";
import { useAuth } from "@/contexts/AuthContext";
import type { CroPillar } from "@/types";
import type { ScanResultRow } from "@/types/database";

// ── Types ────────────────────────────────────────────────────────

export interface PillarDelta {
  pillar: CroPillar;
  previous: number;
  current: number;
  delta: number;
}

export interface ScanComparison {
  hasPrevious: boolean;
  previousScore: number;
  currentScore: number;
  scoreDelta: number;
  pillarDeltas: PillarDelta[];
  previousScanId: string;
}

const EMPTY_COMPARISON: ScanComparison = {
  hasPrevious: false,
  previousScore: 0,
  currentScore: 0,
  scoreDelta: 0,
  pillarDeltas: [],
  previousScanId: "",
};

// ── Query keys ──────────────────────────────────────────────────

export const scanComparisonKeys = {
  comparison: (scanId: string) => ["scan-comparison", scanId] as const,
};

// ── useScanComparison ────────────────────────────────────────────
// For the current scan, find the previous scan for the same URL and
// compute per-pillar deltas.

export function useScanComparison(
  scanId: string | undefined,
  url: string | undefined,
) {
  const { user } = useAuth();

  return useQuery({
    queryKey: scanComparisonKeys.comparison(scanId ?? ""),
    queryFn: async (): Promise<ScanComparison> => {
      if (!scanId || !url || !user?.id) return EMPTY_COMPARISON;

      const urlHash = url.toLowerCase().trim();

      // 1. Find all history entries for this URL, ordered ASC
      const { data: historyRows, error: historyError } = await supabase
        .from("scan_history")
        .select("scan_id, score, scanned_at")
        .eq("user_id", user.id)
        .eq("url_hash", urlHash)
        .order("scanned_at", { ascending: true });

      if (historyError || !historyRows || historyRows.length < 2) {
        return EMPTY_COMPARISON;
      }

      // 2. Find this scan's position in history
      const currentIndex = (historyRows as Array<{ scan_id: string; score: number; scanned_at: string }>)
        .findIndex((r) => r.scan_id === scanId);

      if (currentIndex <= 0) {
        // Not found or is the first entry — no previous scan
        return EMPTY_COMPARISON;
      }

      const currentEntry = historyRows[currentIndex] as { scan_id: string; score: number };
      const previousEntry = historyRows[currentIndex - 1] as { scan_id: string; score: number };

      // 3. Fetch pillar results for previous scan
      const { data: prevResults, error: prevError } = await supabase
        .from("scan_results")
        .select("pillar, score")
        .eq("scan_id", previousEntry.scan_id);

      if (prevError || !prevResults) return EMPTY_COMPARISON;

      // 4. Fetch pillar results for current scan
      const { data: currResults, error: currError } = await supabase
        .from("scan_results")
        .select("pillar, score")
        .eq("scan_id", scanId);

      if (currError || !currResults) return EMPTY_COMPARISON;

      // 5. Build per-pillar deltas
      const prevMap = new Map<string, number>();
      for (const r of prevResults as Pick<ScanResultRow, "pillar" | "score">[]) {
        prevMap.set(r.pillar, r.score);
      }

      const pillarDeltas: PillarDelta[] = [];
      for (const r of currResults as Pick<ScanResultRow, "pillar" | "score">[]) {
        const prevScore = prevMap.get(r.pillar);
        if (prevScore !== undefined) {
          pillarDeltas.push({
            pillar: r.pillar as CroPillar,
            previous: prevScore,
            current: r.score,
            delta: r.score - prevScore,
          });
        }
      }

      return {
        hasPrevious: true,
        previousScore: previousEntry.score,
        currentScore: currentEntry.score,
        scoreDelta: currentEntry.score - previousEntry.score,
        pillarDeltas,
        previousScanId: previousEntry.scan_id,
      };
    },
    enabled: !!scanId && !!url && !!user?.id,
    staleTime: 1000 * 60 * 5,
  });
}

// ── UrlHistoryGroup ─────────────────────────────────────────────

export interface UrlHistoryGroup {
  url: string;
  url_hash: string;
  entries: ScanHistoryEntry[];
  latestScore: number;
  firstScore: number;
  delta: number;
  latestScanId: string;
}

// ── useAllUrlHistory ────────────────────────────────────────────
// Returns all tracked URLs for the current user, grouped, with deltas.

export function useAllUrlHistory() {
  const { user } = useAuth();

  return useQuery({
    queryKey: scanHistoryKeys.allUrls(user?.id ?? ""),
    queryFn: async (): Promise<UrlHistoryGroup[]> => {
      if (!user?.id) return [];

      const { data, error } = await supabase
        .from("scan_history")
        .select("*")
        .eq("user_id", user.id)
        .order("scanned_at", { ascending: true });

      if (error) throw new Error(error.message);

      const rows = (data as Record<string, unknown>[]).map(rowToEntry);

      // Group by url_hash
      const groupMap = new Map<string, ScanHistoryEntry[]>();
      for (const entry of rows) {
        const existing = groupMap.get(entry.url_hash);
        if (existing) {
          existing.push(entry);
        } else {
          groupMap.set(entry.url_hash, [entry]);
        }
      }

      const groups: UrlHistoryGroup[] = [];
      for (const [urlHash, entries] of groupMap) {
        // entries are already ASC by scanned_at
        const firstEntry = entries[0];
        const lastEntry = entries[entries.length - 1];
        groups.push({
          url: lastEntry.url,
          url_hash: urlHash,
          entries,
          latestScore: lastEntry.score,
          firstScore: firstEntry.score,
          delta: lastEntry.score - firstEntry.score,
          latestScanId: lastEntry.scan_id,
        });
      }

      // Sort groups by latest scan time DESC
      groups.sort((a, b) => {
        const aLast = a.entries[a.entries.length - 1].scanned_at;
        const bLast = b.entries[b.entries.length - 1].scanned_at;
        return bLast.localeCompare(aLast);
      });

      return groups;
    },
    enabled: !!user?.id,
    staleTime: 1000 * 60 * 2,
  });
}

import * as React from "react";

import type { ToastActionElement, ToastProps } from "@/components/ui/toast";

const TOAST_LIMIT = 1;
const TOAST_REMOVE_DELAY = 1000000;

type ToasterToast = ToastProps & {
  id: string;
  title?: React.ReactNode;
  description?: React.ReactNode;
  action?: ToastActionElement;
};

const actionTypes = {
  ADD_TOAST: "ADD_TOAST",
  UPDATE_TOAST: "UPDATE_TOAST",
  DISMISS_TOAST: "DISMISS_TOAST",
  REMOVE_TOAST: "REMOVE_TOAST",
} as const;

let count = 0;

function genId() {
  count = (count + 1) % Number.MAX_SAFE_INTEGER;
  return count.toString();
}

type ActionType = typeof actionTypes;

type Action =
  | {
      type: ActionType["ADD_TOAST"];
      toast: ToasterToast;
    }
  | {
      type: ActionType["UPDATE_TOAST"];
      toast: Partial<ToasterToast>;
    }
  | {
      type: ActionType["DISMISS_TOAST"];
      toastId?: ToasterToast["id"];
    }
  | {
      type: ActionType["REMOVE_TOAST"];
      toastId?: ToasterToast["id"];
    };

interface State {
  toasts: ToasterToast[];
}

const toastTimeouts = new Map<string, ReturnType<typeof setTimeout>>();

const addToRemoveQueue = (toastId: string) => {
  if (toastTimeouts.has(toastId)) {
    return;
  }

  const timeout = setTimeout(() => {
    toastTimeouts.delete(toastId);
    dispatch({
      type: "REMOVE_TOAST",
      toastId: toastId,
    });
  }, TOAST_REMOVE_DELAY);

  toastTimeouts.set(toastId, timeout);
};

export const reducer = (state: State, action: Action): State => {
  switch (action.type) {
    case "ADD_TOAST":
      return {
        ...state,
        toasts: [action.toast, ...state.toasts].slice(0, TOAST_LIMIT),
      };

    case "UPDATE_TOAST":
      return {
        ...state,
        toasts: state.toasts.map((t) => (t.id === action.toast.id ? { ...t, ...action.toast } : t)),
      };

    case "DISMISS_TOAST": {
      const { toastId } = action;

      // ! Side effects ! - This could be extracted into a dismissToast() action,
      // but I'll keep it here for simplicity
      if (toastId) {
        addToRemoveQueue(toastId);
      } else {
        state.toasts.forEach((toast) => {
          addToRemoveQueue(toast.id);
        });
      }

      return {
        ...state,
        toasts: state.toasts.map((t) =>
          t.id === toastId || toastId === undefined
            ? {
                ...t,
                open: false,
              }
            : t,
        ),
      };
    }
    case "REMOVE_TOAST":
      if (action.toastId === undefined) {
        return {
          ...state,
          toasts: [],
        };
      }
      return {
        ...state,
        toasts: state.toasts.filter((t) => t.id !== action.toastId),
      };
  }
};

const listeners: Array<(state: State) => void> = [];

let memoryState: State = { toasts: [] };

function dispatch(action: Action) {
  memoryState = reducer(memoryState, action);
  listeners.forEach((listener) => {
    listener(memoryState);
  });
}

type Toast = Omit<ToasterToast, "id">;

function toast({ ...props }: Toast) {
  const id = genId();

  const update = (props: ToasterToast) =>
    dispatch({
      type: "UPDATE_TOAST",
      toast: { ...props, id },
    });
  const dismiss = () => dispatch({ type: "DISMISS_TOAST", toastId: id });

  dispatch({
    type: "ADD_TOAST",
    toast: {
      ...props,
      id,
      open: true,
      onOpenChange: (open) => {
        if (!open) dismiss();
      },
    },
  });

  return {
    id: id,
    dismiss,
    update,
  };
}

function useToast() {
  const [state, setState] = React.useState<State>(memoryState);

  React.useEffect(() => {
    listeners.push(setState);
    return () => {
      const index = listeners.indexOf(setState);
      if (index > -1) {
        listeners.splice(index, 1);
      }
    };
  }, [state]);

  return {
    ...state,
    toast,
    dismiss: (toastId?: string) => dispatch({ type: "DISMISS_TOAST", toastId }),
  };
}

export { useToast, toast };

      const insert: WebhookEndpointInsert = {
        user_id: userId,
        url,
        secret,
        events,
        is_active: true,
      };

      const { data, error } = await supabase
        .from("webhook_endpoints")
        .insert(insert)
        .select("*")
        .single();

      if (error) throw new Error(error.message);
      return data as WebhookEndpoint;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: webhookKeys.all });
    },
  });
}

// ── useDeleteWebhook ───────────────────────────────────────────
// Deletes a webhook endpoint by ID.

export function useDeleteWebhook() {
  const { profile } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async (id: string): Promise<void> => {
      const userId = profile?.id;
      if (!userId) throw new Error("Not authenticated");

      const { error } = await supabase
        .from("webhook_endpoints")
        .delete()
        .eq("id", id)
        .eq("user_id", userId);

      if (error) throw new Error(error.message);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: webhookKeys.all });
    },
  });
}

// ── useToggleWebhook ───────────────────────────────────────────
// Toggles is_active for a webhook endpoint.

export function useToggleWebhook() {
  const { profile } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({
      id,
      is_active,
    }: {
      id: string;
      is_active: boolean;
    }): Promise<void> => {
      const userId = profile?.id;
      if (!userId) throw new Error("Not authenticated");

      const { error } = await supabase
        .from("webhook_endpoints")
        .update({ is_active })
        .eq("id", id)
        .eq("user_id", userId);

      if (error) throw new Error(error.message);
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: webhookKeys.all });
    },
  });
}
