// ── Query keys ─────────────────────────────────────────────────

export const webhookKeys = {
  all: ["webhook_endpoints"] as const,
};

// ── useWebhooks ────────────────────────────────────────────────
// Fetches all webhook endpoints for the current user.

export function useWebhooks() {
  const { profile } = useAuth();
  const userId = profile?.id;

  return useQuery({
    queryKey: webhookKeys.all,
    queryFn: async (): Promise<WebhookEndpoint[]> => {
      if (!userId) return [];

      const { data, error } = await supabase
        .from("webhook_endpoints")
        .select("*")
        .eq("user_id", userId)
        .order("created_at", { ascending: false });

      if (error) throw new Error(error.message);
      return data as WebhookEndpoint[];
    },
    enabled: !!userId,
    staleTime: 1000 * 60, // 1 minute
  });
}

// ── useCreateWebhook ───────────────────────────────────────────
// Creates a new webhook endpoint with an auto-generated secret.

interface CreateWebhookInput {
  url: string;
  events?: string[];
}

export function useCreateWebhook() {
  const { profile } = useAuth();
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({
      url,
      events = ["scan.completed"],
    }: CreateWebhookInput): Promise<WebhookEndpoint> => {
      const userId = profile?.id;
      if (!userId) throw new Error("Not authenticated");

      const secret = crypto.randomUUID().replace(/-/g, "") + crypto.randomUUID().replace(/-/g, "");

      const insert: WebhookEndpointInsert = {
        user_id: userId,
        url,
        secret,
        events,
        is_active: true,
      };

      const { data, error } = await supabase
        .from("webhook_endpoints")
        .insert(insert)
        .select("*")
        .single();

      if (error) throw new Error(error.message);
      return data as WebhookEndpoint;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: webhookKeys.all });
    },
  });
}

// ── Plan prices ───────────────────────────────────────────────
// Keep in sync with use-admin-actions.ts plan limits

export const PLAN_PRICES: Record<string, number> = {
  free: 0,
  pro: 49,
  agency: 199,
};

// ── Types ─────────────────────────────────────────────────────

export interface PlanDistribution {
  free: number;
  pro: number;
  agency: number;
}

export interface BillingStats {
  mrr: number;
  annualProjection: number;
  activeSubscriptions: number;
  freeUsers: number;
  planDistribution: PlanDistribution;
}

export interface PlanChangeEntry extends AuditLogEntry {
  previous_plan?: string;
  new_plan?: string;
}

// ── Fetch helpers ─────────────────────────────────────────────

async function fetchBillingStats(): Promise<BillingStats> {
  const { data, error } = await supabase
    .from("profiles")
    .select("plan");

  if (error) throw new Error(error.message);

  const profiles = data ?? [];

  const planDistribution: PlanDistribution = { free: 0, pro: 0, agency: 0 };

  for (const p of profiles) {
    const plan = p.plan as keyof PlanDistribution;
    if (plan in planDistribution) {
      planDistribution[plan]++;
    }
  }

  const mrr =
    planDistribution.pro * PLAN_PRICES.pro +
    planDistribution.agency * PLAN_PRICES.agency;

  const annualProjection = mrr * 12;

  const activeSubscriptions = planDistribution.pro + planDistribution.agency;

  return {
    mrr,
    annualProjection,
    activeSubscriptions,
    freeUsers: planDistribution.free,
    planDistribution,
  };
}

// ── Extended scan row with joined user profile ────────────────

export interface AdminScanRow extends ScanRow {

  user_name?: string;
}

// ── Filters ───────────────────────────────────────────────────

export interface AdminScansFilter {
  search?: string;
  status?: ScanStatus | "";
  scoreLte?: number;
  scoreGte?: number;
  dateFrom?: string;
  dateTo?: string;
}

export interface AdminScansResult {
  scans: AdminScanRow[];
  total: number;
}

// ── Aggregate stats ───────────────────────────────────────────

export interface AdminScanStats {
  totalScans: number;
  avgScore: number;
  failedScans: number;
  scansToday: number;
}

// ── Fetch helpers ─────────────────────────────────────────────

interface FetchAdminScansParams {
  page: number;
  pageSize: number;
  filter: AdminScansFilter;
  sortColumn: keyof ScanRow;
  sortAsc: boolean;
}

async function fetchAdminScans({
  page,
  pageSize,
  filter,
  sortColumn,
  sortAsc,
}: FetchAdminScansParams): Promise<AdminScansResult> {
  // Fetch scans with joined profile for user email/name
  let query = supabase
    .from("scans")
    .select(
      `
      *,
      profiles:user_id (
        email,
        full_name
      )
    `,
      { count: "exact" }
    )
    .order(sortColumn as string, { ascending: sortAsc });

  if (filter.status) {
    query = query.eq("status", filter.status);
  }

  if (filter.scoreGte !== undefined && filter.scoreGte > 0) {
    query = query.gte("overall_score", filter.scoreGte);
  }

  if (filter.scoreLte !== undefined && filter.scoreLte < 100) {
    query = query.lte("overall_score", filter.scoreLte);
  }

  if (filter.dateFrom) {
    query = query.gte("created_at", filter.dateFrom);
  }

  if (filter.dateTo) {
    query = query.lte("created_at", filter.dateTo);
  }

  const from = page * pageSize;
  const to = from + pageSize - 1;
  query = query.range(from, to);

  const { data, count, error } = await query;

  if (error) throw new Error(error.message);

  const scans: AdminScanRow[] = (data ?? []).map((row) => {
    const profileData = row.profiles as
      | { email: string; full_name: string | null }
      | null;

    // Filter search client-side after fetch if needed
    return {
      id: row.id,
      user_id: row.user_id,
      url: row.url,
      favicon_url: row.favicon_url,
      page_title: row.page_title,
      page_type: row.page_type,
      status: row.status,
      overall_score: row.overall_score,
      issues_found: row.issues_found,
      critical_issues: row.critical_issues,
      warnings: row.warnings,
      quick_wins: row.quick_wins,
      screenshot_url: row.screenshot_url,
      device: row.device,
      error_message: row.error_message,
      created_at: row.created_at,
      completed_at: row.completed_at,
      user_email: profileData?.email,
      user_name: profileData?.full_name ?? undefined,
    };
  });

  // Apply search filter after joining (Supabase can't ilike on joined columns directly)
  const filtered = filter.search
    ? scans.filter(
        (s) =>
          s.url.toLowerCase().includes(filter.search!.toLowerCase()) ||
          (s.user_email ?? "")
            .toLowerCase()
            .includes(filter.search!.toLowerCase())
      )
    : scans;

  return {
    scans: filtered,
    total: filter.search ? filtered.length : (count ?? 0),
  };
}

async function fetchAdminScanStats(): Promise<AdminScanStats> {
  const todayStart = new Date();
  todayStart.setHours(0, 0, 0, 0);

  const [
    { count: totalScans, error: totalError },
    { count: failedScans, error: failedError },
    { count: scansToday, error: todayError },
    { data: scoreData, error: scoreError },
  ] = await Promise.all([
    supabase.from("scans").select("*", { count: "exact", head: true }),
    supabase
      .from("scans")
      .select("*", { count: "exact", head: true })
      .eq("status", "failed"),
    supabase
      .from("scans")
      .select("*", { count: "exact", head: true })
      .gte("created_at", todayStart.toISOString()),
    supabase
      .from("scans")
      .select("overall_score")
      .eq("status", "completed")
      .not("overall_score", "is", null),
  ]);

  if (totalError) throw new Error(totalError.message);
  if (failedError) throw new Error(failedError.message);
  if (todayError) throw new Error(todayError.message);
  if (scoreError) throw new Error(scoreError.message);

  const scores = (scoreData ?? [])
    .map((r) => r.overall_score as number)
    .filter((s) => s !== null);

  const avgScore =
    scores.length > 0
      ? Math.round(scores.reduce((a, b) => a + b, 0) / scores.length)
      : 0;

  return {
    totalScans: totalScans ?? 0,
    avgScore,
    failedScans: failedScans ?? 0,
    scansToday: scansToday ?? 0,
  };
}

export function useAdminRetryScan() {
  const queryClient = useQueryClient();
  const { user } = useAuth();

  return useMutation({
    mutationFn: async (scanId: string) => {
      if (!user) throw new Error("Not authenticated");

      // Reset scan to pending state
      const { error: updateError } = await supabase
        .from("scans")
        .update({ status: "pending", completed_at: null, error_message: null })
        .eq("id", scanId);

      if (updateError) throw new Error(updateError.message);

      // Trigger re-analysis
      const { error: fnError } = await supabase.functions.invoke("analyze-url", {
        body: { scanId },
      });

      // Non-fatal: function might not be available locally
      if (fnError) {
        console.warn("analyze-url function invocation warning:", fnError.message);
      }

      await writeAuditLog(user.id, "retry_scan", "scan", scanId, {});
    },
    onSuccess: () => {
      toast.success("Scan queued for retry");
      queryClient.invalidateQueries({ queryKey: ["admin", "scans"] });
      queryClient.invalidateQueries({ queryKey: ["admin", "scan-stats"] });
    },
    onError: (error: Error) => {
      toast.error(`Failed to retry scan: ${error.message}`);
    },
  });
}

  firstOfMonth.setHours(0, 0, 0, 0);

  const [
    { count: totalUsers, error: usersError },
    { count: totalScans, error: scansError },
    { count: failedScans, error: failedError },
    { count: completedScans, error: completedError },
    { data: activeData, error: activeError },
    { count: newUsersToday, error: newUsersTodayError },
    { count: scansThisMonth, error: scansMonthError },
    { data: monthScores, error: monthScoresError },
  ] = await Promise.all([
    supabase.from("profiles").select("*", { count: "exact", head: true }),
    supabase.from("scans").select("*", { count: "exact", head: true }),
    supabase.from("scans").select("*", { count: "exact", head: true }).eq("status", "failed"),
    supabase.from("scans").select("*", { count: "exact", head: true }).eq("status", "completed"),
    supabase.from("scans").select("user_id").gte("created_at", sevenDaysAgo),
    supabase
      .from("profiles")
      .select("*", { count: "exact", head: true })
      .gte("created_at", todayMidnightUTC.toISOString()),
    supabase
      .from("scans")
      .select("*", { count: "exact", head: true })
      .gte("created_at", firstOfMonth.toISOString()),
    supabase
      .from("scans")
      .select("overall_score")
      .eq("status", "completed")
      .gte("created_at", firstOfMonth.toISOString())
      .not("overall_score", "is", null),
  ]);

  if (usersError) throw new Error(usersError.message);
  if (scansError) throw new Error(scansError.message);
  if (failedError) throw new Error(failedError.message);
  if (completedError) throw new Error(completedError.message);
  if (activeError) throw new Error(activeError.message);
  if (newUsersTodayError) throw new Error(newUsersTodayError.message);
  if (scansMonthError) throw new Error(scansMonthError.message);
  if (monthScoresError) throw new Error(monthScoresError.message);

  const uniqueActiveUsers = new Set((activeData ?? []).map((r) => r.user_id)).size;

  const total = totalScans ?? 0;
  const completed = completedScans ?? 0;
  const scanSuccessRate = total > 0 ? Math.round((completed / total) * 1000) / 10 : 0;

  const scores = (monthScores ?? [])
    .map((r) => r.overall_score as number)
    .filter((s) => s !== null && s !== undefined);
  const avgScoreThisMonth =
    scores.length > 0
      ? Math.round(scores.reduce((a, b) => a + b, 0) / scores.length)
      : 0;

  return {
    totalUsers: totalUsers ?? 0,
    totalScans: total,
    activeUsers: uniqueActiveUsers,
    failedScans: failedScans ?? 0,
    scanSuccessRate,
    newUsersToday: newUsersToday ?? 0,
    avgScoreThisMonth,
    scansThisMonth: scansThisMonth ?? 0,
  };
}

async function fetchUserGrowth(): Promise<UserGrowthDataPoint[]> {
  const thirtyDaysAgo = subDays(new Date(), 30).toISOString();

  const { data, error } = await supabase
    .from("profiles")
    .select("created_at")
    .gte("created_at", thirtyDaysAgo)
    .order("created_at", { ascending: true });

  if (error) throw new Error(error.message);

  // Group by date
  const countsByDate: Record<string, number> = {};
  for (const row of data ?? []) {
    const day = format(new Date(row.created_at), "MMM d");
    countsByDate[day] = (countsByDate[day] ?? 0) + 1;
  }

  // Build 30-day series with zeros for missing days
  const result: UserGrowthDataPoint[] = [];
  let cumulative = 0;
  for (let i = 29; i >= 0; i--) {
    const d = subDays(new Date(), i);
    const dateLabel = format(d, "MMM d");
    const count = countsByDate[dateLabel] ?? 0;
    cumulative += count;
    result.push({ date: dateLabel, count, cumulative });
  }

  return result;
}

async function fetchScanVolume(): Promise<ScanVolumeDataPoint[]> {
  const sevenDaysAgo = subDays(startOfDay(new Date()), 6).toISOString();

  const { data, error } = await supabase
    .from("scans")
    .select("created_at, status")
    .gte("created_at", sevenDaysAgo)
    .order("created_at", { ascending: true });

  if (error) throw new Error(error.message);

  const byDay: Record<string, { total: number; completed: number; failed: number }> = {};

  for (const row of data ?? []) {
    const day = format(new Date(row.created_at), "EEE");
    if (!byDay[day]) byDay[day] = { total: 0, completed: 0, failed: 0 };
    byDay[day].total++;
    if (row.status === "completed") byDay[day].completed++;
    if (row.status === "failed") byDay[day].failed++;
  }

  // Build 7-day series
  const result: ScanVolumeDataPoint[] = [];
  for (let i = 6; i >= 0; i--) {
    const day = format(subDays(new Date(), i), "EEE");
    const entry = byDay[day] ?? { total: 0, completed: 0, failed: 0 };
    result.push({ day, ...entry });
  }

  return result;
}

async function fetchAtRiskUsers(): Promise<AtRiskUser[]> {
  const sevenDaysAgo = new Date(
    Date.now() - 7 * 24 * 60 * 60 * 1000
  ).toISOString();

  const { data: profiles, error: profilesError } = await supabase
    .from("profiles")
    .select("*")
    .order("created_at", { ascending: false })
    .limit(200);

  if (profilesError) throw new Error(profilesError.message);

  const { data: failedScansData, error: failedError } = await supabase
    .from("scans")
    .select("user_id")
    .eq("status", "failed")
    .gte("created_at", sevenDaysAgo);

  if (failedError) throw new Error(failedError.message);

  const { data: lastScans, error: lastScansError } = await supabase
    .from("scans")
    .select("user_id, created_at")
    .order("created_at", { ascending: false });

  if (lastScansError) throw new Error(lastScansError.message);

  const failedUserIds = new Set(
    (failedScansData ?? []).map((r) => r.user_id)
  );

  const lastScanMap = new Map<string, string>();
  for (const scan of lastScans ?? []) {
    if (!lastScanMap.has(scan.user_id)) {
      lastScanMap.set(scan.user_id, scan.created_at);
    }
  }

  const atRisk: AtRiskUser[] = [];

  for (const profile of profiles ?? []) {
    const p = profile as ProfileRow;
    const atLimit = p.scans_used >= p.scan_limit;
    const hasFailed = failedUserIds.has(p.id);
    const neverScanned = !lastScanMap.has(p.id);
    const lastScanDate = lastScanMap.get(p.id) ?? null;

    if (atLimit || hasFailed || neverScanned) {
      atRisk.push({
        ...p,
        has_failed_scans: hasFailed,
        never_scanned: neverScanned,
        at_scan_limit: atLimit,
        last_scan_date: lastScanDate,
      });
    }
  }

  return atRisk.sort((a, b) => {
    const scoreA = (a.at_scan_limit ? 3 : 0) + (a.has_failed_scans ? 2 : 0) + (a.never_scanned ? 1 : 0);
    const scoreB = (b.at_scan_limit ? 3 : 0) + (b.has_failed_scans ? 2 : 0) + (b.never_scanned ? 1 : 0);
    return scoreB - scoreA;
  });
}

      `email.ilike.%${filter.search}%,full_name.ilike.%${filter.search}%`
    );
  }

  if (filter.plan) {
    query = query.eq("plan", filter.plan);
  }

  if (filter.role) {
    query = query.eq("role", filter.role);
  }

  const from = page * pageSize;
  const to = from + pageSize - 1;

  query = query
    .order(sortColumn as string, { ascending: sortAsc })
    .range(from, to);

  const { data, count, error } = await query;

  if (error) throw new Error(error.message);

  return {
    users: (data ?? []) as ProfileRow[],
    total: count ?? 0,
  };
}

async function fetchAllUsersForExport(filter: AdminUsersFilter): Promise<string> {
  let query = supabase
    .from("profiles")
    .select("*")
    .order("created_at", { ascending: false });

  if (filter.search) {
    query = query.or(
      `email.ilike.%${filter.search}%,full_name.ilike.%${filter.search}%`
    );
  }
  if (filter.plan) {
    query = query.eq("plan", filter.plan);
  }
  if (filter.role) {
    query = query.eq("role", filter.role);
  }

  const { data, error } = await query;
  if (error) throw new Error(error.message);

  const users = (data ?? []) as ProfileRow[];

  const headers = ["Email", "Full Name", "Plan", "Role", "Scans Used", "Scan Limit", "Created At", "Is Banned"];
  const rows = users.map((u) => [
    u.email ?? "",
    u.full_name ?? "",
    u.plan,
    u.role,
    String(u.scans_used),
    String(u.scan_limit),
    format(new Date(u.created_at), "yyyy-MM-dd HH:mm:ss"),
    u.is_banned ? "Yes" : "No",
  ]);

  const escape = (v: string) => `"${v.replace(/"/g, '""')}"`;
  const csvLines = [headers.map(escape).join(","), ...rows.map((r) => r.map(escape).join(","))];
  return csvLines.join("\n");
}

export function useExportUsers(filter: AdminUsersFilter) {
  return useQuery({
    queryKey: ["admin", "users", "export", filter],
    queryFn: () => fetchAllUsersForExport(filter),
    enabled: false, // Only run when refetch() is called manually
    staleTime: 0,
  });
}

async function fetchAuditLog({
  page,
  pageSize,
  actionFilter,
}: FetchAuditLogParams): Promise<{ entries: AuditLogEntry[]; total: number }> {
  let query = supabase
    .from("audit_logs")
    .select(
      `
      *,
      profiles:admin_id (
        email,
        full_name
      )
    `,
      { count: "exact" }
    )
    .order("created_at", { ascending: false });

  }

  const from = page * pageSize;

      admin_name: profileData?.full_name,

import { useQuery, useMutation, useQueryClient } from "@tanstack/react-query";

export interface FlagDefinition {

}

export const FLAG_DEFINITIONS: FlagDefinition[] = [

    defaultValue: false,

    requiresSuperAdmin: true,
  },

const DEFAULT_FLAGS: Record<string, boolean> = Object.fromEntries(
  FLAG_DEFINITIONS.map((f) => [f.key, f.defaultValue])
);

export interface FeatureFlagsData {
  flags: Record<string, boolean>;
  updatedAt: string | null;
}

async function fetchFeatureFlags(): Promise<FeatureFlagsData> {
  const { data, error } = await supabase
    .from("system_config")
    .select("value, updated_at")
    .eq("key", "feature_flags")
    .single();

  if (error && error.code !== "PGRST116") {
    // PGRST116 = row not found — that's fine, we use defaults
    throw new Error(error.message);
  }

  const stored = (data?.value ?? {}) as Record<string, boolean>;

  // Merge stored values with defaults (stored values take priority)
  const flags: Record<string, boolean> = { ...DEFAULT_FLAGS, ...stored };

  return {
    flags,
    updatedAt: data?.updated_at ?? null,
  };
}

// ── Admin hook: reads + writes ────────────────────────────────

export function useFeatureFlags() {
  return useQuery({
    queryKey: ["admin", "feature-flags"],
    queryFn: fetchFeatureFlags,
    staleTime: 60_000,
  });
}

export function useUpdateFeatureFlag() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({
      key,
      enabled,
      currentFlags,
    }: {
      key: string;
      enabled: boolean;
      currentFlags: Record<string, boolean>;
    }) => {
      const mergedFlags = { ...currentFlags, [key]: enabled };

      const { error } = await supabase.from("system_config").upsert(
        {
          key: "feature_flags",
          value: mergedFlags,
          updated_at: new Date().toISOString(),
        },
        { onConflict: "key" }
      );

      if (error) throw new Error(error.message);

      return { key, enabled, mergedFlags };
    },
    onSuccess: (result) => {
      const flagDef = FLAG_DEFINITIONS.find((f) => f.key === result.key);
      const label = flagDef?.label ?? result.key;
      toast.success(`${label} ${result.enabled ? "enabled" : "disabled"}`);
      // Invalidate both admin and app-level queries
      queryClient.invalidateQueries({ queryKey: ["admin", "feature-flags"] });
      queryClient.invalidateQueries({ queryKey: ["feature-flags"] });
    },
    onError: (error: Error) => {
      toast.error(`Failed to update flag: ${error.message}`);
    },
  });
}
