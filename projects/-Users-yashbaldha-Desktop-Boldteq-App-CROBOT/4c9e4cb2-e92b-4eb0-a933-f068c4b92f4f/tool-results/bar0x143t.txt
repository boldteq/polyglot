    39	function CopyInline({ text }: { text: string }) {
    40	  const [copied, setCopied] = useState(false);
    41	  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    42	
    43	  useEffect(
    44	    () => () => {
    45	      if (timerRef.current) clearTimeout(timerRef.current);
    46	    },
    47	    [],
    48	  );
    49	
    50	  const handleCopy = async () => {
    51	    if (timerRef.current) clearTimeout(timerRef.current);
    52	    try {
    53	      await navigator.clipboard.writeText(text);
    54	    } catch {
    55	      /* clipboard API unavailable */
    56	    }
    57	    setCopied(true);
    58	    timerRef.current = setTimeout(() => setCopied(false), 2000);
    59	  };
    60	
    61	  return (
    62	    <button
    63	      onClick={handleCopy}
    64	      className={cn(
    65	        "inline-flex items-center gap-1 rounded-md px-2 py-0.5 text-2xs font-medium border transition-all",
    66	        copied
    67	          ? "bg-green-50 text-green-600 border-green-200"
    68	          : "bg-card text-muted-foreground border-border/50 hover:bg-muted/40 hover:text-foreground",

    71	      {copied ? (

   266	                          key={color.hex}
   267	                          type="button"
   268	                          title={color.label}
   269	                          onClick={() => {
   270	                            setAccentColor(color.hex);
   271	                            setCustomHex("");
   272	                          }}
   273	                          className={cn(
   274	                            "h-7 w-7 rounded-full border-2 transition-all",
   275	                            effectiveAccent === color.hex
   276	                              ? "border-foreground scale-110 shadow-md"
   277	                              : "border-transparent hover:scale-105"
   278	                          )}
   279	                          style={{ backgroundColor: color.hex }}
   280	                        />
   281	                      ))}
   282	                    </div>
   283	                    <div className="flex items-center gap-2">
   284	                      <Input
   285	                        value={customHex}
   286	                        onChange={(e) => setCustomHex(e.target.value)}
   287	                        placeholder="#1a2b3c"
   288	                        maxLength={7}
   289	                        className="h-8 w-28 font-mono text-xs"
   290	                      />
   291	                      <span className="text-xs text-muted-foreground">Custom hex</span>
   292	                      {customHex && !customHex.match(/^#[0-9a-fA-F]{6}$/) && (
   293	                        <span className="text-xs text-red-500">Invalid format</span>
   294	                      )}
   295	                    </div>
   296	                  </div>
   297	
   298	                  {/* Section toggles */}
   299	                  <div className="space-y-3">
   300	                    <Label className="text-sm font-medium">Include Sections</Label>
   301	
   302	                    <div className="flex items-center gap-3">
   303	                      <Checkbox
   304	                        id="include-rewrites"
   305	                        checked={includeRewrites}
   306	                        onCheckedChange={(v) => setIncludeRewrites(v === true)}
   307	                        disabled={!hasRewrites}
   308	                      />
   309	                      <label
   310	                        htmlFor="include-rewrites"
   311	                        className={cn(
   312	                          "text-sm cursor-pointer",
   313	                          !hasRewrites && "text-muted-foreground cursor-not-allowed"
   314	                        )}
   315	                      >
   316	                        AI Copy Rewrites
   317	                        {!hasRewrites && (
   318	                          <span className="ml-2 text-xs text-muted-foreground">
   319	                            (no rewrites for this scan)
   320	                          </span>
   321	                        )}
   322	                      </label>
   323	                    </div>
   324	
   325	                    <div className="flex items-center gap-3">
   326	                      <Checkbox
   327	                        id="include-history"
   328	                        checked={includeHistory}
   329	                        onCheckedChange={(v) => setIncludeHistory(v === true)}
   330	                        disabled={!hasHistory}
   331	                      />
   332	                      <label
   333	                        htmlFor="include-history"
   334	                        className={cn(
   335	                          "text-sm cursor-pointer",
   336	                          !hasHistory && "text-muted-foreground cursor-not-allowed"
   337	                        )}
   338	                      >
   339	                        Score History
   340	                        {!hasHistory && (
   341	                          <span className="ml-2 text-xs text-muted-foreground">
   342	                            (requires 2+ scans of this URL)
   343	                          </span>
   344	                        )}
   345	                      </label>
   346	                    </div>
   347	                  </div>
   348	                </div>
   349	              ) : (
   350	                // Not Pro+ — show upgrade prompt
   351	                <div className="flex items-start gap-3 rounded-lg border border-amber-200 bg-amber-50 p-4">
   352	                  <Lock className="mt-0.5 h-4 w-4 shrink-0 text-amber-600" />
   353	                  <div>
   354	                    <p className="text-sm font-semibold text-amber-800">Pro plan required</p>
   355	                    <p className="mt-0.5 text-xs text-amber-700">
   356	                      Upgrade to Pro to unlock enhanced branded reports, custom colors,
   357	                      AI copy rewrite sections, and score history pages.
   358	                    </p>
   359	                  </div>
   360	                </div>
   361	              )}
   362	            </>
   363	          )}
   364	        </div>
   365	
   366	        <DialogFooter className="gap-2 sm:gap-0">
   367	          <Button
   368	            variant="outline"
   369	            size="sm"
   370	            onClick={() => onOpenChange(false)}
   371	            disabled={isGenerating}
   372	          >
   373	            Cancel
   374	          </Button>
   375	          <Button
   376	            size="sm"
   377	            onClick={handleDownload}
   378	            disabled={isGenerating || (pdfType === "enhanced" && !isProUser)}
   379	            className="min-w-[120px]"
   380	          >
   381	            {isGenerating ? (
   382	              <>
   383	                <RefreshCw className="mr-1.5 h-3.5 w-3.5 animate-spin" />
   384	                Generating…
   385	              </>
   386	            ) : (
   387	              <>
   388	                <Download className="mr-1.5 h-3.5 w-3.5" />
   389	                Download PDF
   390	              </>
   391	            )}
   392	          </Button>
   393	        </DialogFooter>
   394	      </DialogContent>
   395	    </Dialog>
   396	  );
