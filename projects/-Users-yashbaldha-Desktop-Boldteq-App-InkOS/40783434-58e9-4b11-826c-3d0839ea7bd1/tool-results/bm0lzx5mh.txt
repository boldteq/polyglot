-- =============================================================
-- SECTION 9 — RPC: increment_usage_meter
-- =============================================================
-- SECURITY DEFINER: bypasses RLS. Called by API routes to atomically
-- increment usage. Returns jsonb: { used, quota, exceeded }.
-- Period = calendar month (UTC). Creates row if missing.
-- =============================================================

CREATE OR REPLACE FUNCTION public.increment_usage_meter(
  p_studio_id uuid,
  p_meter     text,
  p_amount    int DEFAULT 1
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_period_start  timestamptz;
  v_period_end    timestamptz;
  v_used          int;
  v_quota         int;
  v_plan_tier     text;
  v_plan_row      public.plans%ROWTYPE;
BEGIN
  -- Compute current calendar month window (UTC)
  v_period_start := date_trunc('month', now() AT TIME ZONE 'UTC');
  v_period_end   := v_period_start + interval '1 month';

  -- Resolve quota from the studio's current plan
  SELECT s.plan_tier INTO v_plan_tier
  FROM public.subscriptions s
  WHERE s.studio_id = p_studio_id
  LIMIT 1;

  -- Default to 'solo' if no subscription row yet
  v_plan_tier := COALESCE(v_plan_tier, 'solo');

  SELECT * INTO v_plan_row
  FROM public.plans
  WHERE tier = v_plan_tier
    AND billing_cycle = 'monthly'   -- quota is per-period regardless of billing cycle
  LIMIT 1;

  -- Derive quota from plan columns
  v_quota := CASE p_meter
    WHEN 'sms_sent'          THEN v_plan_row.sms_monthly_quota
    WHEN 'ai_generations'    THEN v_plan_row.ai_generations_quota
    WHEN 'bookings_created'  THEN NULL  -- unlimited bookings at all tiers
    WHEN 'active_artists'    THEN v_plan_row.max_artists
    WHEN 'active_locations'  THEN v_plan_row.max_locations
    WHEN 'storage_mb'        THEN NULL  -- unlimited storage (monitored, not enforced yet)
    ELSE NULL
  END;

  -- Upsert: increment if row exists, insert if not
  INSERT INTO public.usage_meters
    (studio_id, period_start, period_end, meter, used, quota)
  VALUES
    (p_studio_id, v_period_start, v_period_end, p_meter, p_amount, v_quota)
  ON CONFLICT (studio_id, period_start, meter) DO UPDATE
    SET used       = usage_meters.used + p_amount,
        quota      = v_quota,           -- refresh quota snapshot on every write
        updated_at = now()
  RETURNING used INTO v_used;

  RETURN jsonb_build_object(
    'used',     v_used,
    'quota',    v_quota,
    'exceeded', CASE WHEN v_quota IS NOT NULL AND v_used > v_quota THEN true ELSE false END
  );
END;
$$;

-- SECTION 10 — RPC: check_feature_access
-- =============================================================
-- Resolves: active feature_flags override takes precedence;
-- otherwise falls through to plan-level features jsonb array.
-- Used by lib/billing/feature-gate.ts.
-- =============================================================

CREATE OR REPLACE FUNCTION public.check_feature_access(
  p_studio_id   uuid,
  p_feature_key text
)
RETURNS boolean
LANGUAGE plpgsql
SECURITY DEFINER
STABLE
SET search_path = public
AS $$
DECLARE
  v_flag_enabled  boolean;
  v_flag_exists   boolean;
  v_plan_tier     text;
  v_features      jsonb;
BEGIN
  -- 1. Check explicit feature_flag override (not expired)
  SELECT enabled, true
  INTO v_flag_enabled, v_flag_exists
  FROM public.feature_flags
  WHERE studio_id = p_studio_id
    AND flag_key  = p_feature_key
    AND (expires_at IS NULL OR expires_at > now())
  LIMIT 1;

  IF v_flag_exists THEN
    RETURN v_flag_enabled;
  END IF;

  -- 2. Fall back to plan features array
  SELECT s.plan_tier INTO v_plan_tier
  FROM public.subscriptions s
  WHERE s.studio_id = p_studio_id
  LIMIT 1;

  v_plan_tier := COALESCE(v_plan_tier, 'solo');

  SELECT pl.features INTO v_features
  FROM public.plans pl
  WHERE pl.tier          = v_plan_tier
    AND pl.billing_cycle = 'monthly'   -- features are identical across cycles
    AND pl.is_active     = true
  LIMIT 1;

  RETURN COALESCE(v_features ? p_feature_key, false);
END;
$$;

-- =============================================================
-- SECTION 11 — RPC: get_studio_subscription
-- =============================================================
-- Returns subscription + plan + current usage meters for the studio.
-- Caller must be a member of p_studio_id.
-- =============================================================

CREATE OR REPLACE FUNCTION public.get_studio_subscription(
  p_studio_id uuid
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
STABLE
SET search_path = public
AS $$
DECLARE
  v_sub       public.subscriptions%ROWTYPE;
  v_plan      public.plans%ROWTYPE;
  v_meters    jsonb;
  v_period_start timestamptz;
  v_period_end   timestamptz;
BEGIN
  -- Verify caller is a member of this studio
  IF NOT EXISTS (
    SELECT 1 FROM public.studio_members
    WHERE studio_id = p_studio_id AND user_id = auth.uid()
  ) THEN
    RAISE EXCEPTION 'access_denied' USING ERRCODE = '42501';
  END IF;

  -- Fetch subscription
  SELECT * INTO v_sub
  FROM public.subscriptions
  WHERE studio_id = p_studio_id
  LIMIT 1;

  -- Fetch matching plan (prefer same billing_cycle; fallback monthly)
  SELECT * INTO v_plan
  FROM public.plans
  WHERE tier          = COALESCE(v_sub.plan_tier, 'solo')
    AND billing_cycle = COALESCE(v_sub.billing_cycle, 'monthly')
    AND is_active     = true
  LIMIT 1;

  -- Current period usage meters
  v_period_start := date_trunc('month', now() AT TIME ZONE 'UTC');
  v_period_end   := v_period_start + interval '1 month';

  SELECT COALESCE(jsonb_object_agg(meter, jsonb_build_object(
    'used',   used,
    'quota',  quota
  )), '{}')
  INTO v_meters
  FROM public.usage_meters
  WHERE studio_id   = p_studio_id
    AND period_start = v_period_start;

  RETURN jsonb_build_object(
    'subscription', to_jsonb(v_sub),
    'plan',         to_jsonb(v_plan),
    'usage',        v_meters
  );
END;
$$;

-- =============================================================
-- SECTION 12 — RPC: get_billing_summary
-- =============================================================
-- Studio portal summary: current plan, next renewal, last 3 invoices,
-- trial status, cancel_at_period_end flag, current-period usage.
-- =============================================================

CREATE OR REPLACE FUNCTION public.get_billing_summary(
  p_studio_id uuid
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
STABLE
SET search_path = public
AS $$
DECLARE
  v_sub           public.subscriptions%ROWTYPE;
  v_plan          public.plans%ROWTYPE;
  v_invoices      jsonb;
  v_meters        jsonb;
  v_period_start  timestamptz;
BEGIN
  -- Verify caller is a member of this studio
  IF NOT EXISTS (
    SELECT 1 FROM public.studio_members
    WHERE studio_id = p_studio_id AND user_id = auth.uid()
  ) THEN
    RAISE EXCEPTION 'access_denied' USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_sub
  FROM public.subscriptions
  WHERE studio_id = p_studio_id
  LIMIT 1;

  SELECT * INTO v_plan
  FROM public.plans
  WHERE tier          = COALESCE(v_sub.plan_tier, 'solo')
    AND billing_cycle = COALESCE(v_sub.billing_cycle, 'monthly')
    AND is_active     = true
  LIMIT 1;

  -- Last 3 invoices
  SELECT COALESCE(jsonb_agg(
    jsonb_build_object(
      'id',           i.id,
      'dodo_invoice_id', i.dodo_invoice_id,
      'amount_cents', i.amount_cents,
      'status',       i.status,
      'paid_at',      i.paid_at,
      'pdf_url',      i.pdf_url,
      'hosted_url',   i.hosted_url
    )
    ORDER BY i.created_at DESC
  ), '[]')
  INTO v_invoices
  FROM (
    SELECT * FROM public.invoices
    WHERE studio_id = p_studio_id
    ORDER BY created_at DESC
    LIMIT 3
  ) i;

  -- Current period usage
  v_period_start := date_trunc('month', now() AT TIME ZONE 'UTC');

  SELECT COALESCE(jsonb_object_agg(meter, jsonb_build_object(
    'used',  used,
    'quota', quota
  )), '{}')
  INTO v_meters
  FROM public.usage_meters
  WHERE studio_id    = p_studio_id
    AND period_start = v_period_start;

  RETURN jsonb_build_object(
    'plan',                to_jsonb(v_plan),
    'status',              v_sub.status,
    'billing_cycle',       v_sub.billing_cycle,
    'current_period_end',  v_sub.current_period_end,
    'cancel_at_period_end', v_sub.cancel_at_period_end,
    'trial_end',           v_sub.trial_end,
    'in_trial',            (v_sub.status = 'trialing' AND v_sub.trial_end > now()),
    'last_invoices',       v_invoices,
    'current_usage',       v_meters
  );
END;
$$;

-- 13a: bookings INSERT → increment bookings_created (non-blocking)
CREATE OR REPLACE FUNCTION public.handle_booking_meter()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  -- Non-blocking: catch any error and log; never abort the booking INSERT
  BEGIN
    PERFORM public.increment_usage_meter(NEW.studio_id, 'bookings_created', 1);
  EXCEPTION WHEN OTHERS THEN
    -- Silent failure: meter increment is best-effort
    NULL;
  END;
  RETURN NEW;
END;
$$;

CREATE OR REPLACE TRIGGER on_booking_inserted_meter
  AFTER INSERT ON public.bookings
  FOR EACH ROW
  EXECUTE FUNCTION public.handle_booking_meter();

-- 13b: artists INSERT/UPDATE → SET active_artists to exact count
-- (set, not increment, to stay accurate after soft-deletes / status changes)
CREATE OR REPLACE FUNCTION public.handle_artist_meter()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_count         int;
  v_period_start  timestamptz;
  v_period_end    timestamptz;
  v_studio_id     uuid;
BEGIN
  -- Work on the studio that changed
  v_studio_id := COALESCE(NEW.studio_id, OLD.studio_id);

  -- Only recalculate when the status column touches 'active'
  IF (TG_OP = 'INSERT' AND NEW.status = 'active')
  OR (TG_OP = 'UPDATE' AND (NEW.status = 'active' OR OLD.status = 'active'))
  THEN
    v_period_start := date_trunc('month', now() AT TIME ZONE 'UTC');
    v_period_end   := v_period_start + interval '1 month';

    SELECT COUNT(*)
    INTO v_count
    FROM public.artists
    WHERE studio_id = v_studio_id
      AND status    = 'active';

    INSERT INTO public.usage_meters
      (studio_id, period_start, period_end, meter, used)
    VALUES
      (v_studio_id, v_period_start, v_period_end, 'active_artists', v_count)
    ON CONFLICT (studio_id, period_start, meter) DO UPDATE
      SET used       = v_count,
          updated_at = now();
  END IF;

  RETURN COALESCE(NEW, OLD);
END;
$$;

CREATE OR REPLACE TRIGGER on_artist_active_meter
  AFTER INSERT OR UPDATE OF status ON public.artists
  FOR EACH ROW
  EXECUTE FUNCTION public.handle_artist_meter();

-- 13c: locations INSERT → SET active_locations to exact count

AS $$

  ON CONFLICT (studio_id, period_start, meter) DO UPDATE
    SET used       = v_count,
        updated_at = now();

CREATE OR REPLACE TRIGGER on_location_inserted_meter
