   390→  const rankAndPoll = async (jobId: string, resumeIds: string[]) => {
   391→    const response = await fetch(
   392→      `https://zkzontawprihrzpayeam.supabase.co/functions/v1/rank-resumes`,
   393→      {
   394→        method: "POST",
   395→        headers: {
   396→          "Content-Type": "application/json",
   397→          Authorization: `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
   398→        },
   399→        body: JSON.stringify({ job_id: jobId, resume_ids: resumeIds }),
   400→      }
   401→    );
   402→
   403→    if (!response.ok) {
   404→      const errBody = await response.json().catch(() => ({}));
   405→      const err: any = new Error(errBody.error || "Ranking failed");
   406→      err.status = response.status;
   407→      throw err;
   408→    }

   412→      await new Promise((r) => setTimeout(r, 2000));

   414→      if (resumeProgress) {
   415→        const done = resumeProgress.filter((r: any) => r.status === "done" || r.status === "failed").length;
   416→        setScoredCount(done);
   417→        const pct = resumeIds.length > 0 ? Math.round((done / resumeIds.length) * 35) : 0;
   418→        setRankingProgress(Math.min(60 + pct, 95));
   419→      }
   420→
   421→      const { data: jobData } = await supabase.from("jobs").select("status").eq("id", jobId).single();
   422→      if (jobData?.status === "completed") {
   423→         const { data: allResumes } = await supabase.from("resumes").select("id, filename, file_path, candidate_name, parsed_json").eq("job_id", jobId);
   424→         if (allResumes) {
   425→           const allResumeIds = allResumes.map((r: any) => r.id);
   426→            const { data: resultData } = await supabase
   427→              .from("results")
   428→              .select("id, resume_id, overall_score, missing_skills, summary, interview_questions, hard_fit, category_scores_json, confidence, strengths_json, gaps_json, explanation_json, matched_must_haves, missing_must_haves, scoring_version")
   429→              .in("resume_id", allResumeIds);
   430→
   431→            const resumeMap = new Map(allResumes.map((r: any) => [r.id, r]));
   432→            const merged: Result[] = (resultData ?? []).map((r: any) => {
   433→              const resume = resumeMap.get(r.resume_id);
   434→              const parsed = resume?.parsed_json as any;
   435→              return {
   436→                ...r,
   437→                overall_score: r.overall_score ?? 0,
   438→                missing_skills: r.missing_skills ?? [],
   439→                filename: resume?.filename ?? "Unknown",
   440→                file_path: resume?.file_path ?? null,
   441→                candidate_name: resume?.candidate_name ?? null,
   442→                candidate_email: parsed?.email ?? parsed?.contact?.email ?? null,
   443→                candidate_phone: parsed?.phone ?? parsed?.contact?.phone ?? null,
   444→                candidate_location: parsed?.location ?? parsed?.contact?.location ?? null,
   445→              };
   446→            });
   447→          merged.sort((a, b) => b.overall_score - a.overall_score);
   448→          setResults(merged);
   449→        }
   450→        setRankingProgress(100);
   451→        await refreshProfile();
   452→        setTimeout(() => { resultsRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }); }, 200);
   453→        return;
   454→      }
   455→      if (jobData?.status === "failed") throw new Error("Ranking failed on server");
   456→    }
   457→    throw new Error("Ranking timed out");
   458→  };

   460→  const handleRerank = async () => {
   461→    if (!user || !selectedJobId || existingResumes.length === 0) return;

   470→      for (const rid of resumeIds) {
   471→        await supabase.from("resumes").update({ parsed_text: null, parsed_json: null } as any).eq("id", rid);
   472→      }
   473→      setResults([]); setRankingProgress(30);
