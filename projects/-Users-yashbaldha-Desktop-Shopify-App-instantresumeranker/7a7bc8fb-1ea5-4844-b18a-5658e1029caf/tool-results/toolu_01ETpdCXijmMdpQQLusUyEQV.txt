   125→    setExistingResumes((prev: ExistingResume[]) => prev.filter((r) => r.id !== resumeId));

   128→
   129→  const handleDeleteConfirm = async () => {
   130→    if (!deleteConfirm) return;
   131→    setDeleting(true);
   132→    try {
   133→      if (deleteConfirm.type === "single" && deleteConfirm.resumeId) {
   134→        await deleteResume(deleteConfirm.resumeId, deleteConfirm.filePath ?? null);
   135→        toast.success("Resume deleted");

   137→        const toDelete = allResults.filter((r) => selectedResumes.has(r.resume_id));

   139→        toast.success(`${toDelete.length} resume(s) deleted`);
   140→      }

   142→    finally { setDeleting(false); setDeleteConfirm(null); }

   144→

   148→    const dropped = Array.from(e.dataTransfer.files).filter(
   149→      (f) => f.type === "application/pdf" || f.type === "application/vnd.openxmlformats-officedocument.wordprocessingml.document"

   158→  const addFiles = (newFiles: File[]) => {

   160→    const valid = newFiles.filter((f) => {
   161→      if (f.size > maxFileSizeMb * 1024 * 1024) { toast.error(`${f.name} exceeds ${maxFileSizeMb}MB limit`); return false; }
   162→      return true;
   163→    });
   164→    const remaining = maxResumes - currentCount;
   165→    if (valid.length > remaining) toast.error(`Max ${maxResumes} resumes per job`);
   166→    const toAdd = valid.slice(0, Math.max(0, remaining)).map((f) => ({ file: f, name: f.name }));
   167→    setFiles((prev) => [...prev, ...toAdd]);

   169→
   170→  const removeFile = (index: number) => setFiles((prev) => prev.filter((_, i) => i !== index));
   171→  const toggleRow = (id: string) => {
   172→    setExpandedRows((prev) => { const next = new Set(prev); if (next.has(id)) next.delete(id); else next.add(id); return next; });
   173→  };

   179→  const validate = (): string[] => {

   182→    const filesToCheck = files.filter(f => !rankingFilesRef.current.has(f.name));
   183→    if (filesToCheck.length === 0) errors.push("Upload at least one resume.");
   184→    if (user && (profile?.credits ?? 0) < Math.min(filesToCheck.length, 20)) {

   190→  // ── Analyze JD ──────────────────────────────────────────────
   191→  const analyzeJd = async (jobId: string) => {

   195→        `https://zkzontawprihrzpayeam.supabase.co/functions/v1/analyze-jd`,

   199→            "Content-Type": "application/json",
   200→            Authorization: `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
   201→          },
   202→          body: JSON.stringify({ job_id: jobId }),
   203→        }
   204→      );
   205→      if (!response.ok) {
   206→        const err = await response.json().catch(() => ({}));
   207→        throw new Error(err.error || "JD analysis failed");
   208→      }
   209→      const data = await response.json();
   210→      if (onJobUpdated) {
   211→        onJobUpdated(jobId, {
   212→          jd_analysis_json: data.jd_analysis,
   213→          dynamic_weights_json: data.dynamic_weights,

   217→    } catch (err: any) {
   218→      toast.error(err.message || "JD analysis failed");

   221→      setAnalyzingJd(false);
   222→    }

   224→
   225→  // ── Load dismissed resumes from localStorage when job changes ──
   226→  useEffect(() => {
   227→    if (selectedJobId) {
   228→      try {
   229→        const stored = localStorage.getItem(`dismissed_${selectedJobId}`);
   230→        if (stored) {
   231→          setDismissedResumes(new Set(JSON.parse(stored) as string[]));
   232→        } else {

   236→        setDismissedResumes(new Set());
   237→      }
   238→    } else {
   239→      setDismissedResumes(new Set());
   240→    }

   243→  // ── Auto-rank ───────────────────────────────────────────────
   244→  const autoRankTriggered = useRef(false);
   245→  useEffect(() => {

   247→    if (unrankedFiles.length > 0 && !ranking && !autoRankTriggered.current) {
   248→      if (isNewJobMode && !jobDescription.trim()) return;
   249→      autoRankTriggered.current = true;

   252→    if (files.length === 0 && !ranking) autoRankTriggered.current = false;

   254→
   255→  const handleRankingError = (err: any) => {
   256→    const msg: string = err?.message || "Ranking failed.";
   257→    if (msg.includes("Insufficient credits") || err?.status === 402) {
   258→      setRankingError({ type: "credits", message: "You don't have enough credits to rank these resumes." });
   259→    } else if (msg.includes("Rate limit") || err?.status === 429) {
   260→      setRankingError({ type: "rate_limit", message: "Too many jobs in a short time. Please wait a minute and try again." });
   261→    } else {
   262→      setRankingError({ type: "generic", message: msg });
   263→      toast.error(msg);
   264→    }

   267→  const handleRankClick = () => {
   268→    if (!user) { onAuthRequired(); return; }
   269→    setRankingError(null);
   270→    const errors = validate();
   271→    setValidationErrors(errors);
   272→    if (errors.length > 0) return;
   273→    if (isExistingJob) handleRankExisting(); else handleRankNew();

   275→
   276→  const handleRankNew = async () => {
   277→    if (!user) return;
   278→    const filesToRank = [...files];
   279→    rankingFilesRef.current = new Set(filesToRank.map(f => f.name));
   280→    setRanking(true); setResults([]); setRankingError(null);

   287→      })();

   289→      const { data: job, error: jobError } = await supabase
   290→        .from("jobs")
   291→        .insert({ user_id: user.id, job_description: jobDescription, status: "pending", title: autoTitle } as any)
   292→        .select("id")
   293→        .single();

   295→      setRankingProgress(10);
   296→
   297→      const analysisData = await analyzeJd(job.id);

   307→        jd_analysis_json: analysisData.jd_analysis,
   308→        dynamic_weights_json: analysisData.dynamic_weights,
   309→      });
   310→
   311→      const resumeIds = await uploadFilesFromSnapshot(job.id, filesToRank);
   312→      setRankingProgress(40);
   313→
   314→      await rankAndPoll(job.id, resumeIds);
   315→
   316→      setJobDescription("");
   317→      setFiles(prev => prev.filter(f => !rankingFilesRef.current.has(f.name)));
   318→      toast.success("Candidates ranked!");
   319→    } catch (err: any) {
   320→      handleRankingError(err);
   321→    } finally {
   322→      rankingFilesRef.current = new Set();
   323→      setRanking(false); setRankingProgress(0); setScoredCount(0); setTotalToScore(0);
   324→      autoRankTriggered.current = false;
   325→    }
   326→  };
   327→
   328→  const handleRankExisting = async () => {
   329→    if (!user || !selectedJobId) return;
   330→    const filesToRank = [...files];
   331→    rankingFilesRef.current = new Set(filesToRank.map(f => f.name));

   347→        .from("resumes")
   348→        .select("id, filename, file_path, candidate_name, created_at")

   350→        .order("created_at", { ascending: false });

   356→      handleRankingError(err);
   357→    } finally {

   364→  const uploadFilesFromSnapshot = async (jobId: string, filesToUpload: UploadedFile[]): Promise<string[]> => {
   365→    if (!user) return [];
   366→    const resumeIds: string[] = [];
   367→    for (const { file, name } of filesToUpload) {
   368→      const filePath = `${user.id}/${jobId}/${Date.now()}_${name}`;
   369→      const { error: uploadError } = await supabase.storage.from("resumes").upload(filePath, file);
   370→      if (uploadError) throw new Error(`Upload failed: ${uploadError.message}`);
   371→      const { data: resume, error: resumeError } = await supabase
   372→        .from("resumes")
   373→        .insert({ job_id: jobId, filename: name, file_path: filePath })
   374→        .select("id")
   375→        .single();
   376→      if (resumeError || !resume) throw new Error(resumeError?.message || "Failed to save resume");
   377→      resumeIds.push(resume.id);
   378→    }
   379→    return resumeIds;
   380→  };
   381→
   382→  const rankAndPoll = async (jobId: string, resumeIds: string[]) => {
   383→    const response = await fetch(
   384→      `https://zkzontawprihrzpayeam.supabase.co/functions/v1/rank-resumes`,
   385→      {
   386→        method: "POST",
   387→        headers: {
   388→          "Content-Type": "application/json",
   389→          Authorization: `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
   390→        },
   391→        body: JSON.stringify({ job_id: jobId, resume_ids: resumeIds }),
   392→      }
   393→    );
   394→
   395→    if (!response.ok) {
   396→      const errBody = await response.json().catch(() => ({}));
   397→      const err: any = new Error(errBody.error || "Ranking failed");
   398→      err.status = response.status;
   399→      throw err;
   400→    }
   401→    setRankingProgress(60);
   402→
   403→    for (let i = 0; i < 150; i++) {
   404→      await new Promise((r) => setTimeout(r, 2000));
   405→      const { data: resumeProgress } = await supabase.from("resumes").select("id, status").in("id", resumeIds);
   406→      if (resumeProgress) {
   407→        const done = resumeProgress.filter((r: any) => r.status === "done" || r.status === "failed").length;
   408→        setScoredCount(done);
   409→        const pct = resumeIds.length > 0 ? Math.round((done / resumeIds.length) * 35) : 0;
   410→        setRankingProgress(Math.min(60 + pct, 95));
   411→      }
   412→
   413→      const { data: jobData } = await supabase.from("jobs").select("status").eq("id", jobId).single();
   414→      if (jobData?.status === "completed") {
   415→         const { data: allResumes } = await supabase.from("resumes").select("id, filename, file_path, candidate_name, parsed_json").eq("job_id", jobId);
   416→         if (allResumes) {
   417→           const allResumeIds = allResumes.map((r: any) => r.id);
   418→            const { data: resultData } = await supabase
   419→              .from("results")
   420→              .select("id, resume_id, overall_score, missing_skills, summary, interview_questions, hard_fit, category_scores_json, confidence, strengths_json, gaps_json, explanation_json, matched_must_haves, missing_must_haves, scoring_version")
   421→              .in("resume_id", allResumeIds);
   422→
   423→            const resumeMap = new Map(allResumes.map((r: any) => [r.id, r]));
   424→            const merged: Result[] = (resultData ?? []).map((r: any) => {
   425→              const resume = resumeMap.get(r.resume_id);
   426→              const parsed = resume?.parsed_json as any;
   427→              return {
   428→                ...r,
   429→                overall_score: r.overall_score ?? 0,
   430→                missing_skills: r.missing_skills ?? [],
   431→                filename: resume?.filename ?? "Unknown",
   432→                file_path: resume?.file_path ?? null,
   433→                candidate_name: resume?.candidate_name ?? null,
   434→                candidate_email: parsed?.email ?? parsed?.contact?.email ?? null,
   435→                candidate_phone: parsed?.phone ?? parsed?.contact?.phone ?? null,
   436→                candidate_location: parsed?.location ?? parsed?.contact?.location ?? null,
   437→              };
   438→            });
   439→          merged.sort((a, b) => b.overall_score - a.overall_score);
   440→          setResults(merged);
   441→        }
   442→        setRankingProgress(100);
   443→        await refreshProfile();
   444→        setTimeout(() => { resultsRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }); }, 200);
   445→        return;
   446→      }
   447→      if (jobData?.status === "failed") throw new Error("Ranking failed on server");
   448→    }
   449→    throw new Error("Ranking timed out");
   450→  };
   451→
   452→  const handleRerank = async () => {
   453→    if (!user || !selectedJobId || existingResumes.length === 0) return;
   454→    setRanking(true); setRankingProgress(10);
   455→    try {
   456→      if (!selectedJob?.jd_analysis_json) {
   457→        await analyzeJd(selectedJobId);
   458→        setRankingProgress(20);
   459→      }
   460→      const resumeIds = existingResumes.map((r) => r.id);
   461→      await supabase.from("results").delete().in("resume_id", resumeIds);
   462→      for (const rid of resumeIds) {
   463→        await supabase.from("resumes").update({ parsed_text: null, parsed_json: null } as any).eq("id", rid);
   464→      }
   465→      setResults([]); setRankingProgress(30);
   466→      await supabase.from("jobs").update({ status: "processing" }).eq("id", selectedJobId);
   467→      await rankAndPoll(selectedJobId, resumeIds);
   468→      const { data: allResumes } = await supabase
   469→        .from("resumes")
   470→        .select("id, filename, file_path, candidate_name, created_at")
   471→        .eq("job_id", selectedJobId)
   472→        .order("created_at", { ascending: false });
   473→      if (allResumes) setExistingResumes(allResumes);

   916→                    )}
   917→                    {isExistingJob && existingResumes.length > 0 && (
   918→                      <Button variant="outline" size="sm" className="gap-1.5" disabled={ranking} onClick={handleRerank}>
   919→                        {ranking ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <RefreshCw className="h-3.5 w-3.5" />}
   920→                        Re-rank
   921→                      </Button>
   922→                    )}
   923→                  </div>
   924→                </div>
   925→              </CardHeader>
   926→              <CardContent className="p-0">
   927→                <div className="overflow-x-auto">
   928→                  <Table>
   929→                    <TableHeader className="sticky top-0 bg-card z-10">
   930→                      <TableRow>
   931→                        <TableHead className="w-10">
   932→                          <Checkbox checked={visibleResults.length > 0 && selectedResumes.size === visibleResults.length} onCheckedChange={toggleSelectAll} />
   933→                        </TableHead>
   934→                        <TableHead className="w-14 text-center">#</TableHead>
   935→                        <TableHead>Candidate</TableHead>
   936→                        <TableHead className="w-32">Match</TableHead>
   937→                        <TableHead className="w-20">Strength</TableHead>
   938→                        <TableHead className="w-36"></TableHead>
   939→                      </TableRow>
   940→                    </TableHeader>
   941→                    <TableBody>
   942→                      {visibleResults.map((r, index) => (
   943→                        <CandidateRow
   944→                          key={r.id}
   945→                          result={r}
   946→                          index={index}
   947→                          isExpanded={expandedRows.has(r.id)}
   948→                          isSelected={selectedResumes.has(r.resume_id)}
   949→                          jobTitle={selectedJob?.title}
   950→                          onToggleRow={toggleRow}
   951→                          onToggleSelect={toggleSelectResume}
   952→                          onDownload={downloadResume}
   953→                          onDelete={(resumeId, filePath) => setDeleteConfirm({ type: "single", resumeId, filePath })}
   954→                          onDismiss={handleDismiss}
   955→                          formatCandidateName={formatCandidateName}
   956→                        />
   957→                      ))}
   958→                    </TableBody>
   959→                  </Table>
   960→                </div>
   961→              </CardContent>
   962→            </Card>
   963→          </div>
   964→        )}
   965→      </div>
   966→
   967→      {/* Delete Confirmation */}
   968→      <AlertDialog open={!!deleteConfirm} onOpenChange={(open) => !open && setDeleteConfirm(null)}>
   969→        <AlertDialogContent>
   970→          <AlertDialogHeader>
   971→            <AlertDialogTitle>Delete {deleteCount} resume{deleteCount > 1 ? "s" : ""}?</AlertDialogTitle>
   972→            <AlertDialogDescription>
   973→              This will permanently delete the resume{deleteCount > 1 ? "s" : ""} and any associated results.
   974→            </AlertDialogDescription>
   975→          </AlertDialogHeader>
   976→          <AlertDialogFooter>
   977→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   978→            <AlertDialogAction onClick={handleDeleteConfirm} disabled={deleting} className="bg-destructive text-destructive-foreground hover:bg-destructive/90">
   979→              {deleting ? <Loader2 className="h-4 w-4 animate-spin mr-2" /> : null}
   980→              Delete
   981→            </AlertDialogAction>
   982→          </AlertDialogFooter>
   983→        </AlertDialogContent>
   984→      </AlertDialog>
   985→    </div>
   986→  );
   987→};
