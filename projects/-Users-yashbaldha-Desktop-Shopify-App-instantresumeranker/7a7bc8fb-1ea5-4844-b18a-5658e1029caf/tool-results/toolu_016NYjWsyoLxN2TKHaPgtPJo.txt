   181→  const validate = (): string[] => {

   187→    const totalCredits = splitTotal > 0 ? splitTotal : (profile?.credits ?? 0);
   188→    if (user && totalCredits < Math.min(filesToCheck.length, 20)) {
   189→      errors.push(`Not enough credits. You need ${Math.min(filesToCheck.length, 20)} but have ${totalCredits}.`);

   206→          body: JSON.stringify({ job_id: jobId }),
   207→        }
   208→      );

   210→        const err = await response.json().catch(() => ({}));
   211→        throw new Error(err.error || "JD analysis failed");
   212→      }
   213→      const data = await response.json();
   214→      if (onJobUpdated) {
   215→        onJobUpdated(jobId, {
   216→          jd_analysis_json: data.jd_analysis,
   217→          dynamic_weights_json: data.dynamic_weights,

   219→      }
   220→      return data;
   221→    } catch (err: any) {
   222→      toast.error(err.message || "JD analysis failed");
   223→      throw err;
   224→    } finally {
   225→      setAnalyzingJd(false);
   226→    }
   227→  };
   228→
   229→  // ── Load dismissed resumes from localStorage when job changes ──
   230→  useEffect(() => {

   233→      try {
   234→        const stored = localStorage.getItem(`dismissed_${selectedJobId}`);
   235→        if (stored) {

   249→  const autoRankTriggered = useRef(false);

   260→  const handleRankingError = (err: any) => {
   261→    const msg: string = err?.message || "Ranking failed.";
   262→    if (msg.includes("Insufficient credits") || err?.status === 402) {
   263→      setRankingError({ type: "credits", message: "You don't have enough credits to rank these resumes." });
   264→    } else if (msg.includes("Rate limit") || err?.status === 429) {
   265→      setRankingError({ type: "rate_limit", message: "Too many jobs in a short time. Please wait a minute and try again." });

   267→      setRankingError({ type: "generic", message: msg });

   269→    }

   277→    if (errors.length > 0) return;
   278→    if (isExistingJob) handleRankExisting(); else handleRankNew();

   280→
   281→  const handleRankNew = async () => {
   282→    if (!user) return;
   283→    const filesToRank = [...files];

   292→      })();
   293→
   294→      const { data: job, error: jobError } = await supabase
   295→        .from("jobs")
   296→        .insert({ user_id: user.id, job_description: jobDescription, status: "pending", title: autoTitle } as any)
   297→        .select("id, inbox_email")
   298→        .single();
   299→      if (jobError || !job) throw new Error(jobError?.message || "Failed to create job");
   300→      setRankingProgress(10);
   301→

   311→        resume_count: filesToRank.length,
   312→        jd_analysis_json: analysisData.jd_analysis,
   313→        dynamic_weights_json: analysisData.dynamic_weights,
   314→        inbox_email: job.inbox_email ?? null,
   315→      });
   316→
   317→      const resumeIds = await uploadFilesFromSnapshot(job.id, filesToRank);
   318→      setRankingProgress(40);
   319→
   320→      await rankAndPoll(job.id, resumeIds);
   321→
   322→      setJobDescription("");
   323→      setFiles(prev => prev.filter(f => !rankingFilesRef.current.has(f.name)));
   324→      toast.success("Candidates ranked!");
   325→    } catch (err: any) {
   326→      handleRankingError(err);
   327→    } finally {
   328→      rankingFilesRef.current = new Set();
   329→      setRanking(false); setRankingProgress(0); setScoredCount(0); setTotalToScore(0);
   330→      autoRankTriggered.current = false;

   370→  const uploadFilesFromSnapshot = async (jobId: string, filesToUpload: UploadedFile[]): Promise<string[]> => {

   375→      const { error: uploadError } = await supabase.storage.from("resumes").upload(filePath, file);
   376→      if (uploadError) throw new Error(`Upload failed: ${uploadError.message}`);
   377→      const { data: resume, error: resumeError } = await supabase
   378→        .from("resumes")
   379→        .insert({ job_id: jobId, filename: name, file_path: filePath })
   380→        .select("id")
   381→        .single();
   382→      if (resumeError || !resume) throw new Error(resumeError?.message || "Failed to save resume");
   383→      resumeIds.push(resume.id);
   384→    }
   385→    return resumeIds;

   388→  const rankAndPoll = async (jobId: string, resumeIds: string[]) => {
   389→    const response = await fetch(
   390→      `https://zkzontawprihrzpayeam.supabase.co/functions/v1/rank-resumes`,
   391→      {
   392→        method: "POST",
   393→        headers: {
   394→          "Content-Type": "application/json",
   395→          Authorization: `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
   396→        },
   397→        body: JSON.stringify({ job_id: jobId, resume_ids: resumeIds }),
   398→      }
   399→    );
   400→
   401→    if (!response.ok) {
   402→      const errBody = await response.json().catch(() => ({}));
   403→      const err: any = new Error(errBody.error || "Ranking failed");
   404→      err.status = response.status;
   405→      throw err;
   406→    }
   407→    setRankingProgress(60);
   408→
   409→    for (let i = 0; i < 150; i++) {
   410→      await new Promise((r) => setTimeout(r, 2000));
   411→      const { data: resumeProgress } = await supabase.from("resumes").select("id, status").in("id", resumeIds);
   412→      if (resumeProgress) {
   413→        const done = resumeProgress.filter((r: any) => r.status === "done" || r.status === "failed").length;
   414→        setScoredCount(done);
   415→        const pct = resumeIds.length > 0 ? Math.round((done / resumeIds.length) * 35) : 0;
   416→        setRankingProgress(Math.min(60 + pct, 95));
   417→      }
   418→
   419→      const { data: jobData } = await supabase.from("jobs").select("status").eq("id", jobId).single();
   420→      if (jobData?.status === "completed") {
   421→         const { data: allResumes } = await supabase.from("resumes").select("id, filename, file_path, candidate_name, parsed_json").eq("job_id", jobId);
   422→         if (allResumes) {
   423→           const allResumeIds = allResumes.map((r: any) => r.id);
   424→            const { data: resultData } = await supabase
   425→              .from("results")
   426→              .select("id, resume_id, overall_score, missing_skills, summary, interview_questions, hard_fit, category_scores_json, confidence, strengths_json, gaps_json, explanation_json, matched_must_haves, missing_must_haves, scoring_version")
   427→              .in("resume_id", allResumeIds);
   428→
   429→            const resumeMap = new Map(allResumes.map((r: any) => [r.id, r]));
   430→            const merged: Result[] = (resultData ?? []).map((r: any) => {
   431→              const resume = resumeMap.get(r.resume_id);
   432→              const parsed = resume?.parsed_json as any;
   433→              return {
   434→                ...r,
   435→                overall_score: r.overall_score ?? 0,
   436→                missing_skills: r.missing_skills ?? [],
   437→                filename: resume?.filename ?? "Unknown",
   438→                file_path: resume?.file_path ?? null,
   439→                candidate_name: resume?.candidate_name ?? null,
   440→                candidate_email: parsed?.email ?? parsed?.contact?.email ?? null,
   441→                candidate_phone: parsed?.phone ?? parsed?.contact?.phone ?? null,
   442→                candidate_location: parsed?.location ?? parsed?.contact?.location ?? null,
   443→              };
   444→            });
   445→          merged.sort((a, b) => b.overall_score - a.overall_score);
   446→          setResults(merged);
   447→        }
   448→        setRankingProgress(100);
   449→        await refreshProfile();
   450→        setTimeout(() => { resultsRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }); }, 200);
   451→        return;
   452→      }
   453→      if (jobData?.status === "failed") throw new Error("Ranking failed on server");
   454→    }
   455→    throw new Error("Ranking timed out");
   456→  };
   457→
   458→  const handleRerank = async () => {
   459→    if (!user || !selectedJobId || existingResumes.length === 0) return;
   460→    setRanking(true); setRankingProgress(10);
   461→    try {
   462→      if (!selectedJob?.jd_analysis_json) {
   463→        await analyzeJd(selectedJobId);
   464→        setRankingProgress(20);
   465→      }
   466→      const resumeIds = existingResumes.map((r) => r.id);
   467→      await supabase.from("results").delete().in("resume_id", resumeIds);
   468→      for (const rid of resumeIds) {
   469→        await supabase.from("resumes").update({ parsed_text: null, parsed_json: null } as any).eq("id", rid);
   470→      }
   471→      setResults([]); setRankingProgress(30);
   472→      await supabase.from("jobs").update({ status: "processing" }).eq("id", selectedJobId);
   473→      await rankAndPoll(selectedJobId, resumeIds);

   921→                            if (UUID_REGEX.test(name)) return "Candidate";

   943→                      >

   945→                        Export CSV
   946→                      </Button>
   947→                    )}
   948→                    {isExistingJob && existingResumes.length > 0 && (
   949→                      <Button variant="outline" size="sm" className="gap-1.5" disabled={ranking} onClick={handleRerank}>
   950→                        {ranking ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <RefreshCw className="h-3.5 w-3.5" />}
   951→                        Re-rank
   952→                      </Button>
   953→                    )}
   954→                  </div>
   955→                </div>
   956→              </CardHeader>

   958→                <div className="overflow-x-auto">
   959→                  <Table>
   960→                    <TableHeader className="sticky top-0 bg-card z-10">
   961→                      <TableRow>
   962→                        <TableHead className="w-10">
   963→                          <Checkbox checked={visibleResults.length > 0 && selectedResumes.size === visibleResults.length} onCheckedChange={toggleSelectAll} />
   964→                        </TableHead>
   965→                        <TableHead className="w-14 text-center">#</TableHead>

   967→                        <TableHead className="w-32">Match</TableHead>
   968→                        <TableHead className="w-20">Strength</TableHead>
   969→                        <TableHead className="w-36"></TableHead>
   970→                      </TableRow>
   971→                    </TableHeader>
   972→                    <TableBody>
   973→                      {visibleResults.map((r, index) => (
   974→                        <CandidateRow
   975→                          key={r.id}
   976→                          result={r}
   977→                          index={index}
   978→                          isExpanded={expandedRows.has(r.id)}

   980→                          jobTitle={selectedJob?.title}
   981→                          onToggleRow={toggleRow}
   982→                          onToggleSelect={toggleSelectResume}
   983→                          onDownload={downloadResume}
   984→                          onDelete={(resumeId, filePath) => setDeleteConfirm({ type: "single", resumeId, filePath })}
   985→                          onDismiss={handleDismiss}
   986→                          formatCandidateName={formatCandidateName}
   987→                        />
   988→                      ))}
   989→                    </TableBody>
   990→                  </Table>
   991→                </div>
   992→              </CardContent>
   993→            </Card>
   994→
   995→            {/* Archived Candidates Panel */}
   996→            {showArchived && archivedResults.length > 0 && (
   997→              <Card className="mt-3 border-amber-200 dark:border-amber-800">
   998→                <CardHeader className="pb-3">
   999→                  <div className="flex items-center justify-between">
  1000→                    <CardTitle className="text-base flex items-center gap-2 text-amber-700 dark:text-amber-400">
  1001→                      <Archive className="h-4 w-4" />
  1002→                      Archived Candidates ({archivedResults.length})
  1003→                    </CardTitle>
  1004→                    <button

  1006→                      className="text-muted-foreground hover:text-foreground"
  1007→                    >
  1008→                      <X className="h-4 w-4" />
  1009→                    </button>
  1010→                  </div>
  1011→                </CardHeader>
  1012→                <CardContent className="p-0">
  1013→                  <div className="overflow-x-auto">
  1014→                    <Table>

  1017→                          <TableHead className="w-14 text-center pl-4">#</TableHead>

  1033→                                {r.candidate_name || formatCandidateName(r.filename)}
  1034→                              </TableCell>

  1052→                                <Button

  1054→                                  size="icon"

  1065→                      </TableBody>

  1070→            )}
  1071→          </div>
  1072→        )}
  1073→      </div>
  1074→
  1075→      {/* Delete Confirmation */}
  1076→      <AlertDialog open={!!deleteConfirm} onOpenChange={(open) => !open && setDeleteConfirm(null)}>

  1089→            </AlertDialogAction>
  1090→          </AlertDialogFooter>
  1091→        </AlertDialogContent>

  1093→    </div>
  1094→  );
  1095→};
