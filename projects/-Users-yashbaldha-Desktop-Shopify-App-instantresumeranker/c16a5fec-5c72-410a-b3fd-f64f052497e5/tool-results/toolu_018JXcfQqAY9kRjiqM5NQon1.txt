   399→      const { data: resume, error: resumeError } = await supabase
   400→        .from("resumes")
   401→        .insert({ job_id: jobId, filename: name, file_path: filePath })
   402→        .select("id")
   403→        .single();
   404→      if (resumeError || !resume) throw new Error(resumeError?.message || "Failed to save resume");
   405→      resumeIds.push(resume.id);
   406→    }

   410→  const rankAndPoll = async (jobId: string, resumeIds: string[]) => {
   411→    const response = await fetch(
   412→      `https://zkzontawprihrzpayeam.supabase.co/functions/v1/rank-resumes`,
   413→      {
   414→        method: "POST",
   415→        headers: {
   416→          "Content-Type": "application/json",
   417→          Authorization: `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
   418→        },
   419→        body: JSON.stringify({ job_id: jobId, resume_ids: resumeIds }),
   420→      }
   421→    );
   422→
   423→    if (!response.ok) {
   424→      const errBody = await response.json().catch(() => ({}));
   425→      const err: any = new Error(errBody.error || "Ranking failed");
   426→      err.status = response.status;
   427→      throw err;
   428→    }
   429→    setRankingProgress(60);
   430→
   431→    for (let i = 0; i < 150; i++) {
   432→      await new Promise((r) => setTimeout(r, 2000));
   433→      const { data: resumeProgress } = await supabase.from("resumes").select("id, status").in("id", resumeIds);
   434→      if (resumeProgress) {
   435→        const done = resumeProgress.filter((r: any) => r.status === "done" || r.status === "failed").length;
   436→        setScoredCount(done);
   437→        const pct = resumeIds.length > 0 ? Math.round((done / resumeIds.length) * 35) : 0;
   438→        setRankingProgress(Math.min(60 + pct, 95));
   439→      }
   440→
   441→      const { data: jobData } = await supabase.from("jobs").select("status").eq("id", jobId).single();
   442→      if (jobData?.status === "completed") {
   443→         const { data: allResumes } = await supabase.from("resumes").select("id, filename, file_path, candidate_name, parsed_json").eq("job_id", jobId);
   444→         if (allResumes) {
   445→           const allResumeIds = allResumes.map((r: any) => r.id);
   446→            const { data: resultData } = await supabase
   447→              .from("results")
   448→              .select("id, resume_id, overall_score, missing_skills, summary, interview_questions, hard_fit, category_scores_json, confidence, strengths_json, gaps_json, explanation_json, matched_must_haves, missing_must_haves, scoring_version")
   449→              .in("resume_id", allResumeIds);
   450→
   451→            const resumeMap = new Map(allResumes.map((r: any) => [r.id, r]));
   452→            const merged: Result[] = (resultData ?? []).map((r: any) => {
   453→              const resume = resumeMap.get(r.resume_id);
   454→              const parsed = resume?.parsed_json as any;
   455→              return {
   456→                ...r,
   457→                overall_score: r.overall_score ?? 0,
   458→                missing_skills: r.missing_skills ?? [],
   459→                filename: resume?.filename ?? "Unknown",
   460→                file_path: resume?.file_path ?? null,
   461→                candidate_name: resume?.candidate_name ?? null,
   462→                candidate_email: parsed?.email ?? parsed?.contact?.email ?? null,
   463→                candidate_phone: parsed?.phone ?? parsed?.contact?.phone ?? null,
   464→                candidate_location: parsed?.location ?? parsed?.contact?.location ?? null,
   465→                parsed_json: resume?.parsed_json ?? null,
   466→              };
   467→            });
   468→          merged.sort((a, b) => b.overall_score - a.overall_score);
   469→          setResults(merged);
   470→        }
   471→        setRankingProgress(100);
   472→        await refreshProfile();
   473→        setTimeout(() => { resultsRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }); }, 200);
