    25→async function adminCall(body: any) {
    26→  const { data: { session } } = await supabase.auth.getSession();
    27→  const res = await supabase.functions.invoke("admin-users", {
    28→    headers: { Authorization: `Bearer ${session?.access_token}` },
    29→    body,
    30→  });
    31→  if (res.error) throw new Error(res.error.message || "Admin call failed");
    32→  const data = res.data as any;
    33→  if (data?.error) throw new Error(data.error);
    34→  return data;
    35→}

    88→function HealthBanner({ health, loading, onVerify }: {
    89→  health: PostmarkHealth | null;
    90→  loading: boolean;
    91→  onVerify: () => void;
    92→}) {
    93→  if (loading) {
    94→    return (
    95→      <div className="rounded-lg border border-border bg-muted/30 p-4 flex items-center gap-3">
    96→        <Loader2 className="h-4 w-4 animate-spin text-muted-foreground" />
    97→        <span className="text-sm text-muted-foreground">Checking Postmark connection...</span>
    98→      </div>
    99→    );
   100→  }
   101→
   102→  if (!health) return null;
   103→
   104→  const issues: { icon: React.ElementType; text: string; severity: "error" | "warning" }[] = [];
   105→
   106→  if (!health.tokenValid) {
   107→    issues.push({ icon: XCircle, text: "API token is invalid or not configured", severity: "error" });
   108→  }
   109→  if (health.tokenValid && !health.senderVerified) {
   110→    issues.push({
   111→      icon: AlertTriangle,
   112→      text: `Sender "${health.fromEmail}" is not verified in Postmark. Emails will fail.`,
   113→      severity: "warning",
   114→    });
   115→  }
   116→  if (health.tokenValid && !health.accountApproved) {
   117→    issues.push({
   118→      icon: AlertTriangle,
   119→      text: health.accountMessage || "Postmark account is pending approval — can only send to same-domain recipients.",
   120→      severity: "warning",
   121→    });
   122→  }
   123→
   124→  if (issues.length === 0 && health.tokenValid) {
   125→    return (
   126→      <div className="rounded-lg border border-green-200 bg-green-50 dark:border-green-900 dark:bg-green-950/50 p-4 space-y-2">
   127→        <div className="flex items-center justify-between">
   128→          <div className="flex items-center gap-2">
   129→            <CheckCircle2 className="h-4 w-4 text-green-600" />
   130→            <span className="text-sm font-medium text-green-700 dark:text-green-400">
   131→              Postmark Connected{health.serverName ? ` — ${health.serverName}` : ""}
   132→            </span>
   133→          </div>
   134→          <Button variant="ghost" size="sm" className="gap-1.5 h-7 text-xs" onClick={onVerify}>
   135→            <RefreshCw className="h-3 w-3" /> Recheck
   136→          </Button>
   137→        </div>
   138→        <div className="flex gap-3 text-xs text-green-600 dark:text-green-500">
   139→          <span className="flex items-center gap-1"><CheckCircle2 className="h-3 w-3" /> Token valid</span>
   140→          <span className="flex items-center gap-1"><CheckCircle2 className="h-3 w-3" /> Sender verified</span>
   141→          <span className="flex items-center gap-1"><CheckCircle2 className="h-3 w-3" /> Account approved</span>
   142→        </div>
   143→      </div>
   144→    );
   145→  }
   146→
   147→  return (
   148→    <div className="space-y-2">
   149→      {issues.map((issue, i) => (
   150→        <div
   151→          key={i}
   152→          className={`rounded-lg border p-3 flex items-start gap-2 ${
   153→            issue.severity === "error"
   154→              ? "border-destructive/50 bg-destructive/10 text-destructive"
   155→              : "border-amber-200 bg-amber-50 dark:border-amber-900 dark:bg-amber-950 text-amber-700 dark:text-amber-400"
   156→          }`}
   157→        >
   158→          <issue.icon className="h-4 w-4 shrink-0 mt-0.5" />
   159→          <p className="text-xs">{issue.text}</p>
   160→        </div>
   161→      ))}
   162→      <div className="flex justify-end">
   163→        <Button variant="ghost" size="sm" className="gap-1.5 h-7 text-xs" onClick={onVerify}>
   164→          <RefreshCw className="h-3 w-3" /> Recheck
   165→        </Button>
   166→      </div>
   167→    </div>
   168→  );
   169→}

   976→        {template.description && <CardDescription>{template.description}</CardDescription>}
   977→      </CardHeader>
   978→      <CardContent className="space-y-4">
   979→        {vars.length > 0 && (
   980→          <div className="flex flex-wrap gap-1.5">
   981→            <span className="text-xs text-muted-foreground">Variables:</span>
   982→            {vars.map((v) => (
   983→              <Badge key={v} variant="outline" className="text-[11px] font-mono px-1.5 py-0">{`{{${v}}}`}</Badge>
   984→            ))}
   985→          </div>
   986→        )}
   987→
   988→        {showPreview ? (
   989→          <div className="rounded-lg border bg-white dark:bg-background p-4">
   990→            <p className="text-xs text-muted-foreground mb-2">Subject: <strong>{subject}</strong></p>
   991→            <div className="border rounded p-3 text-sm" dangerouslySetInnerHTML={{ __html: bodyHtml }} />
   992→          </div>
   993→        ) : (
   994→          <>
   995→            <div className="space-y-2">
   996→              <Label className="text-sm font-medium">Subject Line</Label>
   997→              <Input value={subject} onChange={(e) => setSubject(e.target.value)} />
   998→            </div>
   999→            <div className="space-y-2">
  1000→              <Label className="text-sm font-medium">Email Body (HTML)</Label>
  1001→              <Textarea value={bodyHtml} onChange={(e) => setBodyHtml(e.target.value)} className="font-mono text-xs min-h-[180px]" />
  1002→            </div>
  1003→          </>
  1004→        )}
  1005→
  1006→        <Button onClick={() => onSave(subject, bodyHtml)} disabled={!dirty || saving} className="gap-2">
  1007→          <Save className="h-4 w-4" />
  1008→          {saving ? "Saving..." : "Save Template"}
  1009→        </Button>
  1010→      </CardContent>
  1011→    </Card>
  1012→  );
  1013→}
