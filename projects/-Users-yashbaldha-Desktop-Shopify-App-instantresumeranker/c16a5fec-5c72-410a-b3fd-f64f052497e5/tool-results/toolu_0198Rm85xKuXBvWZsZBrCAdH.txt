   397→      if (uploadError) throw new Error(`Upload failed: ${uploadError.message}`);
   398→      const { data: resume, error: resumeError } = await supabase
   399→        .from("resumes")
   400→        .insert({ job_id: jobId, filename: name, file_path: filePath })
   401→        .select("id")
   402→        .single();
   403→      if (resumeError || !resume) throw new Error(resumeError?.message || "Failed to save resume");
   404→      resumeIds.push(resume.id);
   405→    }

   409→  const rankAndPoll = async (jobId: string, resumeIds: string[]) => {
   410→    const response = await fetch(
   411→      `https://zkzontawprihrzpayeam.supabase.co/functions/v1/rank-resumes`,
   412→      {
   413→        method: "POST",
   414→        headers: {
   415→          "Content-Type": "application/json",
   416→          Authorization: `Bearer ${(await supabase.auth.getSession()).data.session?.access_token}`,
   417→        },
   418→        body: JSON.stringify({ job_id: jobId, resume_ids: resumeIds }),
   419→      }
   420→    );
   421→
   422→    if (!response.ok) {
   423→      const errBody = await response.json().catch(() => ({}));
   424→      const err: any = new Error(errBody.error || "Ranking failed");
   425→      err.status = response.status;
   426→      throw err;
   427→    }
   428→    setRankingProgress(60);
   429→
   430→    for (let i = 0; i < 150; i++) {
   431→      await new Promise((r) => setTimeout(r, 2000));
   432→      const { data: resumeProgress } = await supabase.from("resumes").select("id, status").in("id", resumeIds);
   433→      if (resumeProgress) {
   434→        const done = resumeProgress.filter((r: any) => r.status === "done" || r.status === "failed").length;
   435→        setScoredCount(done);
   436→        const pct = resumeIds.length > 0 ? Math.round((done / resumeIds.length) * 35) : 0;
   437→        setRankingProgress(Math.min(60 + pct, 95));
   438→      }
   439→
   440→      const { data: jobData } = await supabase.from("jobs").select("status").eq("id", jobId).single();
   441→      if (jobData?.status === "completed") {
   442→         const { data: allResumes } = await supabase.from("resumes").select("id, filename, file_path, candidate_name, parsed_json").eq("job_id", jobId);
   443→         if (allResumes) {
   444→           const allResumeIds = allResumes.map((r: any) => r.id);
   445→            const { data: resultData } = await supabase
   446→              .from("results")
   447→              .select("id, resume_id, overall_score, missing_skills, summary, interview_questions, hard_fit, category_scores_json, confidence, strengths_json, gaps_json, explanation_json, matched_must_haves, missing_must_haves, scoring_version")
   448→              .in("resume_id", allResumeIds);
   449→
   450→            const resumeMap = new Map(allResumes.map((r: any) => [r.id, r]));
   451→            const merged: Result[] = (resultData ?? []).map((r: any) => {
   452→              const resume = resumeMap.get(r.resume_id);
   453→              const parsed = resume?.parsed_json as any;
   454→              return {
   455→                ...r,
   456→                overall_score: r.overall_score ?? 0,
   457→                missing_skills: r.missing_skills ?? [],
   458→                filename: resume?.filename ?? "Unknown",
   459→                file_path: resume?.file_path ?? null,
   460→                candidate_name: resume?.candidate_name ?? null,
   461→                candidate_email: parsed?.email ?? parsed?.contact?.email ?? null,
   462→                candidate_phone: parsed?.phone ?? parsed?.contact?.phone ?? null,
   463→                candidate_location: parsed?.location ?? parsed?.contact?.location ?? null,
   464→                parsed_json: resume?.parsed_json ?? null,
   465→              };
   466→            });
   467→          merged.sort((a, b) => b.overall_score - a.overall_score);
   468→          setResults(merged);
   469→        }
   470→        setRankingProgress(100);
   471→        await refreshProfile();
   472→        setTimeout(() => { resultsRef.current?.scrollIntoView({ behavior: "smooth", block: "start" }); }, 200);
   473→        return;
   474→      }
   475→      if (jobData?.status === "failed") throw new Error("Ranking failed on server");
   476→    }
   477→    throw new Error("Ranking timed out");
   478→  };

  1055→                          onRewriteJd={() => setJdRewriteOpen(true)}

  1058→                    </TableBody>
  1059→                  </Table>
  1060→                </div>
  1061→              </CardContent>
  1062→            </Card>
  1063→
  1064→            {/* Archived Candidates Panel */}
  1065→            {showArchived && archivedResults.length > 0 && (
  1066→              <Card className="mt-3 border-amber-200 dark:border-amber-800">
  1067→                <CardHeader className="pb-3">
  1068→                  <div className="flex items-center justify-between">
  1069→                    <CardTitle className="text-base flex items-center gap-2 text-amber-700 dark:text-amber-400">

  1071→                      Archived Candidates ({archivedResults.length})
