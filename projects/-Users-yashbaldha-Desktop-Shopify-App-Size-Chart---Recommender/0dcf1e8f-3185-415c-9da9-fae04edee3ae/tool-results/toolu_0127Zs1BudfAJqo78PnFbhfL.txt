    28→export const loader = async ({ request, params }: LoaderFunctionArgs) => {
    29→  const { session } = await authenticate.admin(request);
    30→  const shop = session.shop;
    31→  const id = params.id;
    32→
    33→  if (!id) throw new Response("Not found", { status: 404 });
    34→
    35→  if (id === "new") {
    36→    return {
    37→      chart: null as null,
    38→      assignments: [] as Assignment[],
    39→      isNew: true,
    40→    };
    41→  }
    42→
    43→  const chart = await db.sizeChart.findFirst({
    44→    where: { id, shop },
    45→    include: { assignments: true },
    46→  });
    47→
    48→  if (!chart) {
    49→    throw new Response("Not found", { status: 404 });
    50→  }
    51→
    52→  return {
    53→    chart: {
    54→      id: chart.id,
    55→      name: chart.name,
    56→      category: chart.category,
    57→      status: chart.status,
    58→      data: chart.data,
    59→      createdAt: chart.createdAt.toISOString(),
    60→      updatedAt: chart.updatedAt.toISOString(),
    61→    },
    62→    assignments: chart.assignments.map((a) => ({
    63→      id: a.id,
    64→      assignmentType: a.assignmentType,
    65→      assignmentValue: a.assignmentValue,
    66→    })),
    67→    isNew: false,
    68→  };
    69→};

    75→export const action = async ({ request, params }: ActionFunctionArgs) => {
    76→  const { session } = await authenticate.admin(request);
    77→  const shop = session.shop;
    78→  const id = params.id;
    79→  if (!id) return { error: "Missing chart ID" };
    80→  const formData = await request.formData();
    81→  const intent = formData.get("intent") as string;
    82→
    83→  try {
    84→    switch (intent) {
    85→      case "save": {
    86→        const name = formData.get("name") as string;
    87→        const category = formData.get("category") as string;
    88→        const status = formData.get("status") as string;
    89→        const dataRaw = formData.get("data") as string;
    90→
    91→        if (!name?.trim()) return { error: "Chart name is required" };
    92→
    93→        let parsedData: ChartData;
    94→        try {
    95→          parsedData = JSON.parse(dataRaw) as ChartData;
    96→        } catch {
    97→          return { error: "Invalid chart data. Please try again." };
    98→        }
    99→
   100→        if (id === "new") {
   101→          const chart = await db.sizeChart.create({
   102→            data: {
   103→              shop,
   104→              name: name.trim(),
   105→              category,
   106→              status,
   107→              data: JSON.stringify(parsedData),
   108→            },
   109→          });
   110→          return redirect(`/app/size-charts/${chart.id}?saved=1`);
   111→        }
   112→
   113→        const existing = await db.sizeChart.findFirst({ where: { id, shop } });
   114→        if (!existing) return { error: "Chart not found" };
   115→
   116→        await db.sizeChart.update({
   117→          where: { id },
   118→          data: {
   119→            name: name.trim(),
   120→            category,
   121→            status,
   122→            data: JSON.stringify(parsedData),
   123→          },
   124→        });
   125→        return { ok: true, message: "Chart saved successfully" };
   126→      }
   127→
   128→      case "toggle-status": {
   129→        const chart = await db.sizeChart.findFirst({ where: { id, shop } });
   130→        if (!chart) return { error: "Chart not found" };
   131→        const newStatus = chart.status === "active" ? "draft" : "active";
   132→        await db.sizeChart.update({ where: { id }, data: { status: newStatus } });
   133→        return {
   134→          ok: true,
   135→          message: newStatus === "active" ? "Chart published" : "Chart unpublished",
   136→          newStatus,
   137→        };
   138→      }
   139→
   140→      case "add-assignment": {
   141→        const assignmentType = formData.get("assignmentType") as string;
   142→        const assignmentValue = formData.get("assignmentValue") as string;
   143→
   144→        if (!assignmentValue?.trim()) {
   145→          return { error: "Assignment value is required" };
   146→        }
   147→
   148→        await db.chartAssignment.upsert({
   149→          where: {
   150→            shop_sizeChartId_assignmentType_assignmentValue: {
   151→              shop,
   152→              sizeChartId: id,
   153→              assignmentType,
   154→              assignmentValue: assignmentValue.trim(),
   155→            },
   156→          },
   157→          create: {
   158→            shop,
   159→            sizeChartId: id,
   160→            assignmentType,
   161→            assignmentValue: assignmentValue.trim(),
   162→          },
   163→          update: {},
   164→        });
   165→        return { ok: true, message: "Assignment added" };
   166→      }
   167→
   168→      case "remove-assignment": {
   169→        const assignmentId = formData.get("assignmentId") as string;
   170→        await db.chartAssignment.deleteMany({
   171→          where: { id: assignmentId, shop },
   172→        });
   173→        return { ok: true, message: "Assignment removed" };
   174→      }
   175→
   176→      case "delete": {
   177→        await db.sizeChart.deleteMany({ where: { id, shop } });
   178→        return redirect("/app/size-charts");
   179→      }
   180→
   181→      default:
   182→        return { error: "Unknown action" };
   183→    }
   184→  } catch (error) {
   185→    const message =
   186→      error instanceof Error ? error.message : "An unexpected error occurred";
   187→    if (message.includes("Unique constraint")) {
   188→      return { error: "A chart with this name already exists." };
   189→    }
   190→    return { error: message };
   191→  }
   192→};

   227→function makeDefaultData(): ChartData {
   228→  return {
   229→    headers: ["Size", "Chest", "Waist", "Hips"],
   230→    units: { default: "in", available: ["in", "cm"] },
   231→    rows: [

   239→function parseChartData(raw: string): ChartData {
   240→  try {
   241→    const parsed = JSON.parse(raw) as ChartData;
   242→    if (
   243→      Array.isArray(parsed.headers) &&
   244→      parsed.units &&
   245→      Array.isArray(parsed.rows)
   246→    ) {
   247→      return parsed;
   248→    }
   249→  } catch {
   250→    // fall through
   251→  }
   252→  return makeDefaultData();
   253→}

   352→interface TableEditorProps {

   357→function TableEditor({ data, onChange }: TableEditorProps) {
   358→  const headerInputRefs = useRef<(HTMLInputElement | null)[]>([]);
   359→
   360→  const updateCell = (rowIndex: number, header: string, value: string) => {
   361→    const newRows = data.rows.map((row, i) => {
   362→      if (i !== rowIndex) return row;
   363→      return { ...row, [header]: value };
   364→    });
   365→    onChange({ ...data, rows: newRows });
   366→  };
   367→
   368→  const updateHeader = (headerIndex: number, value: string) => {
   369→    const oldHeader = data.headers[headerIndex];
   370→    const newHeaders = data.headers.map((h, i) =>
   371→      i === headerIndex ? value : h,
   372→    );
   373→    const newRows = data.rows.map((row) => {
   374→      const newRow: Record<string, string> = {};
   375→      data.headers.forEach((h, i) => {
   376→        const key = i === headerIndex ? value : h;
   377→        newRow[key] = row[h] ?? "";
   378→      });
   379→      return newRow;
   380→    });
   381→    onChange({ ...data, headers: newHeaders, rows: newRows });
   382→    void oldHeader;
   383→  };
   384→
   385→  const addRow = () => {
   386→    const emptyRow: Record<string, string> = {};
   387→    data.headers.forEach((h) => {
   388→      emptyRow[h] = "";
   389→    });
   390→    onChange({ ...data, rows: [...data.rows, emptyRow] });
   391→  };
   392→
   393→  const removeRow = (rowIndex: number) => {
   394→    onChange({ ...data, rows: data.rows.filter((_, i) => i !== rowIndex) });
   395→  };
   396→
   397→  const addColumn = () => {
   398→    const newHeader = `Column ${data.headers.length + 1}`;
   399→    const newHeaders = [...data.headers, newHeader];
   400→    const newRows = data.rows.map((row) => ({ ...row, [newHeader]: "" }));
   401→    onChange({ ...data, headers: newHeaders, rows: newRows });
   402→  };
   403→
   404→  const removeColumn = (headerIndex: number) => {
   405→    if (data.headers.length <= 1) return;
   406→    const removedHeader = data.headers[headerIndex];
   407→    const newHeaders = data.headers.filter((_, i) => i !== headerIndex);
   408→    const newRows = data.rows.map((row) => {
   409→      const newRow: Record<string, string> = {};
   410→      newHeaders.forEach((h) => {
   411→        newRow[h] = row[h] ?? "";
   412→      });
   413→      return newRow;
   414→    });
   415→    onChange({ ...data, headers: newHeaders, rows: newRows });
   416→    void removedHeader;
   417→  };
   418→
   419→  const toggleUnit = (unit: string) => {
   420→    onChange({ ...data, units: { ...data.units, default: unit } });
   421→  };
   422→
   423→  return (
   424→    <div>
   425→      {/* Toolbar — unit toggle + row/col count */}
   426→      <div
   427→        style={{
   428→          display: "flex",
   429→          alignItems: "center",
   430→          gap: "8px",
   431→          marginBottom: "10px",
   432→        }}
   433→      >
   434→        <span
   435→          style={{
   436→            fontSize: "12px",
   437→            fontWeight: 600,
   438→            color: "#6d7175",
   439→            textTransform: "uppercase",
   440→            letterSpacing: "0.04em",
   441→          }}
   442→        >
   443→          Units
   444→        </span>
   445→        <div
   446→          style={{
   447→            display: "flex",
   448→            gap: "4px",
   449→            background: "#f6f6f7",
   450→            borderRadius: "20px",
   451→            padding: "3px",
   452→            border: "1px solid #e1e3e5",
   453→          }}
   454→        >
   455→          {data.units.available.map((unit) => (
   456→            <button
   457→              key={unit}
   458→              type="button"
   459→              onClick={() => toggleUnit(unit)}
   460→              style={{
   461→                padding: "3px 14px",
   462→                borderRadius: "16px",
   463→                border: "none",
   464→                background:
   465→                  data.units.default === unit ? "#2c6ecb" : "transparent",
   466→                color: data.units.default === unit ? "#ffffff" : "#6d7175",
   467→                fontWeight: data.units.default === unit ? 700 : 500,
   468→                cursor: "pointer",
   469→                fontSize: "12px",
   470→                lineHeight: "20px",
   471→                transition: "all 0.12s ease",
   472→                fontFamily: "inherit",
   473→              }}
   474→            >
   475→              {unit}
   476→            </button>
   477→          ))}
   478→        </div>
   479→        <div
   480→          style={{
   481→            marginLeft: "auto",
   482→            fontSize: "12px",
   483→            color: "#8c9196",
   484→            fontWeight: 500,
   485→            background: "#f6f6f7",
