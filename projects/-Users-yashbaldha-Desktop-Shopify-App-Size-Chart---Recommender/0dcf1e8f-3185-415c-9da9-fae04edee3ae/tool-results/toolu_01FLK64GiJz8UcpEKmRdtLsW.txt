    28→export const loader = async ({ request, params }: LoaderFunctionArgs) => {
    29→  const { session } = await authenticate.admin(request);
    30→  const shop = session.shop;
    31→  const id = params.id;
    32→
    33→  if (!id) throw new Response("Not found", { status: 404 });
    34→
    35→  if (id === "new") {
    36→    return {
    37→      chart: null as null,
    38→      assignments: [] as Assignment[],
    39→      isNew: true,
    40→    };
    41→  }
    42→
    43→  const chart = await db.sizeChart.findFirst({
    44→    where: { id, shop },
    45→    include: { assignments: true },
    46→  });
    47→
    48→  if (!chart) {
    49→    throw new Response("Not found", { status: 404 });
    50→  }
    51→
    52→  return {
    53→    chart: {
    54→      id: chart.id,
    55→      name: chart.name,
    56→      category: chart.category,
    57→      status: chart.status,
    58→      data: chart.data,
    59→      createdAt: chart.createdAt.toISOString(),
    60→      updatedAt: chart.updatedAt.toISOString(),
    61→    },
    62→    assignments: chart.assignments.map((a) => ({
    63→      id: a.id,
    64→      assignmentType: a.assignmentType,
    65→      assignmentValue: a.assignmentValue,
    66→    })),
    67→    isNew: false,
    68→  };
    69→};

    75→export const action = async ({ request, params }: ActionFunctionArgs) => {
    76→  const { session } = await authenticate.admin(request);
    77→  const shop = session.shop;
    78→  const id = params.id;
    79→  if (!id) return { error: "Missing chart ID" };
    80→  const formData = await request.formData();
    81→  const intent = formData.get("intent") as string;
    82→
    83→  try {
    84→    switch (intent) {
    85→      case "save": {
    86→        const name = formData.get("name") as string;
    87→        const category = formData.get("category") as string;
    88→        const status = formData.get("status") as string;
    89→        const dataRaw = formData.get("data") as string;
    90→
    91→        if (!name?.trim()) return { error: "Chart name is required" };
    92→
    93→        let parsedData: ChartData;
    94→        try {
    95→          parsedData = JSON.parse(dataRaw) as ChartData;
    96→        } catch {
    97→          return { error: "Invalid chart data. Please try again." };
    98→        }
    99→
   100→        if (id === "new") {
   101→          const chart = await db.sizeChart.create({
   102→            data: {
   103→              shop,
   104→              name: name.trim(),
   105→              category,
   106→              status,
   107→              data: JSON.stringify(parsedData),
   108→            },
   109→          });
   110→          return redirect(`/app/size-charts/${chart.id}?saved=1`);
   111→        }
   112→
   113→        const existing = await db.sizeChart.findFirst({ where: { id, shop } });
   114→        if (!existing) return { error: "Chart not found" };
   115→
   116→        await db.sizeChart.update({
   117→          where: { id },
   118→          data: {
   119→            name: name.trim(),
   120→            category,
   121→            status,
   122→            data: JSON.stringify(parsedData),
   123→          },
   124→        });
   125→        return { ok: true, message: "Chart saved successfully" };
   126→      }
   127→
   128→      case "toggle-status": {
   129→        const chart = await db.sizeChart.findFirst({ where: { id, shop } });
   130→        if (!chart) return { error: "Chart not found" };
   131→        const newStatus = chart.status === "active" ? "draft" : "active";
   132→        await db.sizeChart.update({ where: { id }, data: { status: newStatus } });
   133→        return {
   134→          ok: true,
   135→          message: newStatus === "active" ? "Chart published" : "Chart unpublished",
   136→          newStatus,
   137→        };
   138→      }
   139→
   140→      case "add-assignment": {
   141→        const assignmentType = formData.get("assignmentType") as string;
   142→        const assignmentValue = formData.get("assignmentValue") as string;
   143→
   144→        if (!assignmentValue?.trim()) {
   145→          return { error: "Assignment value is required" };
   146→        }
   147→
   148→        await db.chartAssignment.upsert({
   149→          where: {
   150→            shop_sizeChartId_assignmentType_assignmentValue: {
   151→              shop,
   152→              sizeChartId: id,
   153→              assignmentType,
   154→              assignmentValue: assignmentValue.trim(),
   155→            },
   156→          },
   157→          create: {
   158→            shop,
   159→            sizeChartId: id,
   160→            assignmentType,
   161→            assignmentValue: assignmentValue.trim(),
   162→          },
   163→          update: {},
   164→        });
   165→        return { ok: true, message: "Assignment added" };
   166→      }
   167→
   168→      case "remove-assignment": {
   169→        const assignmentId = formData.get("assignmentId") as string;
   170→        await db.chartAssignment.deleteMany({
   171→          where: { id: assignmentId, shop },
   172→        });
   173→        return { ok: true, message: "Assignment removed" };
   174→      }
   175→
   176→      case "delete": {
   177→        await db.sizeChart.deleteMany({ where: { id, shop } });
   178→        return redirect("/app/size-charts");
   179→      }
   180→
   181→      default:
   182→        return { error: "Unknown action" };
   183→    }
   184→  } catch (error) {
   185→    const message =
   186→      error instanceof Error ? error.message : "An unexpected error occurred";
   187→    if (message.includes("Unique constraint")) {
   188→      return { error: "A chart with this name already exists." };
   189→    }
   190→    return { error: message };
   191→  }
   192→};

   239→function parseChartData(raw: string): ChartData {
   240→  try {
   241→    const parsed = JSON.parse(raw) as ChartData;
   242→    if (
   243→      Array.isArray(parsed.headers) &&
   244→      parsed.units &&
   245→      Array.isArray(parsed.rows)
   246→    ) {
   247→      return parsed;
   248→    }
   249→  } catch {
   250→    // fall through
   251→  }
   252→  return makeDefaultData();
   253→}

   340→function TableEditor({ data, onChange }: TableEditorProps) {
   341→  const headerInputRefs = useRef<(HTMLInputElement | null)[]>([]);
   342→
   343→  const updateCell = (rowIndex: number, header: string, value: string) => {
   344→    const newRows = data.rows.map((row, i) => {
   345→      if (i !== rowIndex) return row;
   346→      return { ...row, [header]: value };
   347→    });
   348→    onChange({ ...data, rows: newRows });
   349→  };
   350→
   351→  const updateHeader = (headerIndex: number, value: string) => {
   352→    const oldHeader = data.headers[headerIndex];
   353→    const newHeaders = data.headers.map((h, i) => (i === headerIndex ? value : h));
   354→    const newRows = data.rows.map((row) => {
   355→      const newRow: Record<string, string> = {};
   356→      data.headers.forEach((h, i) => {
   357→        const key = i === headerIndex ? value : h;
   358→        newRow[key] = row[h] ?? "";
   359→      });
   360→      return newRow;
   361→    });
   362→    onChange({ ...data, headers: newHeaders, rows: newRows });
   363→    void oldHeader;
   364→  };
   365→
   366→  const addRow = () => {
   367→    const emptyRow: Record<string, string> = {};
   368→    data.headers.forEach((h) => {
   369→      emptyRow[h] = "";
   370→    });
   371→    onChange({ ...data, rows: [...data.rows, emptyRow] });
   372→  };
   373→
   374→  const removeRow = (rowIndex: number) => {
   375→    onChange({ ...data, rows: data.rows.filter((_, i) => i !== rowIndex) });
   376→  };
   377→
   378→  const addColumn = () => {
   379→    const newHeader = `Column ${data.headers.length + 1}`;
   380→    const newHeaders = [...data.headers, newHeader];
   381→    const newRows = data.rows.map((row) => ({ ...row, [newHeader]: "" }));
   382→    onChange({ ...data, headers: newHeaders, rows: newRows });
   383→  };
   384→
   385→  const removeColumn = (headerIndex: number) => {
   386→    if (data.headers.length <= 1) return;
   387→    const removedHeader = data.headers[headerIndex];
   388→    const newHeaders = data.headers.filter((_, i) => i !== headerIndex);
   389→    const newRows = data.rows.map((row) => {
   390→      const newRow: Record<string, string> = {};
   391→      newHeaders.forEach((h) => {
   392→        newRow[h] = row[h] ?? "";
   393→      });
   394→      return newRow;
   395→    });
   396→    onChange({ ...data, headers: newHeaders, rows: newRows });
   397→    void removedHeader;
   398→  };
   399→
   400→  const toggleUnit = (unit: string) => {
   401→    onChange({ ...data, units: { ...data.units, default: unit } });
   402→  };
   403→
   404→  return (
   405→    <div>
   406→      {/* Toolbar */}
   407→      <div
   408→        style={{
   409→          display: "flex",
   410→          alignItems: "center",
   411→          gap: "8px",
   412→          background: "#f6f6f7",
   413→          border: "1px solid #e1e3e5",
   414→          borderBottom: "none",
   415→          borderRadius: "8px 8px 0 0",
   416→          padding: "10px 14px",
   417→        }}
   418→      >
   419→        <span
   420→          style={{
   421→            fontSize: "12px",
   422→            fontWeight: 600,
   423→            color: "#6d7175",
   424→            marginRight: "4px",
   425→            textTransform: "uppercase",
   426→            letterSpacing: "0.04em",
   427→          }}
   428→        >
   429→          Units
   430→        </span>
   431→        <div style={{ display: "flex", gap: "4px" }}>
   432→          {data.units.available.map((unit) => (
   433→            <button
   434→              key={unit}
   435→              type="button"
   436→              onClick={() => toggleUnit(unit)}
   437→              style={{
   438→                padding: "3px 12px",
   439→                borderRadius: "20px",
   440→                border:
   441→                  data.units.default === unit
   442→                    ? "1.5px solid #2c6ecb"
   443→                    : "1.5px solid #c9cccf",
   444→                background:
   445→                  data.units.default === unit ? "#2c6ecb" : "#ffffff",
   446→                color:
   447→                  data.units.default === unit ? "#ffffff" : "#6d7175",
   448→                fontWeight: data.units.default === unit ? 700 : 500,
   449→                cursor: "pointer",
   450→                fontSize: "12px",
   451→                lineHeight: "20px",
   452→                transition: "all 0.12s ease",
   453→                fontFamily: "inherit",
   454→              }}
   455→            >
   456→              {unit}
   457→            </button>
   458→          ))}
   459→        </div>
   460→        <div
   461→          style={{
   462→            marginLeft: "auto",
   463→            fontSize: "12px",
   464→            color: "#8c9196",
   465→            fontWeight: 500,
   466→          }}
   467→        >
   468→          {data.rows.length} rows &middot; {data.headers.length} columns
