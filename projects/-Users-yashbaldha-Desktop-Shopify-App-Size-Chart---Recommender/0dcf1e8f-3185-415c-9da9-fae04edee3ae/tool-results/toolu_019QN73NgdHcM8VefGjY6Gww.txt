    10→export const loader = async ({ request }: LoaderFunctionArgs) => {
    11→  const { session } = await authenticate.admin(request);
    12→  const shop = session.shop;
    13→
    14→  const [charts, subscription] = await Promise.all([
    15→    db.sizeChart.findMany({
    16→      where: { shop },
    17→      orderBy: { updatedAt: "desc" },
    18→      include: { assignments: true },
    19→    }),
    20→    db.subscription.findUnique({ where: { shop } }),
    21→  ]);
    22→
    23→  const plan = getPlan(subscription?.planId ?? "free");
    24→  const templates = SIZE_CHART_TEMPLATES.map((t) => ({
    25→    name: t.name,
    26→    category: t.category,
    27→    gender: t.gender,
    28→    rowCount: t.data.rows.length,
    29→  }));
    30→  const categories = getCategories();
    31→
    32→  return {
    33→    charts,
    34→    plan,
    35→    templates,
    36→    categories,
    37→    canCreate: canCreateChart(plan.id, charts.length),
    38→  };
    39→};

    41→export const action = async ({ request }: ActionFunctionArgs) => {
    42→  const { session } = await authenticate.admin(request);
    43→  const shop = session.shop;
    44→  const formData = await request.formData();
    45→  const intent = formData.get("intent") as string;
    46→
    47→  try {
    48→    switch (intent) {
    49→      case "create-from-template": {
    50→        const templateName = formData.get("templateName") as string;
    51→        const template = SIZE_CHART_TEMPLATES.find((t) => t.name === templateName);
    52→        if (!template) return { error: "Template not found" };
    53→
    54→        const subscription = await db.subscription.findUnique({ where: { shop } });
    55→        const plan = getPlan(subscription?.planId ?? "free");
    56→        const chartCount = await db.sizeChart.count({ where: { shop } });
    57→        if (!canCreateChart(plan.id, chartCount)) {
    58→          return {
    59→            error: `Free plan allows max ${plan.limits.maxCharts} charts. Upgrade to create more.`,
    60→          };
    61→        }
    62→
    63→        let name = template.name;
    64→        const existing = await db.sizeChart.findUnique({
    65→          where: { shop_name: { shop, name } },
    66→        });
    67→        if (existing) name = `${template.name} (${Date.now().toString(36)})`;
    68→
    69→        const created = await db.sizeChart.create({
    70→          data: {
    71→            shop,
    72→            name,
    73→            category: template.category,
    74→            data: JSON.stringify(template.data),
    75→            status: "active",
    76→          },
    77→        });
    78→        return { ok: true, message: `"${name}" created successfully`, redirectId: created.id };
    79→      }
    80→
    81→      case "create-blank": {
    82→        const name = (formData.get("name") as string) || "Untitled Size Chart";
    83→        const category = (formData.get("category") as string) || "general";
    84→
    85→        const subscription = await db.subscription.findUnique({ where: { shop } });
    86→        const plan = getPlan(subscription?.planId ?? "free");
    87→        const chartCount = await db.sizeChart.count({ where: { shop } });
    88→        if (!canCreateChart(plan.id, chartCount)) {
    89→          return {
    90→            error: `Free plan allows max ${plan.limits.maxCharts} charts. Upgrade to create more.`,
    91→          };
    92→        }
    93→
    94→        let uniqueName = name;
    95→        const existing = await db.sizeChart.findUnique({
    96→          where: { shop_name: { shop, name } },
    97→        });
    98→        if (existing) uniqueName = `${name} (${Date.now().toString(36)})`;
    99→
   100→        const defaultData = {
   101→          headers: ["Size", "Chest", "Waist", "Hips"],
   102→          units: { default: "in", available: ["in", "cm"] },
   103→          rows: [
   104→            { Size: "S", Chest: "", Waist: "", Hips: "" },
   105→            { Size: "M", Chest: "", Waist: "", Hips: "" },
   106→            { Size: "L", Chest: "", Waist: "", Hips: "" },
   107→          ],
   108→        };
   109→
   110→        const created = await db.sizeChart.create({
   111→          data: {
   112→            shop,
   113→            name: uniqueName,
   114→            category,
   115→            data: JSON.stringify(defaultData),
   116→            status: "draft",
   117→          },
   118→        });
   119→        return { ok: true, message: `"${uniqueName}" created`, redirectId: created.id };
   120→      }
   121→
   122→      case "delete": {
   123→        const id = formData.get("id") as string;
   124→        await db.sizeChart.deleteMany({ where: { id, shop } });
   125→        return { ok: true, message: "Chart deleted" };
   126→      }
   127→
   128→      case "import-csv": {
   129→        const csvRaw = formData.get("csvData") as string;
   130→        const chartName = (formData.get("name") as string) || "Imported Size Chart";
   131→        const category = (formData.get("category") as string) || "general";
   132→
   133→        if (!csvRaw?.trim()) return { error: "CSV data is required" };
   134→
   135→        const subscription = await db.subscription.findUnique({ where: { shop } });
   136→        const plan = getPlan(subscription?.planId ?? "free");
   137→        const chartCount = await db.sizeChart.count({ where: { shop } });
   138→        if (!canCreateChart(plan.id, chartCount)) {
   139→          return { error: `Free plan allows max ${plan.limits.maxCharts} charts. Upgrade to create more.` };
   140→        }
   141→
   142→        // Parse CSV
   143→        const lines = csvRaw.trim().split(/\r?\n/).filter((l) => l.trim());
   144→        if (lines.length < 2) return { error: "CSV must have a header row and at least one data row." };
   145→
   146→        const parseRow = (line: string): string[] => {
   147→          const result: string[] = [];
   148→          let current = "";
   149→          let inQuotes = false;
   150→          for (let i = 0; i < line.length; i++) {
   151→            const ch = line[i];
   152→            if (ch === '"') {
   153→              if (inQuotes && line[i + 1] === '"') { current += '"'; i++; }
   154→              else { inQuotes = !inQuotes; }
   155→            } else if (ch === "," && !inQuotes) {
   156→              result.push(current.trim());
   157→              current = "";
   158→            } else {
   159→              current += ch;
   160→            }
   161→          }
   162→          result.push(current.trim());
   163→          return result;
   164→        };
   165→
   166→        const headers = parseRow(lines[0]);
   167→        if (headers.length < 1) return { error: "Could not parse CSV headers." };
   168→
   169→        const rows = lines.slice(1).map((line) => {
   170→          const cells = parseRow(line);
   171→          const row: Record<string, string> = {};
   172→          headers.forEach((h, i) => { row[h] = cells[i] ?? ""; });
   173→          return row;
   174→        });
   175→
   176→        const data = {
   177→          headers,
   178→          units: { default: "in", available: ["in", "cm"] },
   179→          rows,
   180→        };
   181→
   182→        let uniqueName = chartName;
   183→        const existingName = await db.sizeChart.findUnique({ where: { shop_name: { shop, name: chartName } } });
   184→        if (existingName) uniqueName = `${chartName} (${Date.now().toString(36)})`;
   185→
   186→        const created = await db.sizeChart.create({
   187→          data: { shop, name: uniqueName, category, data: JSON.stringify(data), status: "draft" },
   188→        });
   189→        return { ok: true, message: `Imported "${uniqueName}"`, redirectId: created.id };
   190→      }
   191→
   192→      case "duplicate": {
   193→        const sourceId = formData.get("id") as string;
   194→        const source = await db.sizeChart.findFirst({ where: { id: sourceId, shop } });
   195→        if (!source) return { error: "Chart not found" };
   196→
   197→        const subscription = await db.subscription.findUnique({ where: { shop } });
   198→        const plan = getPlan(subscription?.planId ?? "free");
   199→        const chartCount = await db.sizeChart.count({ where: { shop } });
   200→        if (!canCreateChart(plan.id, chartCount)) {
   201→          return { error: "Chart limit reached. Upgrade your plan." };
   202→        }
   203→
   204→        const newName = `${source.name} (Copy)`;
   205→        await db.sizeChart.create({
   206→          data: {
   207→            shop,
   208→            name: newName,
   209→            type: source.type,
   210→            category: source.category,
   211→            data: source.data,
   212→            status: "draft",
   213→          },
   214→        });
   215→        return { ok: true, message: `Duplicated as "${newName}"` };
   216→      }
   217→
   218→      case "toggle-status": {
   219→        const chartId = formData.get("id") as string;
   220→        const chart = await db.sizeChart.findFirst({ where: { id: chartId, shop } });
   221→        if (!chart) return { error: "Chart not found" };
   222→
   223→        const newStatus = chart.status === "active" ? "draft" : "active";
   224→        await db.sizeChart.update({ where: { id: chartId }, data: { status: newStatus } });
   225→        return {
   226→          ok: true,
   227→          message: `Chart ${newStatus === "active" ? "published" : "unpublished"}`,
   228→        };
   229→      }
   230→
   231→      default:
   232→        return { error: "Unknown action" };
   233→    }
   234→  } catch (error) {
   235→    const message = error instanceof Error ? error.message : "An unexpected error occurred";
   236→    if (message.includes("Unique constraint")) {
   237→      return { error: "A chart with this name already exists." };
   238→    }
   239→    return { error: message };
   240→  }
   241→};

   548→export default function SizeCharts() {
   549→  const { charts, plan, templates, categories, canCreate } =
   550→    useLoaderData<typeof loader>();
   551→  const fetcher = useFetcher<typeof action>();
   552→  const shopify = useAppBridge();
   553→
   554→  const [showTemplates, setShowTemplates] = useState(false);
   555→  const [showImport, setShowImport] = useState(false);
   556→  const [importName, setImportName] = useState("Imported Size Chart");
   557→  const [importCategory, setImportCategory] = useState("general");
   558→  const [importCsvText, setImportCsvText] = useState("");
   559→  const [statusFilter, setStatusFilter] = useState<"all" | "active" | "draft">("all");
   560→  const [categoryFilter, setCategoryFilter] = useState("all");
   561→  const [searchQuery, setSearchQuery] = useState("");
   562→
   563→  const isLoading = fetcher.state !== "idle";
   564→
   565→  useEffect(() => {
   566→    if (!fetcher.data) return;
   567→    const data = fetcher.data as ActionData;
   568→    if ("ok" in data && data.ok) {
   569→      shopify.toast.show(data.message);
   570→      if ("redirectId" in data && data.redirectId) {
   571→        window.location.href = `/app/size-charts/${data.redirectId}`;
   572→      } else {
   573→        setShowTemplates(false);
   574→        setShowImport(false);
   575→        setImportCsvText("");
   576→        setImportName("Imported Size Chart");
   577→      }
   578→    } else if ("error" in data) {
   579→      shopify.toast.show(data.error, { isError: true });
   580→    }
   581→  }, [fetcher.data, shopify]);
   582→
   583→  const filteredCharts = charts.filter((chart) => {
   584→    const matchesStatus = statusFilter === "all" || chart.status === statusFilter;
   585→    const matchesCategory = categoryFilter === "all" || chart.category === categoryFilter;
   586→    const matchesSearch =
   587→      searchQuery.trim() === "" ||
   588→      chart.name.toLowerCase().includes(searchQuery.toLowerCase());
   589→    return matchesStatus && matchesCategory && matchesSearch;
   590→  });
   591→
   592→  const publishedCount = charts.filter((c) => c.status === "active").length;
   593→  const draftCount = charts.filter((c) => c.status === "draft").length;
   594→  const activeFilteredCount = charts.filter((c) => c.status === "active").length;
   595→  const draftFilteredCount = charts.filter((c) => c.status === "draft").length;
   596→  const maxCharts = plan.limits.maxCharts;
   597→  const isFreePlan = plan.id === "free";
   598→
   599→  const progressPct =
