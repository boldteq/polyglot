    28→    28→export const loader = async ({ request, params }: LoaderFunctionArgs) => {
    29→    29→  const { session } = await authenticate.admin(request);
    30→    30→  const shop = session.shop;

    32→    32→
    33→    33→  if (!id) throw new Response("Not found", { status: 404 });
    34→    34→
    35→    35→  if (id === "new") {
    36→    36→    return {
    37→    37→      chart: null as null,
    38→    38→      assignments: [] as Assignment[],
    39→    39→      isNew: true,
    40→    40→    };
    41→    41→  }
    42→    42→
    43→    43→  const chart = await db.sizeChart.findFirst({
    44→    44→    where: { id, shop },
    45→    45→    include: { assignments: true },
    46→    46→  });
    47→    47→
    48→    48→  if (!chart) {
    49→    49→    throw new Response("Not found", { status: 404 });
    50→    50→  }
    51→    51→

    54→    54→      id: chart.id,
    55→    55→      name: chart.name,
    56→    56→      category: chart.category,
    57→    57→      status: chart.status,

    75→    75→export const action = async ({ request, params }: ActionFunctionArgs) => {
    76→    76→  const { session } = await authenticate.admin(request);
    77→    77→  const shop = session.shop;
    78→    78→  const id = params.id;
    79→    79→  if (!id) return { error: "Missing chart ID" };
    80→    80→  const formData = await request.formData();
    81→    81→  const intent = formData.get("intent") as string;
    82→    82→
    83→    83→  try {
    84→    84→    switch (intent) {
    85→    85→      case "save": {
    86→    86→        const name = formData.get("name") as string;
    87→    87→        const category = formData.get("category") as string;
    88→    88→        const status = formData.get("status") as string;
    89→    89→        const dataRaw = formData.get("data") as string;
    90→    90→
    91→    91→        if (!name?.trim()) return { error: "Chart name is required" };
    92→    92→
    93→    93→        let parsedData: ChartData;
    94→    94→        try {
    95→    95→          parsedData = JSON.parse(dataRaw) as ChartData;
    96→    96→        } catch {
    97→    97→          return { error: "Invalid chart data. Please try again." };
    98→    98→        }
    99→    99→
   100→   100→        if (id === "new") {
   101→   101→          const chart = await db.sizeChart.create({
   102→   102→            data: {
   103→   103→              shop,
   104→   104→              name: name.trim(),
   105→   105→              category,
   106→   106→              status,
   107→   107→              data: JSON.stringify(parsedData),
   108→   108→            },
   109→   109→          });
   110→   110→          return redirect(`/app/size-charts/${chart.id}?saved=1`);
   111→   111→        }
   112→   112→
   113→   113→        const existing = await db.sizeChart.findFirst({ where: { id, shop } });
   114→   114→        if (!existing) return { error: "Chart not found" };
   115→   115→
   116→   116→        await db.sizeChart.update({
   117→   117→          where: { id },
   118→   118→          data: {
   119→   119→            name: name.trim(),
   120→   120→            category,
   121→   121→            status,
   122→   122→            data: JSON.stringify(parsedData),
   123→   123→          },
   124→   124→        });
   125→   125→        return { ok: true, message: "Chart saved successfully" };
   126→   126→      }
   127→   127→
   128→   128→      case "toggle-status": {
   129→   129→        const chart = await db.sizeChart.findFirst({ where: { id, shop } });
   130→   130→        if (!chart) return { error: "Chart not found" };
   131→   131→        const newStatus = chart.status === "active" ? "draft" : "active";
   132→   132→        await db.sizeChart.update({ where: { id }, data: { status: newStatus } });
   133→   133→        return {
   134→   134→          ok: true,
   135→   135→          message: newStatus === "active" ? "Chart published" : "Chart unpublished",
   136→   136→          newStatus,
   137→   137→        };
   138→   138→      }
   139→   139→
   140→   140→      case "add-assignment": {
   141→   141→        const assignmentType = formData.get("assignmentType") as string;
   142→   142→        const assignmentValue = formData.get("assignmentValue") as string;
   143→   143→
   144→   144→        if (!assignmentValue?.trim()) {
   145→   145→          return { error: "Assignment value is required" };
   146→   146→        }
   147→   147→
   148→   148→        await db.chartAssignment.upsert({
   149→   149→          where: {
   150→   150→            shop_sizeChartId_assignmentType_assignmentValue: {
   151→   151→              shop,
   152→   152→              sizeChartId: id,
   153→   153→              assignmentType,
   154→   154→              assignmentValue: assignmentValue.trim(),
   155→   155→            },
   156→   156→          },
   157→   157→          create: {
   158→   158→            shop,
   159→   159→            sizeChartId: id,
   160→   160→            assignmentType,
   161→   161→            assignmentValue: assignmentValue.trim(),
   162→   162→          },
   163→   163→          update: {},
   164→   164→        });
   165→   165→        return { ok: true, message: "Assignment added" };
   166→   166→      }
   167→   167→
   168→   168→      case "remove-assignment": {
   169→   169→        const assignmentId = formData.get("assignmentId") as string;
   170→   170→        await db.chartAssignment.deleteMany({
   171→   171→          where: { id: assignmentId, shop },
   172→   172→        });
   173→   173→        return { ok: true, message: "Assignment removed" };
   174→   174→      }
   175→   175→
   176→   176→      case "delete": {
   177→   177→        await db.sizeChart.deleteMany({ where: { id, shop } });
   178→   178→        return redirect("/app/size-charts");
   179→   179→      }
   180→   180→
   181→   181→      default:
   182→   182→        return { error: "Unknown action" };
   183→   183→    }
   184→   184→  } catch (error) {
   185→   185→    const message =
   186→   186→      error instanceof Error ? error.message : "An unexpected error occurred";
   187→   187→    if (message.includes("Unique constraint")) {
   188→   188→      return { error: "A chart with this name already exists." };
   189→   189→    }
   190→   190→    return { error: message };
   191→   191→  }
   192→   192→};

   240→   240→  try {
   241→   241→    const parsed = JSON.parse(raw) as ChartData;
   242→   242→    if (
   243→   243→      Array.isArray(parsed.headers) &&
   244→   244→      parsed.units &&
   245→   245→      Array.isArray(parsed.rows)
   246→   246→    ) {
   247→   247→      return parsed;
   248→   248→    }
   249→   249→  } catch {
   250→   250→    // fall through
   251→   251→  }
   252→   252→  return makeDefaultData();

   359→   359→
   360→   360→  const updateCell = (rowIndex: number, header: string, value: string) => {

   362→   362→      if (i !== rowIndex) return row;
   363→   363→      return { ...row, [header]: value };
   364→   364→    });
   365→   365→    onChange({ ...data, rows: newRows });
   366→   366→  };
   367→   367→
   368→   368→  const updateHeader = (headerIndex: number, value: string) => {
   369→   369→    const oldHeader = data.headers[headerIndex];
   370→   370→    const newHeaders = data.headers.map((h, i) =>
   371→   371→      i === headerIndex ? value : h,
   372→   372→    );

   374→   374→      const newRow: Record<string, string> = {};
   375→   375→      data.headers.forEach((h, i) => {
   376→   376→        const key = i === headerIndex ? value : h;
   377→   377→        newRow[key] = row[h] ?? "";
   378→   378→      });
   379→   379→      return newRow;
   380→   380→    });
   381→   381→    onChange({ ...data, headers: newHeaders, rows: newRows });
