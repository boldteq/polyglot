    37→export const loader = async ({ request, params }: LoaderFunctionArgs) => {
    38→  const { session } = await authenticate.admin(request);
    39→  const shop = session.shop;
    40→  const id = params.id;
    41→
    42→  if (!id) throw new Response("Not found", { status: 404 });
    43→
    44→  if (id === "new") {
    45→    return {
    46→      chart: null as null,
    47→      assignments: [] as Assignment[],
    48→      isNew: true,
    49→    };
    50→  }
    51→
    52→  const chart = await db.sizeChart.findFirst({
    53→    where: { id, shop },
    54→    include: { assignments: true },
    55→  });
    56→
    57→  if (!chart) {
    58→    throw new Response("Not found", { status: 404 });
    59→  }
    60→
    61→  return {
    62→    chart: {
    63→      id: chart.id,
    64→      name: chart.name,
    65→      category: chart.category,
    66→      status: chart.status,
    67→      data: chart.data,
    68→      createdAt: chart.createdAt.toISOString(),
    69→      updatedAt: chart.updatedAt.toISOString(),
    70→    },
    71→    assignments: chart.assignments.map((a) => ({
    72→      id: a.id,
    73→      assignmentType: a.assignmentType,
    74→      assignmentValue: a.assignmentValue,
    75→    })),
    76→    isNew: false,
    77→  };
    78→};

    84→export const action = async ({ request, params }: ActionFunctionArgs) => {
    85→  const { session } = await authenticate.admin(request);
    86→  const shop = session.shop;
    87→  const id = params.id;
    88→  if (!id) return { error: "Missing chart ID" };
    89→  const formData = await request.formData();
    90→  const intent = formData.get("intent") as string;
    91→
    92→  try {
    93→    switch (intent) {
    94→      case "save": {
    95→        const name = formData.get("name") as string;
    96→        const category = formData.get("category") as string;
    97→        const status = formData.get("status") as string;
    98→        const dataRaw = formData.get("data") as string;
    99→
   100→        if (!name?.trim()) return { error: "Chart name is required" };
   101→
   102→        let parsedData: ChartData;
   103→        try {
   104→          parsedData = JSON.parse(dataRaw) as ChartData;
   105→        } catch {
   106→          return { error: "Invalid chart data. Please try again." };
   107→        }
   108→
   109→        if (id === "new") {
   110→          const chart = await db.sizeChart.create({
   111→            data: {
   112→              shop,
   113→              name: name.trim(),
   114→              category,
   115→              status,
   116→              data: JSON.stringify(parsedData),
   117→            },
   118→          });
   119→          return redirect(`/app/size-charts/${chart.id}?saved=1`);
   120→        }
   121→
   122→        const existing = await db.sizeChart.findFirst({ where: { id, shop } });
   123→        if (!existing) return { error: "Chart not found" };
   124→
   125→        await db.sizeChart.update({
   126→          where: { id },
   127→          data: {
   128→            name: name.trim(),
   129→            category,
   130→            status,
   131→            data: JSON.stringify(parsedData),
   132→          },
   133→        });
   134→        return { ok: true, message: "Chart saved successfully" };
   135→      }
   136→
   137→      case "toggle-status": {
   138→        const chart = await db.sizeChart.findFirst({ where: { id, shop } });
   139→        if (!chart) return { error: "Chart not found" };
   140→        const newStatus = chart.status === "active" ? "draft" : "active";
   141→        await db.sizeChart.update({ where: { id }, data: { status: newStatus } });
   142→        return {
   143→          ok: true,
   144→          message: newStatus === "active" ? "Chart published" : "Chart unpublished",
   145→          newStatus,
   146→        };
   147→      }
   148→
   149→      case "add-assignment": {
   150→        const assignmentType = formData.get("assignmentType") as string;
   151→        const assignmentValue = formData.get("assignmentValue") as string;
   152→
   153→        if (!assignmentValue?.trim()) {
   154→          return { error: "Assignment value is required" };
   155→        }
   156→
   157→        await db.chartAssignment.upsert({
   158→          where: {
   159→            shop_sizeChartId_assignmentType_assignmentValue: {
   160→              shop,
   161→              sizeChartId: id,
   162→              assignmentType,
   163→              assignmentValue: assignmentValue.trim(),
   164→            },
   165→          },
   166→          create: {
   167→            shop,
   168→            sizeChartId: id,
   169→            assignmentType,
   170→            assignmentValue: assignmentValue.trim(),
   171→          },
   172→          update: {},
   173→        });
   174→        return { ok: true, message: "Assignment added" };
   175→      }
   176→
   177→      case "remove-assignment": {
   178→        const assignmentId = formData.get("assignmentId") as string;
   179→        await db.chartAssignment.deleteMany({
   180→          where: { id: assignmentId, shop },
   181→        });
   182→        return { ok: true, message: "Assignment removed" };
   183→      }
   184→
   185→      case "delete": {
   186→        await db.sizeChart.deleteMany({ where: { id, shop } });
   187→        return redirect("/app/size-charts");
   188→      }
   189→
   190→      default:
   191→        return { error: "Unknown action" };
   192→    }
   193→  } catch (error) {
   194→    const message =
   195→      error instanceof Error ? error.message : "An unexpected error occurred";
   196→    if (message.includes("Unique constraint")) {
   197→      return { error: "A chart with this name already exists." };
   198→    }
   199→    return { error: message };
   200→  }
   201→};

   236→function makeDefaultData(): ChartData {
   237→  return {
   238→    headers: ["Size", "Chest", "Waist", "Hips"],
   239→    units: { default: "in", available: ["in", "cm"] },
   240→    rows: [

   248→function parseChartData(raw: string): ChartData {
   249→  try {
   250→    const parsed = JSON.parse(raw) as ChartData;
   251→    if (
   252→      Array.isArray(parsed.headers) &&
   253→      parsed.units &&
   254→      Array.isArray(parsed.rows)
   255→    ) {
   256→      return parsed;
   257→    }
   258→  } catch {
   259→    // fall through
   260→  }
   261→  return makeDefaultData();
   262→}

   366→function TableEditor({ data, onChange }: TableEditorProps) {
   367→  const headerInputRefs = useRef<(HTMLInputElement | null)[]>([]);
   368→
   369→  const updateCell = (rowIndex: number, header: string, value: string) => {
   370→    const newRows = data.rows.map((row, i) => {
   371→      if (i !== rowIndex) return row;
   372→      return { ...row, [header]: value };
   373→    });
   374→    onChange({ ...data, rows: newRows });
   375→  };
   376→
   377→  const updateHeader = (headerIndex: number, value: string) => {
   378→    const oldHeader = data.headers[headerIndex];
   379→    const newHeaders = data.headers.map((h, i) =>
   380→      i === headerIndex ? value : h,
   381→    );
   382→    const newRows = data.rows.map((row) => {
   383→      const newRow: Record<string, string> = {};
   384→      data.headers.forEach((h, i) => {
   385→        const key = i === headerIndex ? value : h;
   386→        newRow[key] = row[h] ?? "";
   387→      });
   388→      return newRow;
   389→    });
   390→    onChange({ ...data, headers: newHeaders, rows: newRows });
   391→    void oldHeader;
   392→  };
   393→
   394→  const addRow = () => {
   395→    const emptyRow: Record<string, string> = {};
   396→    data.headers.forEach((h) => {
   397→      emptyRow[h] = "";
   398→    });
   399→    onChange({ ...data, rows: [...data.rows, emptyRow] });
   400→  };
   401→
   402→  const removeRow = (rowIndex: number) => {
   403→    onChange({ ...data, rows: data.rows.filter((_, i) => i !== rowIndex) });
   404→  };
   405→
   406→  const addColumn = () => {
   407→    const newHeader = `Column ${data.headers.length + 1}`;
   408→    const newHeaders = [...data.headers, newHeader];
   409→    const newRows = data.rows.map((row) => ({ ...row, [newHeader]: "" }));
   410→    onChange({ ...data, headers: newHeaders, rows: newRows });
   411→  };
   412→
   413→  const removeColumn = (headerIndex: number) => {
   414→    if (data.headers.length <= 1) return;
   415→    const removedHeader = data.headers[headerIndex];
   416→    const newHeaders = data.headers.filter((_, i) => i !== headerIndex);
   417→    const newRows = data.rows.map((row) => {
   418→      const newRow: Record<string, string> = {};
   419→      newHeaders.forEach((h) => {
   420→        newRow[h] = row[h] ?? "";
   421→      });
   422→      return newRow;
   423→    });
   424→    onChange({ ...data, headers: newHeaders, rows: newRows });
   425→    void removedHeader;
   426→  };
   427→
   428→  const toggleUnit = (unit: string) => {
   429→    onChange({ ...data, units: { ...data.units, default: unit } });
   430→  };
   431→
   432→  return (
   433→    <div>
   434→      {/* Toolbar — unit toggle + row/col count */}
   435→      <div
   436→        style={{
   437→          display: "flex",
   438→          alignItems: "center",
   439→          gap: "8px",
   440→          marginBottom: "10px",
   441→        }}
   442→      >
   443→        <span
   444→          style={{
   445→            fontSize: "12px",
   446→            fontWeight: 600,
   447→            color: "#6d7175",
   448→            textTransform: "uppercase",
   449→            letterSpacing: "0.04em",
   450→          }}
   451→        >
   452→          Units
   453→        </span>
   454→        <div
   455→          style={{
   456→            display: "flex",
   457→            gap: "4px",
   458→            background: "#f6f6f7",
   459→            borderRadius: "20px",
   460→            padding: "3px",
   461→            border: "1px solid #e1e3e5",
   462→          }}
   463→        >
   464→          {data.units.available.map((unit) => (
   465→            <button
   466→              key={unit}
   467→              type="button"
   468→              onClick={() => toggleUnit(unit)}
   469→              style={{
   470→                padding: "3px 14px",
   471→                borderRadius: "16px",
   472→                border: "none",
   473→                background:
   474→                  data.units.default === unit ? "#2c6ecb" : "transparent",
   475→                color: data.units.default === unit ? "#ffffff" : "#6d7175",
   476→                fontWeight: data.units.default === unit ? 700 : 500,
   477→                cursor: "pointer",
   478→                fontSize: "12px",
   479→                lineHeight: "20px",
   480→                transition: "all 0.12s ease",
   481→                fontFamily: "inherit",
   482→              }}
   483→            >
   484→              {unit}
   485→            </button>
   486→          ))}
   487→        </div>
   488→        <div
   489→          style={{
   490→            marginLeft: "auto",
   491→            fontSize: "12px",
   492→            color: "#8c9196",
   493→            fontWeight: 500,
   494→            background: "#f6f6f7",
   495→            border: "1px solid #e1e3e5",
