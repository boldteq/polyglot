    46→    46→      chart: null as null,
    47→    47→      assignments: [] as Assignment[],
    48→    48→      isNew: true,
    49→    49→    };

    84→    84→export const action = async ({ request, params }: ActionFunctionArgs) => {
    85→    85→  const { session } = await authenticate.admin(request);
    86→    86→  const shop = session.shop;
    87→    87→  const id = params.id;
    88→    88→  if (!id) return { error: "Missing chart ID" };
    89→    89→  const formData = await request.formData();
    90→    90→  const intent = formData.get("intent") as string;
    91→    91→
    92→    92→  try {
    93→    93→    switch (intent) {
    94→    94→      case "save": {
    95→    95→        const name = formData.get("name") as string;
    96→    96→        const category = formData.get("category") as string;
    97→    97→        const status = formData.get("status") as string;
    98→    98→        const dataRaw = formData.get("data") as string;
    99→    99→
   100→   100→        if (!name?.trim()) return { error: "Chart name is required" };
   101→   101→
   102→   102→        let parsedData: ChartData;
   103→   103→        try {
   104→   104→          parsedData = JSON.parse(dataRaw) as ChartData;
   105→   105→        } catch {
   106→   106→          return { error: "Invalid chart data. Please try again." };
   107→   107→        }
   108→   108→
   109→   109→        if (id === "new") {
   110→   110→          const chart = await db.sizeChart.create({
   111→   111→            data: {
   112→   112→              shop,
   113→   113→              name: name.trim(),
   114→   114→              category,
   115→   115→              status,
   116→   116→              data: JSON.stringify(parsedData),
   117→   117→            },
   118→   118→          });
   119→   119→          return redirect(`/app/size-charts/${chart.id}?saved=1`);
   120→   120→        }
   121→   121→
   122→   122→        const existing = await db.sizeChart.findFirst({ where: { id, shop } });
   123→   123→        if (!existing) return { error: "Chart not found" };
   124→   124→
   125→   125→        await db.sizeChart.update({
   126→   126→          where: { id },
   127→   127→          data: {
   128→   128→            name: name.trim(),
   129→   129→            category,
   130→   130→            status,
   131→   131→            data: JSON.stringify(parsedData),
   132→   132→          },
   133→   133→        });
   134→   134→        return { ok: true, message: "Chart saved successfully" };
   135→   135→      }
   136→   136→
   137→   137→      case "toggle-status": {
   138→   138→        const chart = await db.sizeChart.findFirst({ where: { id, shop } });
   139→   139→        if (!chart) return { error: "Chart not found" };
   140→   140→        const newStatus = chart.status === "active" ? "draft" : "active";
   141→   141→        await db.sizeChart.update({ where: { id }, data: { status: newStatus } });
   142→   142→        return {
   143→   143→          ok: true,
   144→   144→          message: newStatus === "active" ? "Chart published" : "Chart unpublished",
   145→   145→          newStatus,
   146→   146→        };
   147→   147→      }
   148→   148→
   149→   149→      case "add-assignment": {
   150→   150→        const assignmentType = formData.get("assignmentType") as string;
   151→   151→        const assignmentValue = formData.get("assignmentValue") as string;
   152→   152→
   153→   153→        if (!assignmentValue?.trim()) {
   154→   154→          return { error: "Assignment value is required" };
   155→   155→        }
   156→   156→
   157→   157→        await db.chartAssignment.upsert({
   158→   158→          where: {
   159→   159→            shop_sizeChartId_assignmentType_assignmentValue: {
   160→   160→              shop,
   161→   161→              sizeChartId: id,
   162→   162→              assignmentType,
   163→   163→              assignmentValue: assignmentValue.trim(),
   164→   164→            },
   165→   165→          },
   166→   166→          create: {
   167→   167→            shop,
   168→   168→            sizeChartId: id,
   169→   169→            assignmentType,
   170→   170→            assignmentValue: assignmentValue.trim(),
   171→   171→          },
   172→   172→          update: {},
   173→   173→        });
   174→   174→        return { ok: true, message: "Assignment added" };
   175→   175→      }
   176→   176→
   177→   177→      case "remove-assignment": {
   178→   178→        const assignmentId = formData.get("assignmentId") as string;
   179→   179→        await db.chartAssignment.deleteMany({
   180→   180→          where: { id: assignmentId, shop },
   181→   181→        });
   182→   182→        return { ok: true, message: "Assignment removed" };
   183→   183→      }
   184→   184→
   185→   185→      case "delete": {
   186→   186→        await db.sizeChart.deleteMany({ where: { id, shop } });
   187→   187→        return redirect("/app/size-charts");
   188→   188→      }
   189→   189→
   190→   190→      default:
   191→   191→        return { error: "Unknown action" };
   192→   192→    }
   193→   193→  } catch (error) {
   194→   194→    const message =
   195→   195→      error instanceof Error ? error.message : "An unexpected error occurred";
   196→   196→    if (message.includes("Unique constraint")) {
   197→   197→      return { error: "A chart with this name already exists." };
   198→   198→    }
   199→   199→    return { error: message };
   200→   200→  }
   201→   201→};

   248→   248→function parseChartData(raw: string): ChartData {
   249→   249→  try {
   250→   250→    const parsed = JSON.parse(raw) as ChartData;
   251→   251→    if (
   252→   252→      Array.isArray(parsed.headers) &&
   253→   253→      parsed.units &&
   254→   254→      Array.isArray(parsed.rows)
   255→   255→    ) {
   256→   256→      return parsed;
   257→   257→    }
   258→   258→  } catch {
   259→   259→    // fall through
   260→   260→  }
   261→   261→  return makeDefaultData();
   262→   262→}

   277→   277→function TableEditor({ data, onChange }: TableEditorProps) {
   278→   278→  const headerInputRefs = useRef<(HTMLInputElement | null)[]>([]);
   279→   279→
   280→   280→  const updateCell = (rowIndex: number, header: string, value: string) => {
   281→   281→    const newRows = data.rows.map((row, i) => {
   282→   282→      if (i !== rowIndex) return row;
   283→   283→      return { ...row, [header]: value };
   284→   284→    });
   285→   285→    onChange({ ...data, rows: newRows });
   286→   286→  };
   287→   287→
   288→   288→  const updateHeader = (headerIndex: number, value: string) => {
   289→   289→    const oldHeader = data.headers[headerIndex];
   290→   290→    const newHeaders = data.headers.map((h, i) =>
   291→   291→      i === headerIndex ? value : h,
   292→   292→    );
   293→   293→    const newRows = data.rows.map((row) => {
   294→   294→      const newRow: Record<string, string> = {};
   295→   295→      data.headers.forEach((h, i) => {
   296→   296→        const key = i === headerIndex ? value : h;
   297→   297→        newRow[key] = row[h] ?? "";
   298→   298→      });
   299→   299→      return newRow;
   300→   300→    });
   301→   301→    onChange({ ...data, headers: newHeaders, rows: newRows });
   302→   302→    void oldHeader;
   303→   303→  };
   304→   304→
   305→   305→  const addRow = () => {
   306→   306→    const emptyRow: Record<string, string> = {};
   307→   307→    data.headers.forEach((h) => {
   308→   308→      emptyRow[h] = "";
   309→   309→    });
   310→   310→    onChange({ ...data, rows: [...data.rows, emptyRow] });
   311→   311→  };
   312→   312→
   313→   313→  const removeRow = (rowIndex: number) => {
   314→   314→    onChange({ ...data, rows: data.rows.filter((_, i) => i !== rowIndex) });
   315→   315→  };
   316→   316→
   317→   317→  const addColumn = () => {
   318→   318→    const newHeader = `Column ${data.headers.length + 1}`;
   319→   319→    const newHeaders = [...data.headers, newHeader];
   320→   320→    const newRows = data.rows.map((row) => ({ ...row, [newHeader]: "" }));
   321→   321→    onChange({ ...data, headers: newHeaders, rows: newRows });
   322→   322→  };
   323→   323→
   324→   324→  const removeColumn = (headerIndex: number) => {
   325→   325→    if (data.headers.length <= 1) return;
   326→   326→    const removedHeader = data.headers[headerIndex];
   327→   327→    const newHeaders = data.headers.filter((_, i) => i !== headerIndex);
   328→   328→    const newRows = data.rows.map((row) => {
   329→   329→      const newRow: Record<string, string> = {};
   330→   330→      newHeaders.forEach((h) => {
   331→   331→        newRow[h] = row[h] ?? "";
   332→   332→      });
   333→   333→      return newRow;
   334→   334→    });
   335→   335→    onChange({ ...data, headers: newHeaders, rows: newRows });
   336→   336→    void removedHeader;
   337→   337→  };
   338→   338→
   339→   339→  const toggleUnit = (unit: string) => {
   340→   340→    onChange({ ...data, units: { ...data.units, default: unit } });
   341→   341→  };
   342→   342→
   343→   343→  return (
   344→   344→    <div>
   345→   345→      {/* Toolbar — unit toggle + row/col count */}
   346→   346→      <div
   347→   347→        style={{
   348→   348→          display: "flex",
   349→   349→          alignItems: "center",
   350→   350→          gap: "8px",
   351→   351→          marginBottom: "10px",
   352→   352→        }}
   353→   353→      >
   354→   354→        <span
   355→   355→          style={{
   356→   356→            fontSize: "12px",
   357→   357→            fontWeight: 600,
   358→   358→            color: "#6d7175",
   359→   359→            textTransform: "uppercase",
   360→   360→            letterSpacing: "0.04em",
   361→   361→          }}
   362→   362→        >
   363→   363→          Units
   364→   364→        </span>
   365→   365→        <div
   366→   366→          style={{
   367→   367→            display: "flex",
   368→   368→            gap: "4px",
   369→   369→            background: "#f6f6f7",
   370→   370→            borderRadius: "20px",
   371→   371→            padding: "3px",
   372→   372→            border: "1px solid #e1e3e5",
   373→   373→          }}
   374→   374→        >
   375→   375→          {data.units.available.map((unit) => (
   376→   376→            <button
   377→   377→              key={unit}
   378→   378→              type="button"
   379→   379→              onClick={() => toggleUnit(unit)}
   380→   380→              style={{

   387→   387→                fontWeight: data.units.default === unit ? 700 : 500,
