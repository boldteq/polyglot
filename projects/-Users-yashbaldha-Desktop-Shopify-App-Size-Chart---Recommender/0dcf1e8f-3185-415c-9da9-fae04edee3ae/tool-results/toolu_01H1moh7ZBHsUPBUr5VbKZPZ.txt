    10→    10→export const loader = async ({ request }: LoaderFunctionArgs) => {
    11→    11→  const { session } = await authenticate.admin(request);
    12→    12→  const shop = session.shop;
    13→    13→
    14→    14→  const [charts, subscription] = await Promise.all([
    15→    15→    db.sizeChart.findMany({
    16→    16→      where: { shop },
    17→    17→      orderBy: { updatedAt: "desc" },
    18→    18→      include: { assignments: true },
    19→    19→    }),
    20→    20→    db.subscription.findUnique({ where: { shop } }),
    21→    21→  ]);
    22→    22→
    23→    23→  const plan = getPlan(subscription?.planId ?? "free");
    24→    24→  const templates = SIZE_CHART_TEMPLATES.map((t) => ({
    25→    25→    name: t.name,
    26→    26→    category: t.category,
    27→    27→    gender: t.gender,
    28→    28→    rowCount: t.data.rows.length,
    29→    29→  }));
    30→    30→  const categories = getCategories();
    31→    31→
    32→    32→  return {
    33→    33→    charts,
    34→    34→    plan,
    35→    35→    templates,
    36→    36→    categories,
    37→    37→    canCreate: canCreateChart(plan.id, charts.length),

    41→    41→export const action = async ({ request }: ActionFunctionArgs) => {
    42→    42→  const { session } = await authenticate.admin(request);
    43→    43→  const shop = session.shop;
    44→    44→  const formData = await request.formData();
    45→    45→  const intent = formData.get("intent") as string;
    46→    46→
    47→    47→  try {
    48→    48→    switch (intent) {
    49→    49→      case "create-from-template": {
    50→    50→        const templateName = formData.get("templateName") as string;
    51→    51→        const template = SIZE_CHART_TEMPLATES.find((t) => t.name === templateName);
    52→    52→        if (!template) return { error: "Template not found" };
    53→    53→
    54→    54→        const subscription = await db.subscription.findUnique({ where: { shop } });
    55→    55→        const plan = getPlan(subscription?.planId ?? "free");
    56→    56→        const chartCount = await db.sizeChart.count({ where: { shop } });
    57→    57→        if (!canCreateChart(plan.id, chartCount)) {
    58→    58→          return {
    59→    59→            error: `Free plan allows max ${plan.limits.maxCharts} charts. Upgrade to create more.`,
    60→    60→          };
    61→    61→        }
    62→    62→
    63→    63→        let name = template.name;
    64→    64→        const existing = await db.sizeChart.findUnique({
    65→    65→          where: { shop_name: { shop, name } },
    66→    66→        });
    67→    67→        if (existing) name = `${template.name} (${Date.now().toString(36)})`;
    68→    68→
    69→    69→        const created = await db.sizeChart.create({
    70→    70→          data: {
    71→    71→            shop,
    72→    72→            name,
    73→    73→            category: template.category,
    74→    74→            data: JSON.stringify(template.data),
    75→    75→            status: "active",
    76→    76→          },
    77→    77→        });
    78→    78→        return { ok: true, message: `"${name}" created successfully`, redirectId: created.id };
    79→    79→      }
    80→    80→
    81→    81→      case "create-blank": {
    82→    82→        const name = (formData.get("name") as string) || "Untitled Size Chart";
    83→    83→        const category = (formData.get("category") as string) || "general";
    84→    84→
    85→    85→        const subscription = await db.subscription.findUnique({ where: { shop } });
    86→    86→        const plan = getPlan(subscription?.planId ?? "free");
    87→    87→        const chartCount = await db.sizeChart.count({ where: { shop } });
    88→    88→        if (!canCreateChart(plan.id, chartCount)) {
    89→    89→          return {
    90→    90→            error: `Free plan allows max ${plan.limits.maxCharts} charts. Upgrade to create more.`,
    91→    91→          };
    92→    92→        }
    93→    93→
    94→    94→        let uniqueName = name;
    95→    95→        const existing = await db.sizeChart.findUnique({
    96→    96→          where: { shop_name: { shop, name } },
    97→    97→        });
    98→    98→        if (existing) uniqueName = `${name} (${Date.now().toString(36)})`;
    99→    99→
   100→   100→        const defaultData = {
   101→   101→          headers: ["Size", "Chest", "Waist", "Hips"],
   102→   102→          units: { default: "in", available: ["in", "cm"] },
   103→   103→          rows: [
   104→   104→            { Size: "S", Chest: "", Waist: "", Hips: "" },
   105→   105→            { Size: "M", Chest: "", Waist: "", Hips: "" },
   106→   106→            { Size: "L", Chest: "", Waist: "", Hips: "" },
   107→   107→          ],
   108→   108→        };
   109→   109→
   110→   110→        const created = await db.sizeChart.create({
   111→   111→          data: {
   112→   112→            shop,
   113→   113→            name: uniqueName,
   114→   114→            category,
   115→   115→            data: JSON.stringify(defaultData),
   116→   116→            status: "draft",
   117→   117→          },
   118→   118→        });
   119→   119→        return { ok: true, message: `"${uniqueName}" created`, redirectId: created.id };
   120→   120→      }
   121→   121→
   122→   122→      case "delete": {
   123→   123→        const id = formData.get("id") as string;
   124→   124→        await db.sizeChart.deleteMany({ where: { id, shop } });
   125→   125→        return { ok: true, message: "Chart deleted" };
   126→   126→      }
   127→   127→
   128→   128→      case "import-csv": {
   129→   129→        const csvRaw = formData.get("csvData") as string;
   130→   130→        const chartName = (formData.get("name") as string) || "Imported Size Chart";
   131→   131→        const category = (formData.get("category") as string) || "general";
   132→   132→
   133→   133→        if (!csvRaw?.trim()) return { error: "CSV data is required" };
   134→   134→
   135→   135→        const subscription = await db.subscription.findUnique({ where: { shop } });
   136→   136→        const plan = getPlan(subscription?.planId ?? "free");
   137→   137→        const chartCount = await db.sizeChart.count({ where: { shop } });
   138→   138→        if (!canCreateChart(plan.id, chartCount)) {
   139→   139→          return { error: `Free plan allows max ${plan.limits.maxCharts} charts. Upgrade to create more.` };
   140→   140→        }
   141→   141→
   142→   142→        // Parse CSV
   143→   143→        const lines = csvRaw.trim().split(/\r?\n/).filter((l) => l.trim());
   144→   144→        if (lines.length < 2) return { error: "CSV must have a header row and at least one data row." };
   145→   145→
   146→   146→        const parseRow = (line: string): string[] => {
   147→   147→          const result: string[] = [];
   148→   148→          let current = "";
   149→   149→          let inQuotes = false;
   150→   150→          for (let i = 0; i < line.length; i++) {
   151→   151→            const ch = line[i];
   152→   152→            if (ch === '"') {
   153→   153→              if (inQuotes && line[i + 1] === '"') { current += '"'; i++; }
   154→   154→              else { inQuotes = !inQuotes; }
   155→   155→            } else if (ch === "," && !inQuotes) {
   156→   156→              result.push(current.trim());
   157→   157→              current = "";
   158→   158→            } else {
   159→   159→              current += ch;
   160→   160→            }
   161→   161→          }
   162→   162→          result.push(current.trim());
   163→   163→          return result;
   164→   164→        };
   165→   165→
   166→   166→        const headers = parseRow(lines[0]);
   167→   167→        if (headers.length < 1) return { error: "Could not parse CSV headers." };
   168→   168→
   169→   169→        const rows = lines.slice(1).map((line) => {
   170→   170→          const cells = parseRow(line);
   171→   171→          const row: Record<string, string> = {};
   172→   172→          headers.forEach((h, i) => { row[h] = cells[i] ?? ""; });
   173→   173→          return row;
   174→   174→        });
   175→   175→
   176→   176→        const data = {
   177→   177→          headers,
   178→   178→          units: { default: "in", available: ["in", "cm"] },
   179→   179→          rows,
   180→   180→        };
   181→   181→
   182→   182→        let uniqueName = chartName;
   183→   183→        const existingName = await db.sizeChart.findUnique({ where: { shop_name: { shop, name: chartName } } });
   184→   184→        if (existingName) uniqueName = `${chartName} (${Date.now().toString(36)})`;
   185→   185→
   186→   186→        const created = await db.sizeChart.create({
   187→   187→          data: { shop, name: uniqueName, category, data: JSON.stringify(data), status: "draft" },
   188→   188→        });
   189→   189→        return { ok: true, message: `Imported "${uniqueName}"`, redirectId: created.id };
   190→   190→      }
   191→   191→
   192→   192→      case "duplicate": {
   193→   193→        const sourceId = formData.get("id") as string;
   194→   194→        const source = await db.sizeChart.findFirst({ where: { id: sourceId, shop } });
   195→   195→        if (!source) return { error: "Chart not found" };
   196→   196→
   197→   197→        const subscription = await db.subscription.findUnique({ where: { shop } });
   198→   198→        const plan = getPlan(subscription?.planId ?? "free");
   199→   199→        const chartCount = await db.sizeChart.count({ where: { shop } });
   200→   200→        if (!canCreateChart(plan.id, chartCount)) {
   201→   201→          return { error: "Chart limit reached. Upgrade your plan." };
   202→   202→        }
   203→   203→
   204→   204→        const newName = `${source.name} (Copy)`;
   205→   205→        await db.sizeChart.create({
   206→   206→          data: {
   207→   207→            shop,
   208→   208→            name: newName,
   209→   209→            type: source.type,
   210→   210→            category: source.category,
   211→   211→            data: source.data,
   212→   212→            status: "draft",
   213→   213→          },
   214→   214→        });
   215→   215→        return { ok: true, message: `Duplicated as "${newName}"` };
   216→   216→      }
   217→   217→
   218→   218→      case "toggle-status": {
   219→   219→        const chartId = formData.get("id") as string;
   220→   220→        const chart = await db.sizeChart.findFirst({ where: { id: chartId, shop } });
   221→   221→        if (!chart) return { error: "Chart not found" };
   222→   222→
   223→   223→        const newStatus = chart.status === "active" ? "draft" : "active";
   224→   224→        await db.sizeChart.update({ where: { id: chartId }, data: { status: newStatus } });
   225→   225→        return {
   226→   226→          ok: true,
   227→   227→          message: `Chart ${newStatus === "active" ? "published" : "unpublished"}`,
   228→   228→        };
   229→   229→      }
   230→   230→
   231→   231→      default:
   232→   232→        return { error: "Unknown action" };
   233→   233→    }
   234→   234→  } catch (error) {
   235→   235→    const message = error instanceof Error ? error.message : "An unexpected error occurred";
   236→   236→    if (message.includes("Unique constraint")) {
   237→   237→      return { error: "A chart with this name already exists." };
   238→   238→    }
   239→   239→    return { error: message };
   240→   240→  }
   241→   241→};
