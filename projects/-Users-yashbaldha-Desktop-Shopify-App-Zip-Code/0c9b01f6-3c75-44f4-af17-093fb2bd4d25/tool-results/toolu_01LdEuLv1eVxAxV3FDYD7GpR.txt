    48→// Helper: sync widget config to an App Installation metafield so the
    49→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    50→//
    51→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    52→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    53→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    54→async function syncConfigMetafield(
    55→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    56→  configData: Record<string, unknown>,
    57→) {
    58→  try {
    59→    // Step 1: get the App Installation GID
    60→    const installResponse = await admin.graphql(
    61→      `query { currentAppInstallation { id } }`,
    62→    );
    63→    const installJson = (await installResponse.json()) as {
    64→      data: { currentAppInstallation: { id: string } };
    65→    };
    66→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    67→
    68→    // Step 2: write the metafield to the App Installation
    69→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    70→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    71→    const metaResponse = await admin.graphql(
    72→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    73→        metafieldsSet(metafields: $metafields) {
    74→          metafields { id namespace key }
    75→          userErrors { field message }
    76→        }
    77→      }`,
    78→      {
    79→        variables: {
    80→          metafields: [
    81→            {
    82→              ownerId: appInstallationGid,
    83→              namespace: "zip_checker",
    84→              key: "widget_config",
    85→              type: "json",
    86→              value: JSON.stringify(configData),
    87→            },
    88→          ],
    89→        },
    90→      },
    91→    );
    92→    const metaJson = (await metaResponse.json()) as {
    93→      data: {
    94→        metafieldsSet: {
    95→          metafields: { id: string; namespace: string; key: string }[];
    96→          userErrors: { field: string; message: string }[];
    97→        };
    98→      };
    99→    };
   100→
   101→    if (metaJson.data.metafieldsSet.userErrors.length > 0) {
   102→      console.error(
   103→        "metafieldsSet userErrors:",
   104→        metaJson.data.metafieldsSet.userErrors,
   105→      );
   106→    }
   107→  } catch (e) {
   108→    // Non-fatal — the API fallback still works
   109→    console.error("Failed to sync widget config metafield:", e);
   110→  }
   111→}
