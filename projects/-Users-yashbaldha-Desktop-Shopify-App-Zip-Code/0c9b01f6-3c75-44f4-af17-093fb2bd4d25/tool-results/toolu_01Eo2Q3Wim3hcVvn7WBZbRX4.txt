    49→/** RFC-4180 compliant CSV row parser — handles quoted fields containing commas. */
    50→function parseCsvRow(line: string): string[] {
    51→  const cols: string[] = [];
    52→  let cur = "";
    53→  let inQuotes = false;
    54→  for (let i = 0; i < line.length; i++) {
    55→    const ch = line[i];
    56→    if (ch === '"') {
    57→      if (inQuotes && line[i + 1] === '"') {
    58→        cur += '"';
    59→        i++;
    60→      } else {
    61→        inQuotes = !inQuotes;
    62→      }
    63→    } else if (ch === "," && !inQuotes) {
    64→      cols.push(cur.trim());
    65→      cur = "";
    66→    } else {
    67→      cur += ch;
    68→    }
    69→  }
    70→  cols.push(cur.trim());
    71→  return cols;
    72→}

   113→export const action = async ({ request }: ActionFunctionArgs) => {

   122→    const zone = String(formData.get("zone") || "").trim();
   123→    const message = String(formData.get("message") || "").trim();
   124→    const eta = String(formData.get("eta") || "").trim();
   125→    const type = String(formData.get("type") || "allowed");
   126→    const codAvailable =
   127→      formData.get("codAvailable") === "true"
   128→        ? true
   129→        : formData.get("codAvailable") === "false"
   130→          ? false
   131→          : null;
   132→    const returnPolicy = (formData.get("returnPolicy") as string | null) || null;
   133→
   134→    if (!zipCode) return { error: "Zip code is required." };
   135→
   136→    const subscription = await getShopSubscription(shop);
   137→    const limits = PLAN_LIMITS[subscription.planTier];
   138→
   139→    if (type === "blocked" && !limits.allowBlocked) {
   140→      return {
   141→        error:
   142→          "Blocked zip codes are not available on your current plan. Upgrade to Pro or Ultimate to use blocked zip codes.",
   143→        upgradeRequired: true,
   144→      };
   145→    }
   146→
   147→    const currentCount = await db.zipCode.count({ where: { shop } });
   148→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   149→      const upgradeTarget =
   150→        subscription.planTier === "free" ? "Starter" : "Pro";
   151→      return {
   152→        error: `You have reached the ${limits.maxZipCodes} zip code limit on the ${limits.label} plan. Upgrade to ${upgradeTarget} for a higher limit.`,
   153→        upgradeRequired: true,
   154→      };
   155→    }
   156→
   157→    try {
   158→      await db.zipCode.create({
   159→        data: {
   160→          shop,
   161→          zipCode,
   162→          label: label || null,
   163→          zone: zone || null,
   164→          message: message || null,
   165→          eta: eta || null,
   166→          type,
   167→          codAvailable: codAvailable ?? undefined,
   168→          returnPolicy: returnPolicy ?? undefined,
   169→        },
   170→      });
   171→      return { success: true, action: "added", zipCode };
   172→    } catch {
   173→      return { error: `Zip code "${zipCode}" already exists.` };
   174→    }
   175→  }
   176→
   177→  if (intent === "delete") {
   178→    const id = String(formData.get("id"));
   179→    try {
   180→      await db.zipCode.delete({ where: { id } });
   181→      return { success: true, action: "deleted" };
   182→    } catch {
   183→      return { error: "Failed to delete zip code." };
   184→    }
   185→  }
   186→
   187→  if (intent === "update") {
   188→    const id = String(formData.get("id"));
   189→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   190→    const label = String(formData.get("label") || "").trim();
   191→    const zone = String(formData.get("zone") || "").trim();
   192→    const message = String(formData.get("message") || "").trim();
   193→    const eta = String(formData.get("eta") || "").trim();
   194→    const type = String(formData.get("type") || "allowed");
   195→    const isActive = formData.get("isActive") === "true";
   196→    const codAvailable =
   197→      formData.get("codAvailable") === "true"
   198→        ? true
   199→        : formData.get("codAvailable") === "false"
   200→          ? false
   201→          : null;
   202→    const returnPolicy = (formData.get("returnPolicy") as string | null) || null;
   203→
   204→    if (!zipCode) return { error: "Zip code is required." };
   205→
   206→    try {
   207→      await db.zipCode.update({
   208→        where: { id },
   209→        data: {
   210→          zipCode,
   211→          label: label || null,
   212→          zone: zone || null,
   213→          message: message || null,
   214→          eta: eta || null,
   215→          type,
   216→          isActive,
   217→          codAvailable,
   218→          returnPolicy,
   219→        },
   220→      });
   221→      return { success: true, action: "updated", zipCode };
   222→    } catch {
   223→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   224→    }
   225→  }
   226→
   227→  if (intent === "toggle") {
   228→    const id = String(formData.get("id"));
   229→    const isActive = formData.get("isActive") === "true";
   230→    try {
   231→      await db.zipCode.update({
   232→        where: { id },
   233→        data: { isActive: !isActive },
   234→      });
   235→      return { success: true, action: "toggled" };
   236→    } catch {
   237→      return { error: "Failed to toggle zip code status." };
   238→    }
   239→  }
   240→
   241→  if (intent === "check") {
   242→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   243→    try {
   244→      const found = await db.zipCode.findUnique({
   245→        where: { shop_zipCode: { shop, zipCode } },
   246→      });
   247→      if (!found) return { checkResult: { found: false, zipCode } };
   248→      return { checkResult: { found: true, zipCode, record: found } };
   249→    } catch {
   250→      return { error: "Failed to check zip code." };
   251→    }
   252→  }
   253→
   254→  if (intent === "bulk-import") {
   255→    const csvData = String(formData.get("csvData") || "");
   256→    if (!csvData.trim()) return { error: "No CSV data provided." };
   257→
   258→    const subscription = await getShopSubscription(shop);
   259→    const limits = PLAN_LIMITS[subscription.planTier];
   260→
   261→    if (!limits.csvImport) {
   262→      const upgradeTarget =
   263→        subscription.planTier === "free" ? "Starter" : "Pro";
   264→      return {
   265→        error: `CSV import is not available on the ${limits.label} plan. Upgrade to ${upgradeTarget} to import zip codes via CSV.`,
   266→        upgradeRequired: true,
   267→      };
   268→    }
   269→
   270→    const lines = csvData
   271→      .split("\n")
   272→      .map((l) => l.trim())
   273→      .filter((l) => l.length > 0);
   274→
   275→    // Skip header row if present
   276→    const startIdx =
   277→      lines.length > 0 &&
   278→      lines[0].toLowerCase().includes("zip")
   279→        ? 1
   280→        : 0;
   281→
   282→    const currentCount = await db.zipCode.count({ where: { shop } });
   283→    let imported = 0;
   284→    let skipped = 0;
   285→    const errors: string[] = [];
   286→
   287→    for (let i = startIdx; i < lines.length; i++) {
   288→      const cols = parseCsvRow(lines[i]);
   289→      const zipCode = (cols[0] || "").toUpperCase();
   290→      if (!zipCode) continue;
   291→
   292→      const zone = cols[1] || null;
   293→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   294→      const message = cols[3] || null;
   295→      const eta = cols[4] || null;
   296→      const codRaw = (cols[5] || "").toLowerCase().trim();
   297→      const codAvailable =
   298→        codRaw === "yes" || codRaw === "true"
   299→          ? true
   300→          : codRaw === "no" || codRaw === "false"
   301→            ? false
   302→            : null;
   303→      const returnPolicy = cols[6] || null;
   304→
   305→      if (type === "blocked" && !limits.allowBlocked) {
   306→        skipped++;
   307→        continue;
   308→      }
   309→
   310→      if (
   311→        limits.maxZipCodes !== Infinity &&
   312→        currentCount + imported >= limits.maxZipCodes
   313→      ) {
   314→        const upgradeTarget =
   315→          subscription.planTier === "free" ? "Starter" : "Pro";
   316→        errors.push(
   317→          `Reached the ${limits.maxZipCodes} zip code limit on the ${limits.label} plan. Upgrade to ${upgradeTarget} for a higher limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   318→        );
   319→        break;
   320→      }
   321→
   322→      try {
   323→        await db.zipCode.upsert({
   324→          where: { shop_zipCode: { shop, zipCode } },
   325→          create: {
   326→            shop,
   327→            zipCode,
   328→            zone,
   329→            type,
   330→            message,
   331→            eta,
   332→            codAvailable,
   333→            returnPolicy,
   334→          },
   335→          update: {
   336→            zone: zone ?? undefined,
   337→            type,
   338→            message: message ?? undefined,
   339→            eta: eta ?? undefined,
   340→            codAvailable,
   341→            returnPolicy: returnPolicy ?? undefined,
   342→          },
   343→        });
   344→        imported++;
   345→      } catch {
   346→        skipped++;
   347→      }
   348→    }
   349→
   350→    return {
   351→      success: true,
   352→      action: "bulk-import",
   353→      imported,
   354→      skipped,
   355→      errors,
   356→    };
   357→  }
   358→
   359→  if (intent === "export") {
   360→    const subscription = await getShopSubscription(shop);
   361→    const limits = PLAN_LIMITS[subscription.planTier];
   362→
   363→    if (!limits.csvExport) {
   364→      return {
   365→        error: `CSV export is not available on the ${limits.label} plan. Upgrade to Pro or Ultimate to export your zip codes.`,
   366→        upgradeRequired: true,
   367→      };
   368→    }
   369→
   370→    try {
   371→      const zipCodes = await db.zipCode.findMany({
   372→        where: { shop },
   373→        orderBy: { createdAt: "desc" },
   374→      });
   375→
   376→      const header = "Zip Code,Zone,Status,Message,ETA,Active,COD,Return Policy";
   377→      const rows = zipCodes.map(
   378→        (z) =>
   379→          `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}","${z.codAvailable === true ? "Yes" : z.codAvailable === false ? "No" : ""}","${z.returnPolicy || ""}"`,
   380→      );
   381→      const csv = [header, ...rows].join("\n");
   382→
   383→      return { success: true, action: "export", csv };
   384→    } catch {
   385→      return { error: "Failed to export zip codes." };
   386→    }
   387→  }
   388→
   389→  if (intent === "range-import") {
   390→    const startZip = String(formData.get("startZip") || "").trim().toUpperCase();
   391→    const endZip = String(formData.get("endZip") || "").trim().toUpperCase();
   392→    const zone = (formData.get("zone") as string | null) || null;
   393→    const type = (formData.get("type") as string) === "blocked" ? "blocked" : "allowed";
   394→    const message = (formData.get("message") as string | null) || null;
   395→    const eta = (formData.get("eta") as string | null) || null;
   396→
   397→    // Validate: must be 5-digit numeric zips
   398→    const ZIP_RE = /^\d{5}$/;
   399→    if (!ZIP_RE.test(startZip) || !ZIP_RE.test(endZip)) {
   400→      return { error: "ZIP range import only supports 5-digit US ZIP codes (e.g., 10001)." };
   401→    }
   402→
   403→    const start = parseInt(startZip, 10);
   404→    const end = parseInt(endZip, 10);
   405→
   406→    if (end < start) {
   407→      return { error: "End ZIP must be greater than or equal to Start ZIP." };
   408→    }
   409→
   410→    const rangeSize = end - start + 1;
   411→    const MAX_RANGE = 500;
   412→    if (rangeSize > MAX_RANGE) {
   413→      return { error: `Range too large. Maximum ${MAX_RANGE} zip codes per import (got ${rangeSize}).` };
   414→    }
   415→
   416→    // Check plan limits
   417→    const subscription = await getShopSubscription(shop);
   418→    const limits = PLAN_LIMITS[subscription.planTier];
   419→
   420→    if (!limits.csvImport) {
   421→      const upgradeTarget =
   422→        subscription.planTier === "free" ? "Starter" : "Pro";
   423→      return {
   424→        error: `ZIP range import is not available on the ${limits.label} plan. Upgrade to ${upgradeTarget} to use bulk import features.`,
   425→        upgradeRequired: true,
   426→      };
   427→    }
   428→
   429→    if (type === "blocked" && !limits.allowBlocked) {
   430→      return {
   431→        error: "Blocked zip codes are not available on your current plan. Upgrade to Pro or Ultimate to use blocked zip codes.",
   432→        upgradeRequired: true,
   433→      };
   434→    }
   435→
   436→    const currentCount = await db.zipCode.count({ where: { shop } });
   437→    if (limits.maxZipCodes !== Infinity && currentCount + rangeSize > limits.maxZipCodes) {
   438→      const upgradeTarget =
   439→        subscription.planTier === "free" ? "Starter" : "Pro";
   440→      return {
   441→        error: `This range would exceed your ${limits.label} plan limit of ${limits.maxZipCodes} zip codes. You have ${limits.maxZipCodes - currentCount} slots remaining. Upgrade to ${upgradeTarget} for a higher limit.`,
   442→        upgradeRequired: true,
   443→      };
   444→    }
   445→
   446→    // Generate and upsert all zips in range
   447→    let imported = 0;
   448→    for (let i = start; i <= end; i++) {
   449→      const zipCode = String(i).padStart(5, "0");
   450→      try {
   451→        await db.zipCode.upsert({
   452→          where: { shop_zipCode: { shop, zipCode } },
   453→          create: { shop, zipCode, zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   454→          update: { zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   455→        });
   456→        imported++;
   457→      } catch {
   458→        // skip individual errors
   459→      }
   460→    }
   461→
   462→    return { success: true, action: "range-import", imported, total: rangeSize };
   463→  }
   464→
   465→  return null;
   466→};
