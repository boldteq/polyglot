    50→/** RFC-4180 compliant CSV row parser — handles quoted fields containing commas. */
    51→function parseCsvRow(line: string): string[] {
    52→  const cols: string[] = [];
    53→  let cur = "";
    54→  let inQuotes = false;
    55→  for (let i = 0; i < line.length; i++) {
    56→    const ch = line[i];
    57→    if (ch === '"') {
    58→      if (inQuotes && line[i + 1] === '"') {
    59→        cur += '"';
    60→        i++;
    61→      } else {
    62→        inQuotes = !inQuotes;
    63→      }
    64→    } else if (ch === "," && !inQuotes) {
    65→      cols.push(cur.trim());
    66→      cur = "";
    67→    } else {
    68→      cur += ch;
    69→    }
    70→  }
    71→  cols.push(cur.trim());
    72→  return cols;
    73→}

   114→export const action = async ({ request }: ActionFunctionArgs) => {
   115→  const { session } = await authenticate.admin(request);
   116→  const shop = session.shop;
   117→  const formData = await request.formData();
   118→  const intent = formData.get("intent");
   119→
   120→  if (intent === "add") {
   121→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   122→    const label = String(formData.get("label") || "").trim();
   123→    const zone = String(formData.get("zone") || "").trim();
   124→    const message = String(formData.get("message") || "").trim();
   125→    const eta = String(formData.get("eta") || "").trim();
   126→    const type = String(formData.get("type") || "allowed");
   127→    const codAvailable =
   128→      formData.get("codAvailable") === "true"
   129→        ? true
   130→        : formData.get("codAvailable") === "false"
   131→          ? false
   132→          : null;
   133→    const returnPolicy = (formData.get("returnPolicy") as string | null) || null;
   134→
   135→    if (!zipCode) return { error: "Zip code is required." };
   136→
   137→    const subscription = await getShopSubscription(shop);
   138→    const limits = PLAN_LIMITS[subscription.planTier];
   139→
   140→    if (type === "blocked" && !limits.allowBlocked) {
   141→      return {
   142→        error:
   143→          "Blocked zip codes are not available on the Free plan. Upgrade to Pro or Ultimate.",
   144→        upgradeRequired: true,
   145→      };
   146→    }
   147→
   148→    const currentCount = await db.zipCode.count({ where: { shop } });
   149→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   150→      return {
   151→        error: `You have reached the ${limits.maxZipCodes} zip code limit for the Free plan. Upgrade to Pro for unlimited zip codes.`,
   152→        upgradeRequired: true,
   153→      };
   154→    }
   155→
   156→    try {
   157→      await db.zipCode.create({
   158→        data: {
   159→          shop,
   160→          zipCode,
   161→          label: label || null,
   162→          zone: zone || null,
   163→          message: message || null,
   164→          eta: eta || null,
   165→          type,
   166→          codAvailable: codAvailable ?? undefined,
   167→          returnPolicy: returnPolicy ?? undefined,
   168→        },
   169→      });
   170→      return { success: true, action: "added", zipCode };
   171→    } catch {
   172→      return { error: `Zip code "${zipCode}" already exists.` };
   173→    }
   174→  }
   175→
   176→  if (intent === "delete") {
   177→    const id = String(formData.get("id"));
   178→    try {
   179→      await db.zipCode.delete({ where: { id } });
   180→      return { success: true, action: "deleted" };
   181→    } catch {
   182→      return { error: "Failed to delete zip code." };
   183→    }
   184→  }
   185→
   186→  if (intent === "update") {
   187→    const id = String(formData.get("id"));
   188→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   189→    const label = String(formData.get("label") || "").trim();
   190→    const zone = String(formData.get("zone") || "").trim();
   191→    const message = String(formData.get("message") || "").trim();
   192→    const eta = String(formData.get("eta") || "").trim();
   193→    const type = String(formData.get("type") || "allowed");
   194→    const isActive = formData.get("isActive") === "true";
   195→
   196→    if (!zipCode) return { error: "Zip code is required." };
   197→
   198→    try {
   199→      await db.zipCode.update({
   200→        where: { id },
   201→        data: {
   202→          zipCode,
   203→          label: label || null,
   204→          zone: zone || null,
   205→          message: message || null,
   206→          eta: eta || null,
   207→          type,
   208→          isActive,
   209→        },
   210→      });
   211→      return { success: true, action: "updated", zipCode };
   212→    } catch {
   213→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   214→    }
   215→  }
   216→
   217→  if (intent === "toggle") {
   218→    const id = String(formData.get("id"));
   219→    const isActive = formData.get("isActive") === "true";
   220→    try {
   221→      await db.zipCode.update({
   222→        where: { id },
   223→        data: { isActive: !isActive },
   224→      });
   225→      return { success: true, action: "toggled" };
   226→    } catch {
   227→      return { error: "Failed to toggle zip code status." };
   228→    }
   229→  }
   230→
   231→  if (intent === "check") {
   232→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   233→    try {
   234→      const found = await db.zipCode.findUnique({
   235→        where: { shop_zipCode: { shop, zipCode } },
   236→      });
   237→      if (!found) return { checkResult: { found: false, zipCode } };
   238→      return { checkResult: { found: true, zipCode, record: found } };
   239→    } catch {
   240→      return { error: "Failed to check zip code." };
   241→    }
   242→  }
   243→
   244→  if (intent === "bulk-import") {
   245→    const csvData = String(formData.get("csvData") || "");
   246→    if (!csvData.trim()) return { error: "No CSV data provided." };
   247→
   248→    const subscription = await getShopSubscription(shop);
   249→    const limits = PLAN_LIMITS[subscription.planTier];
   250→
   251→    const lines = csvData
   252→      .split("\n")
   253→      .map((l) => l.trim())
   254→      .filter((l) => l.length > 0);
   255→
   256→    // Skip header row if present
   257→    const startIdx =
   258→      lines.length > 0 &&
   259→      lines[0].toLowerCase().includes("zip")
   260→        ? 1
   261→        : 0;
   262→
   263→    const currentCount = await db.zipCode.count({ where: { shop } });
   264→    let imported = 0;
   265→    let skipped = 0;
   266→    const errors: string[] = [];
   267→
   268→    for (let i = startIdx; i < lines.length; i++) {
   269→      const cols = parseCsvRow(lines[i]);
   270→      const zipCode = (cols[0] || "").toUpperCase();
   271→      if (!zipCode) continue;
   272→
   273→      const zone = cols[1] || null;
   274→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   275→      const message = cols[3] || null;
   276→      const eta = cols[4] || null;
   277→
   278→      if (type === "blocked" && !limits.allowBlocked) {
   279→        skipped++;
   280→        continue;
   281→      }
   282→
   283→      if (
   284→        limits.maxZipCodes !== Infinity &&
   285→        currentCount + imported >= limits.maxZipCodes
   286→      ) {
   287→        errors.push(
   288→          `Reached the ${limits.maxZipCodes} zip code limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   289→        );
   290→        break;
   291→      }
   292→
   293→      try {
   294→        await db.zipCode.upsert({
   295→          where: { shop_zipCode: { shop, zipCode } },
   296→          create: {
   297→            shop,
   298→            zipCode,
   299→            zone,
   300→            type,
   301→            message,
   302→            eta,
   303→          },
   304→          update: {
   305→            zone: zone ?? undefined,
   306→            type,
   307→            message: message ?? undefined,
   308→            eta: eta ?? undefined,
   309→          },
   310→        });
   311→        imported++;
   312→      } catch {
   313→        skipped++;
   314→      }
   315→    }
   316→
   317→    return {
   318→      success: true,
   319→      action: "bulk-import",
   320→      imported,
   321→      skipped,
   322→      errors,
   323→    };
   324→  }
   325→
   326→  if (intent === "export") {
   327→    try {
   328→      const zipCodes = await db.zipCode.findMany({
   329→        where: { shop },
   330→        orderBy: { createdAt: "desc" },
   331→      });
   332→
   333→      const header = "Zip Code,Zone,Status,Message,ETA,Active";
   334→      const rows = zipCodes.map(
   335→        (z) =>
   336→          `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}"`,
   337→      );
   338→      const csv = [header, ...rows].join("\n");
   339→
   340→      return { success: true, action: "export", csv };
   341→    } catch {
   342→      return { error: "Failed to export zip codes." };
   343→    }
   344→  }
   345→
   346→  if (intent === "range-import") {
   347→    const startZip = String(formData.get("startZip") || "").trim().toUpperCase();
   348→    const endZip = String(formData.get("endZip") || "").trim().toUpperCase();
   349→    const zone = (formData.get("zone") as string | null) || null;
   350→    const type = (formData.get("type") as string) === "blocked" ? "blocked" : "allowed";
   351→    const message = (formData.get("message") as string | null) || null;
   352→    const eta = (formData.get("eta") as string | null) || null;
   353→
   354→    // Validate: must be 5-digit numeric zips
   355→    const ZIP_RE = /^\d{5}$/;
   356→    if (!ZIP_RE.test(startZip) || !ZIP_RE.test(endZip)) {
   357→      return { error: "ZIP range import only supports 5-digit US ZIP codes (e.g., 10001)." };
   358→    }
   359→
   360→    const start = parseInt(startZip, 10);
   361→    const end = parseInt(endZip, 10);
   362→
   363→    if (end < start) {
   364→      return { error: "End ZIP must be greater than or equal to Start ZIP." };
   365→    }
   366→
   367→    const rangeSize = end - start + 1;
   368→    const MAX_RANGE = 500;
   369→    if (rangeSize > MAX_RANGE) {
   370→      return { error: `Range too large. Maximum ${MAX_RANGE} zip codes per import (got ${rangeSize}).` };
   371→    }
   372→
   373→    // Check plan limits
   374→    const subscription = await getShopSubscription(shop);
   375→    const limits = PLAN_LIMITS[subscription.planTier];
   376→
   377→    if (type === "blocked" && !limits.allowBlocked) {
   378→      return { error: "Blocked zip codes require a Pro plan or higher." };
   379→    }
   380→
   381→    const currentCount = await db.zipCode.count({ where: { shop } });
   382→    if (limits.maxZipCodes !== Infinity && currentCount + rangeSize > limits.maxZipCodes) {
   383→      return { error: `This range would exceed your plan limit of ${limits.maxZipCodes} zip codes. You have ${limits.maxZipCodes - currentCount} slots remaining.` };
   384→    }
   385→
   386→    // Generate and upsert all zips in range
   387→    let imported = 0;
   388→    for (let i = start; i <= end; i++) {
   389→      const zipCode = String(i).padStart(5, "0");
   390→      try {
   391→        await db.zipCode.upsert({
   392→          where: { shop_zipCode: { shop, zipCode } },
   393→          create: { shop, zipCode, zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   394→          update: { zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   395→        });
   396→        imported++;
   397→      } catch {
   398→        // skip individual errors
   399→      }
   400→    }
   401→
   402→    return { success: true, action: "range-import", imported, total: rangeSize };
   403→  }
   404→
   405→  return null;
   406→};
