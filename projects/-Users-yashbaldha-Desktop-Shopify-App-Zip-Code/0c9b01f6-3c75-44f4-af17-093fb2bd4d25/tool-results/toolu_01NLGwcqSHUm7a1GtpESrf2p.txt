    49→    49→/** RFC-4180 compliant CSV row parser — handles quoted fields containing commas. */
    50→    50→function parseCsvRow(line: string): string[] {
    51→    51→  const cols: string[] = [];
    52→    52→  let cur = "";
    53→    53→  let inQuotes = false;
    54→    54→  for (let i = 0; i < line.length; i++) {
    55→    55→    const ch = line[i];
    56→    56→    if (ch === '"') {
    57→    57→      if (inQuotes && line[i + 1] === '"') {
    58→    58→        cur += '"';
    59→    59→        i++;
    60→    60→      } else {
    61→    61→        inQuotes = !inQuotes;
    62→    62→      }
    63→    63→    } else if (ch === "," && !inQuotes) {
    64→    64→      cols.push(cur.trim());
    65→    65→      cur = "";
    66→    66→    } else {
    67→    67→      cur += ch;
    68→    68→    }
    69→    69→  }
    70→    70→  cols.push(cur.trim());
    71→    71→  return cols;
    72→    72→}

   327→   327→            codAvailable,

   414→   414→          update: { zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   415→   415→        });
   416→   416→        imported++;
   417→   417→      } catch {
   418→   418→        // skip individual errors
   419→   419→      }
   420→   420→    }
   421→   421→
   422→   422→    return { success: true, action: "range-import", imported, total: rangeSize };
   423→   423→  }
   424→   424→
   425→   425→  return null;
