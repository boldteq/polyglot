    50→/** RFC-4180 compliant CSV row parser — handles quoted fields containing commas. */
    51→function parseCsvRow(line: string): string[] {
    52→  const cols: string[] = [];
    53→  let cur = "";
    54→  let inQuotes = false;
    55→  for (let i = 0; i < line.length; i++) {
    56→    const ch = line[i];
    57→    if (ch === '"') {
    58→      if (inQuotes && line[i + 1] === '"') {
    59→        cur += '"';
    60→        i++;
    61→      } else {
    62→        inQuotes = !inQuotes;
    63→      }
    64→    } else if (ch === "," && !inQuotes) {
    65→      cols.push(cur.trim());
    66→      cur = "";
    67→    } else {
    68→      cur += ch;
    69→    }
    70→  }
    71→  cols.push(cur.trim());
    72→  return cols;
    73→}

   114→export const action = async ({ request }: ActionFunctionArgs) => {
   115→  const { session } = await authenticate.admin(request);
   116→  const shop = session.shop;
   117→  const formData = await request.formData();
   118→  const intent = formData.get("intent");
   119→
   120→  if (intent === "add") {
   121→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   122→    const label = String(formData.get("label") || "").trim();
   123→    const zone = String(formData.get("zone") || "").trim();
   124→    const message = String(formData.get("message") || "").trim();
   125→    const eta = String(formData.get("eta") || "").trim();
   126→    const type = String(formData.get("type") || "allowed");
   127→
   128→    if (!zipCode) return { error: "Zip code is required." };
   129→
   130→    const subscription = await getShopSubscription(shop);
   131→    const limits = PLAN_LIMITS[subscription.planTier];
   132→
   133→    if (type === "blocked" && !limits.allowBlocked) {
   134→      return {
   135→        error:
   136→          "Blocked zip codes are not available on the Free plan. Upgrade to Pro or Ultimate.",
   137→        upgradeRequired: true,
   138→      };
   139→    }
   140→
   141→    const currentCount = await db.zipCode.count({ where: { shop } });
   142→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   143→      return {
   144→        error: `You have reached the ${limits.maxZipCodes} zip code limit for the Free plan. Upgrade to Pro for unlimited zip codes.`,
   145→        upgradeRequired: true,
   146→      };
   147→    }
   148→
   149→    try {
   150→      await db.zipCode.create({
   151→        data: {
   152→          shop,
   153→          zipCode,
   154→          label: label || null,
   155→          zone: zone || null,
   156→          message: message || null,
   157→          eta: eta || null,
   158→          type,
   159→        },
   160→      });
   161→      return { success: true, action: "added", zipCode };
   162→    } catch {
   163→      return { error: `Zip code "${zipCode}" already exists.` };
   164→    }
   165→  }
   166→
   167→  if (intent === "delete") {
   168→    const id = String(formData.get("id"));
   169→    try {
   170→      await db.zipCode.delete({ where: { id } });
   171→      return { success: true, action: "deleted" };
   172→    } catch {
   173→      return { error: "Failed to delete zip code." };
   174→    }
   175→  }
   176→
   177→  if (intent === "update") {
   178→    const id = String(formData.get("id"));
   179→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   180→    const label = String(formData.get("label") || "").trim();
   181→    const zone = String(formData.get("zone") || "").trim();
   182→    const message = String(formData.get("message") || "").trim();
   183→    const eta = String(formData.get("eta") || "").trim();
   184→    const type = String(formData.get("type") || "allowed");
   185→    const isActive = formData.get("isActive") === "true";
   186→
   187→    if (!zipCode) return { error: "Zip code is required." };
   188→
   189→    try {
   190→      await db.zipCode.update({
   191→        where: { id },
   192→        data: {
   193→          zipCode,
   194→          label: label || null,
   195→          zone: zone || null,
   196→          message: message || null,
   197→          eta: eta || null,
   198→          type,
   199→          isActive,
   200→        },
   201→      });
   202→      return { success: true, action: "updated", zipCode };
   203→    } catch {
   204→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   205→    }
   206→  }
   207→
   208→  if (intent === "toggle") {
   209→    const id = String(formData.get("id"));
   210→    const isActive = formData.get("isActive") === "true";
   211→    try {
   212→      await db.zipCode.update({
   213→        where: { id },
   214→        data: { isActive: !isActive },
   215→      });
   216→      return { success: true, action: "toggled" };
   217→    } catch {
   218→      return { error: "Failed to toggle zip code status." };
   219→    }
   220→  }
   221→
   222→  if (intent === "check") {
   223→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   224→    try {
   225→      const found = await db.zipCode.findUnique({
   226→        where: { shop_zipCode: { shop, zipCode } },
   227→      });
   228→      if (!found) return { checkResult: { found: false, zipCode } };
   229→      return { checkResult: { found: true, zipCode, record: found } };
   230→    } catch {
   231→      return { error: "Failed to check zip code." };
   232→    }
   233→  }
   234→
   235→  if (intent === "bulk-import") {
   236→    const csvData = String(formData.get("csvData") || "");
   237→    if (!csvData.trim()) return { error: "No CSV data provided." };
   238→
   239→    const subscription = await getShopSubscription(shop);
   240→    const limits = PLAN_LIMITS[subscription.planTier];
   241→
   242→    const lines = csvData
   243→      .split("\n")
   244→      .map((l) => l.trim())
   245→      .filter((l) => l.length > 0);
   246→
   247→    // Skip header row if present
   248→    const startIdx =
   249→      lines.length > 0 &&
   250→      lines[0].toLowerCase().includes("zip")
   251→        ? 1
   252→        : 0;
   253→
   254→    const currentCount = await db.zipCode.count({ where: { shop } });
   255→    let imported = 0;
   256→    let skipped = 0;
   257→    const errors: string[] = [];
   258→
   259→    for (let i = startIdx; i < lines.length; i++) {
   260→      const cols = parseCsvRow(lines[i]);
   261→      const zipCode = (cols[0] || "").toUpperCase();
   262→      if (!zipCode) continue;
   263→
   264→      const zone = cols[1] || null;
   265→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   266→      const message = cols[3] || null;
   267→      const eta = cols[4] || null;
   268→
   269→      if (type === "blocked" && !limits.allowBlocked) {
   270→        skipped++;
   271→        continue;
   272→      }
   273→
   274→      if (
   275→        limits.maxZipCodes !== Infinity &&
   276→        currentCount + imported >= limits.maxZipCodes
   277→      ) {
   278→        errors.push(
   279→          `Reached the ${limits.maxZipCodes} zip code limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   280→        );
   281→        break;
   282→      }
   283→
   284→      try {
   285→        await db.zipCode.upsert({
   286→          where: { shop_zipCode: { shop, zipCode } },
   287→          create: {
   288→            shop,
   289→            zipCode,
   290→            zone,
   291→            type,
   292→            message,
   293→            eta,
   294→          },
   295→          update: {
   296→            zone: zone ?? undefined,
   297→            type,
   298→            message: message ?? undefined,
   299→            eta: eta ?? undefined,
   300→          },
   301→        });
   302→        imported++;
   303→      } catch {
   304→        skipped++;
   305→      }
   306→    }
   307→
   308→    return {
   309→      success: true,
   310→      action: "bulk-import",
   311→      imported,
   312→      skipped,
   313→      errors,
   314→    };
   315→  }
   316→
   317→  if (intent === "export") {
   318→    try {
   319→      const zipCodes = await db.zipCode.findMany({
   320→        where: { shop },
   321→        orderBy: { createdAt: "desc" },
   322→      });
   323→
   324→      const header = "Zip Code,Zone,Status,Message,ETA,Active";
   325→      const rows = zipCodes.map(
   326→        (z) =>
   327→          `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}"`,
   328→      );
   329→      const csv = [header, ...rows].join("\n");
   330→
   331→      return { success: true, action: "export", csv };
   332→    } catch {
   333→      return { error: "Failed to export zip codes." };
   334→    }
   335→  }
   336→
   337→  if (intent === "range-import") {
   338→    const startZip = String(formData.get("startZip") || "").trim().toUpperCase();
   339→    const endZip = String(formData.get("endZip") || "").trim().toUpperCase();
   340→    const zone = (formData.get("zone") as string | null) || null;
   341→    const type = (formData.get("type") as string) === "blocked" ? "blocked" : "allowed";
   342→    const message = (formData.get("message") as string | null) || null;
   343→    const eta = (formData.get("eta") as string | null) || null;
   344→
   345→    // Validate: must be 5-digit numeric zips
   346→    const ZIP_RE = /^\d{5}$/;
   347→    if (!ZIP_RE.test(startZip) || !ZIP_RE.test(endZip)) {
   348→      return { error: "ZIP range import only supports 5-digit US ZIP codes (e.g., 10001)." };
   349→    }
   350→
   351→    const start = parseInt(startZip, 10);
   352→    const end = parseInt(endZip, 10);
   353→
   354→    if (end < start) {
   355→      return { error: "End ZIP must be greater than or equal to Start ZIP." };
   356→    }
   357→
   358→    const rangeSize = end - start + 1;
   359→    const MAX_RANGE = 500;
   360→    if (rangeSize > MAX_RANGE) {
   361→      return { error: `Range too large. Maximum ${MAX_RANGE} zip codes per import (got ${rangeSize}).` };
   362→    }
   363→
   364→    // Check plan limits
   365→    const subscription = await getShopSubscription(shop);
   366→    const limits = PLAN_LIMITS[subscription.planTier];
   367→
   368→    if (type === "blocked" && !limits.allowBlocked) {
   369→      return { error: "Blocked zip codes require a Pro plan or higher." };
   370→    }
   371→
   372→    const currentCount = await db.zipCode.count({ where: { shop } });
   373→    if (limits.maxZipCodes !== Infinity && currentCount + rangeSize > limits.maxZipCodes) {
   374→      return { error: `This range would exceed your plan limit of ${limits.maxZipCodes} zip codes. You have ${limits.maxZipCodes - currentCount} slots remaining.` };
   375→    }
   376→
   377→    // Generate and upsert all zips in range
   378→    let imported = 0;
   379→    for (let i = start; i <= end; i++) {
   380→      const zipCode = String(i).padStart(5, "0");
   381→      try {
   382→        await db.zipCode.upsert({
   383→          where: { shop_zipCode: { shop, zipCode } },
   384→          create: { shop, zipCode, zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   385→          update: { zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   386→        });
   387→        imported++;
   388→      } catch {
   389→        // skip individual errors
   390→      }
   391→    }
   392→
   393→    return { success: true, action: "range-import", imported, total: rangeSize };
   394→  }
   395→
   396→  return null;
   397→};
