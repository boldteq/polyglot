    48→    48→// Helper: sync widget config to an App Installation metafield so the
    49→    49→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    50→    50→//
    51→    51→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    52→    52→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    53→    53→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    54→    54→async function syncConfigMetafield(
    55→    55→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    56→    56→  configData: Record<string, unknown>,
    57→    57→) {
    58→    58→  try {
    59→    59→    // Step 1: get the App Installation GID
    60→    60→    const installResponse = await admin.graphql(
    61→    61→      `query { currentAppInstallation { id } }`,
    62→    62→    );
    63→    63→    const installJson = (await installResponse.json()) as {
    64→    64→      data: { currentAppInstallation: { id: string } };
    65→    65→    };
    66→    66→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    67→    67→
    68→    68→    // Step 2: write the metafield to the App Installation
    69→    69→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    70→    70→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    71→    71→    const metaResponse = await admin.graphql(
    72→    72→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    73→    73→        metafieldsSet(metafields: $metafields) {
    74→    74→          metafields { id namespace key }
    75→    75→          userErrors { field message }
    76→    76→        }
    77→    77→      }`,
    78→    78→      {
    79→    79→        variables: {
    80→    80→          metafields: [
    81→    81→            {
    82→    82→              ownerId: appInstallationGid,
    83→    83→              namespace: "zip_checker",
    84→    84→              key: "widget_config",
    85→    85→              type: "json",
    86→    86→              value: JSON.stringify(configData),
    87→    87→            },
    88→    88→          ],
    89→    89→        },
    90→    90→      },
    91→    91→    );
    92→    92→    const metaJson = (await metaResponse.json()) as {
    93→    93→      data: {
    94→    94→        metafieldsSet: {
    95→    95→          metafields: { id: string; namespace: string; key: string }[];
    96→    96→          userErrors: { field: string; message: string }[];
    97→    97→        };
    98→    98→      };
    99→    99→    };
   100→   100→
   101→   101→    // userErrors are non-fatal — the API fallback still works
   102→   102→  } catch {
   103→   103→    // Non-fatal — the API fallback still works
   104→   104→  }
   105→   105→}
