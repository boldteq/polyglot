    44→    44→// Helper: sync widget config to an App Installation metafield so the
    45→    45→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    46→    46→//
    47→    47→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    48→    48→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    49→    49→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    50→    50→async function syncConfigMetafield(
    51→    51→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    52→    52→  configData: Record<string, unknown>,
    53→    53→) {
    54→    54→  try {
    55→    55→    // Step 1: get the App Installation GID
    56→    56→    const installResponse = await admin.graphql(
    57→    57→      `query { currentAppInstallation { id } }`,
    58→    58→    );
    59→    59→    const installJson = (await installResponse.json()) as {
    60→    60→      data: { currentAppInstallation: { id: string } };
    61→    61→    };
    62→    62→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    63→    63→
    64→    64→    // Step 2: write the metafield to the App Installation
    65→    65→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    66→    66→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    67→    67→    const metaResponse = await admin.graphql(
    68→    68→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    69→    69→        metafieldsSet(metafields: $metafields) {
    70→    70→          metafields { id namespace key }
    71→    71→          userErrors { field message }
    72→    72→        }
    73→    73→      }`,
    74→    74→      {
    75→    75→        variables: {
    76→    76→          metafields: [
    77→    77→            {
    78→    78→              ownerId: appInstallationGid,
    79→    79→              namespace: "zip_checker",
    80→    80→              key: "widget_config",
    81→    81→              type: "json",
    82→    82→              value: JSON.stringify(configData),
    83→    83→            },
    84→    84→          ],
    85→    85→        },
    86→    86→      },
    87→    87→    );
    88→    88→    const metaJson = (await metaResponse.json()) as {
    89→    89→      data: {
    90→    90→        metafieldsSet: {
    91→    91→          metafields: { id: string; namespace: string; key: string }[];
    92→    92→          userErrors: { field: string; message: string }[];
    93→    93→        };
    94→    94→      };
    95→    95→    };
    96→    96→
    97→    97→    if (metaJson.data.metafieldsSet.userErrors.length > 0) {
    98→    98→      console.error(
    99→    99→        "metafieldsSet userErrors:",
   100→   100→        metaJson.data.metafieldsSet.userErrors,
   101→   101→      );
   102→   102→    }
   103→   103→  } catch (e) {
   104→   104→    // Non-fatal — the API fallback still works
   105→   105→    console.error("Failed to sync widget config metafield:", e);
   106→   106→  }
   107→   107→}
