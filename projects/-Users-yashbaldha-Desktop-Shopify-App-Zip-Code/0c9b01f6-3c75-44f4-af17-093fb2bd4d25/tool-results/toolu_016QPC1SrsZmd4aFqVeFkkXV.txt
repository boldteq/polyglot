    50→    50→/** RFC-4180 compliant CSV row parser — handles quoted fields containing commas. */
    51→    51→function parseCsvRow(line: string): string[] {
    52→    52→  const cols: string[] = [];
    53→    53→  let cur = "";
    54→    54→  let inQuotes = false;
    55→    55→  for (let i = 0; i < line.length; i++) {
    56→    56→    const ch = line[i];
    57→    57→    if (ch === '"') {
    58→    58→      if (inQuotes && line[i + 1] === '"') {
    59→    59→        cur += '"';
    60→    60→        i++;
    61→    61→      } else {
    62→    62→        inQuotes = !inQuotes;
    63→    63→      }
    64→    64→    } else if (ch === "," && !inQuotes) {
    65→    65→      cols.push(cur.trim());
    66→    66→      cur = "";
    67→    67→    } else {
    68→    68→      cur += ch;
    69→    69→    }
    70→    70→  }
    71→    71→  cols.push(cur.trim());
    72→    72→  return cols;
    73→    73→}

   114→   114→export const action = async ({ request }: ActionFunctionArgs) => {
   115→   115→  const { session } = await authenticate.admin(request);
   116→   116→  const shop = session.shop;
   117→   117→  const formData = await request.formData();
   118→   118→  const intent = formData.get("intent");
   119→   119→

   122→   122→    const label = String(formData.get("label") || "").trim();
   123→   123→    const zone = String(formData.get("zone") || "").trim();
   124→   124→    const message = String(formData.get("message") || "").trim();

   126→   126→    const type = String(formData.get("type") || "allowed");
   127→   127→

   130→   130→    const subscription = await getShopSubscription(shop);
   131→   131→    const limits = PLAN_LIMITS[subscription.planTier];

   133→   133→    if (type === "blocked" && !limits.allowBlocked) {
   134→   134→      return {
   135→   135→        error:
   136→   136→          "Blocked zip codes are not available on the Free plan. Upgrade to Pro or Ultimate.",
   137→   137→        upgradeRequired: true,
   138→   138→      };
   139→   139→    }
   140→   140→
   141→   141→    const currentCount = await db.zipCode.count({ where: { shop } });
   142→   142→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   143→   143→      return {
   144→   144→        error: `You have reached the ${limits.maxZipCodes} zip code limit for the Free plan. Upgrade to Pro for unlimited zip codes.`,
   145→   145→        upgradeRequired: true,
   146→   146→      };
   147→   147→    }
   148→   148→
   149→   149→    try {
   150→   150→      await db.zipCode.create({
   151→   151→        data: {
   152→   152→          shop,
   153→   153→          zipCode,
   154→   154→          label: label || null,
   155→   155→          zone: zone || null,
   156→   156→          message: message || null,
   157→   157→          eta: eta || null,
   158→   158→          type,
   159→   159→        },
   160→   160→      });
   161→   161→      return { success: true, action: "added", zipCode };
   162→   162→    } catch {
   163→   163→      return { error: `Zip code "${zipCode}" already exists.` };
   164→   164→    }
   165→   165→  }
   166→   166→
   167→   167→  if (intent === "delete") {
   168→   168→    const id = String(formData.get("id"));
   169→   169→    try {
   170→   170→      await db.zipCode.delete({ where: { id } });
   171→   171→      return { success: true, action: "deleted" };
   172→   172→    } catch {
   173→   173→      return { error: "Failed to delete zip code." };
   174→   174→    }
   175→   175→  }
   176→   176→
   177→   177→  if (intent === "update") {
   178→   178→    const id = String(formData.get("id"));
   179→   179→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   180→   180→    const label = String(formData.get("label") || "").trim();
   181→   181→    const zone = String(formData.get("zone") || "").trim();
   182→   182→    const message = String(formData.get("message") || "").trim();
   183→   183→    const eta = String(formData.get("eta") || "").trim();
   184→   184→    const type = String(formData.get("type") || "allowed");
   185→   185→    const isActive = formData.get("isActive") === "true";
   186→   186→
   187→   187→    if (!zipCode) return { error: "Zip code is required." };
   188→   188→
   189→   189→    try {
   190→   190→      await db.zipCode.update({
   191→   191→        where: { id },
   192→   192→        data: {
   193→   193→          zipCode,
   194→   194→          label: label || null,
   195→   195→          zone: zone || null,
   196→   196→          message: message || null,
   197→   197→          eta: eta || null,
   198→   198→          type,
   199→   199→          isActive,
   200→   200→        },
   201→   201→      });
   202→   202→      return { success: true, action: "updated", zipCode };
   203→   203→    } catch {
   204→   204→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   205→   205→    }
   206→   206→  }
   207→   207→
   208→   208→  if (intent === "toggle") {
   209→   209→    const id = String(formData.get("id"));
   210→   210→    const isActive = formData.get("isActive") === "true";
   211→   211→    try {
   212→   212→      await db.zipCode.update({
   213→   213→        where: { id },
   214→   214→        data: { isActive: !isActive },
   215→   215→      });
   216→   216→      return { success: true, action: "toggled" };
   217→   217→    } catch {
   218→   218→      return { error: "Failed to toggle zip code status." };
   219→   219→    }
   220→   220→  }
   221→   221→
   222→   222→  if (intent === "check") {
   223→   223→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   224→   224→    try {
   225→   225→      const found = await db.zipCode.findUnique({
   226→   226→        where: { shop_zipCode: { shop, zipCode } },
   227→   227→      });
   228→   228→      if (!found) return { checkResult: { found: false, zipCode } };
   229→   229→      return { checkResult: { found: true, zipCode, record: found } };
   230→   230→    } catch {
   231→   231→      return { error: "Failed to check zip code." };
   232→   232→    }
   233→   233→  }
   234→   234→
   235→   235→  if (intent === "bulk-import") {
   236→   236→    const csvData = String(formData.get("csvData") || "");
   237→   237→    if (!csvData.trim()) return { error: "No CSV data provided." };
   238→   238→
   239→   239→    const subscription = await getShopSubscription(shop);
   240→   240→    const limits = PLAN_LIMITS[subscription.planTier];
   241→   241→
   242→   242→    const lines = csvData
   243→   243→      .split("\n")
   244→   244→      .map((l) => l.trim())
   245→   245→      .filter((l) => l.length > 0);
   246→   246→
   247→   247→    // Skip header row if present
   248→   248→    const startIdx =
   249→   249→      lines.length > 0 &&
   250→   250→      lines[0].toLowerCase().includes("zip")
   251→   251→        ? 1
   252→   252→        : 0;
   253→   253→
   254→   254→    const currentCount = await db.zipCode.count({ where: { shop } });
   255→   255→    let imported = 0;
   256→   256→    let skipped = 0;
   257→   257→    const errors: string[] = [];
   258→   258→
   259→   259→    for (let i = startIdx; i < lines.length; i++) {
   260→   260→      const cols = parseCsvRow(lines[i]);
   261→   261→      const zipCode = (cols[0] || "").toUpperCase();
   262→   262→      if (!zipCode) continue;
   263→   263→
   264→   264→      const zone = cols[1] || null;
   265→   265→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   266→   266→      const message = cols[3] || null;
   267→   267→      const eta = cols[4] || null;
   268→   268→
   269→   269→      if (type === "blocked" && !limits.allowBlocked) {
   270→   270→        skipped++;
   271→   271→        continue;
   272→   272→      }
   273→   273→
   274→   274→      if (
   275→   275→        limits.maxZipCodes !== Infinity &&
   276→   276→        currentCount + imported >= limits.maxZipCodes
   277→   277→      ) {
   278→   278→        errors.push(
   279→   279→          `Reached the ${limits.maxZipCodes} zip code limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   280→   280→        );
   281→   281→        break;
   282→   282→      }
   283→   283→
   284→   284→      try {
   285→   285→        await db.zipCode.upsert({
   286→   286→          where: { shop_zipCode: { shop, zipCode } },
   287→   287→          create: {
   288→   288→            shop,
   289→   289→            zipCode,
   290→   290→            zone,
   291→   291→            type,
   292→   292→            message,
   293→   293→            eta,
   294→   294→          },
   295→   295→          update: {
   296→   296→            zone: zone ?? undefined,
   297→   297→            type,
   298→   298→            message: message ?? undefined,
   299→   299→            eta: eta ?? undefined,
   300→   300→          },
   301→   301→        });
   302→   302→        imported++;
   303→   303→      } catch {
   304→   304→        skipped++;
   305→   305→      }
   306→   306→    }
   307→   307→
   308→   308→    return {
   309→   309→      success: true,
   310→   310→      action: "bulk-import",
   311→   311→      imported,
   312→   312→      skipped,
   313→   313→      errors,
   314→   314→    };
   315→   315→  }
   316→   316→
   317→   317→  if (intent === "export") {
   318→   318→    try {
   319→   319→      const zipCodes = await db.zipCode.findMany({
   320→   320→        where: { shop },
   321→   321→        orderBy: { createdAt: "desc" },
   322→   322→      });
   323→   323→
   324→   324→      const header = "Zip Code,Zone,Status,Message,ETA,Active";
   325→   325→      const rows = zipCodes.map(
   326→   326→        (z) =>
   327→   327→          `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}"`,
   328→   328→      );
   329→   329→      const csv = [header, ...rows].join("\n");
   330→   330→
   331→   331→      return { success: true, action: "export", csv };
   332→   332→    } catch {
   333→   333→      return { error: "Failed to export zip codes." };
   334→   334→    }
   335→   335→  }
   336→   336→
   337→   337→  return null;
   338→   338→};
