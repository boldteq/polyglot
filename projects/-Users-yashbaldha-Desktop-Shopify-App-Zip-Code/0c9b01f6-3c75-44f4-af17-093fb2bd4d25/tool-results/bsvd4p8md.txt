    44	// Helper: sync widget config to an App Installation metafield so the
    45	// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    46	//
    47	// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    48	// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    49	// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    50	async function syncConfigMetafield(
    51	  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    52	  configData: Record<string, unknown>,
    53	) {
    54	  try {
    55	    // Step 1: get the App Installation GID
    56	    const installResponse = await admin.graphql(
    57	      `query { currentAppInstallation { id } }`,
    58	    );
    59	    const installJson = (await installResponse.json()) as {
    60	      data: { currentAppInstallation: { id: string } };
    61	    };
    62	    const appInstallationGid = installJson.data.currentAppInstallation.id;
    63	
    64	    // Step 2: write the metafield to the App Installation
    65	    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    66	    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    67	    const metaResponse = await admin.graphql(
    68	      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    69	        metafieldsSet(metafields: $metafields) {
    70	          metafields { id namespace key }
    71	          userErrors { field message }
    72	        }
    73	      }`,
    74	      {
    75	        variables: {
    76	          metafields: [
    77	            {
    78	              ownerId: appInstallationGid,
    79	              namespace: "zip_checker",
    80	              key: "widget_config",
    81	              type: "json",
    82	              value: JSON.stringify(configData),
    83	            },
    84	          ],
    85	        },
    86	      },
    87	    );
    88	    const metaJson = (await metaResponse.json()) as {
    89	      data: {
    90	        metafieldsSet: {
    91	          metafields: { id: string; namespace: string; key: string }[];
    92	          userErrors: { field: string; message: string }[];
    93	        };
    94	      };
    95	    };
    96	
    97	    if (metaJson.data.metafieldsSet.userErrors.length > 0) {
    98	      console.error(
    99	        "metafieldsSet userErrors:",
   100	        metaJson.data.metafieldsSet.userErrors,
   101	      );
   102	    }
   103	  } catch (e) {
   104	    // Non-fatal — the API fallback still works
   105	    console.error("Failed to sync widget config metafield:", e);
   106	  }
   107	}
