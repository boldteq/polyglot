    51→    51→  const shop = session.shop;

    65→    65→    active: zipCodes.filter((z) => z.isActive).length,
    66→    66→  };
    67→    67→

    71→    71→export const action = async ({ request }: ActionFunctionArgs) => {
    72→    72→  const { session } = await authenticate.admin(request);
    73→    73→  const shop = session.shop;
    74→    74→  const formData = await request.formData();
    75→    75→  const intent = formData.get("intent");
    76→    76→
    77→    77→  if (intent === "add") {
    78→    78→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
    79→    79→    const label = String(formData.get("label") || "").trim();
    80→    80→    const zone = String(formData.get("zone") || "").trim();
    81→    81→    const message = String(formData.get("message") || "").trim();
    82→    82→    const eta = String(formData.get("eta") || "").trim();
    83→    83→    const type = String(formData.get("type") || "allowed");
    84→    84→
    85→    85→    if (!zipCode) return { error: "Zip code is required." };
    86→    86→
    87→    87→    const subscription = await getShopSubscription(shop);
    88→    88→    const limits = PLAN_LIMITS[subscription.planTier];
    89→    89→
    90→    90→    if (type === "blocked" && !limits.allowBlocked) {
    91→    91→      return {
    92→    92→        error:
    93→    93→          "Blocked zip codes are not available on the Free plan. Upgrade to Pro or Ultimate.",
    94→    94→        upgradeRequired: true,
    95→    95→      };
    96→    96→    }
    97→    97→
    98→    98→    const currentCount = await db.zipCode.count({ where: { shop } });
    99→    99→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   100→   100→      return {
   101→   101→        error: `You have reached the ${limits.maxZipCodes} zip code limit for the Free plan. Upgrade to Pro for unlimited zip codes.`,
   102→   102→        upgradeRequired: true,
   103→   103→      };
   104→   104→    }
   105→   105→
   106→   106→    try {
   107→   107→      await db.zipCode.create({
   108→   108→        data: {
   109→   109→          shop,
   110→   110→          zipCode,
   111→   111→          label: label || null,
   112→   112→          zone: zone || null,
   113→   113→          message: message || null,
   114→   114→          eta: eta || null,
   115→   115→          type,
   116→   116→        },
   117→   117→      });
   118→   118→      return { success: true, action: "added", zipCode };
   119→   119→    } catch {
   120→   120→      return { error: `Zip code "${zipCode}" already exists.` };
   121→   121→    }
   122→   122→  }
   123→   123→
   124→   124→  if (intent === "delete") {
   125→   125→    const id = String(formData.get("id"));
   126→   126→    try {
   127→   127→      await db.zipCode.delete({ where: { id } });
   128→   128→      return { success: true, action: "deleted" };
   129→   129→    } catch {
   130→   130→      return { error: "Failed to delete zip code." };
   131→   131→    }
   132→   132→  }
   133→   133→
   134→   134→  if (intent === "update") {
   135→   135→    const id = String(formData.get("id"));
   136→   136→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   137→   137→    const label = String(formData.get("label") || "").trim();
   138→   138→    const zone = String(formData.get("zone") || "").trim();
   139→   139→    const message = String(formData.get("message") || "").trim();
   140→   140→    const eta = String(formData.get("eta") || "").trim();
   141→   141→    const type = String(formData.get("type") || "allowed");
   142→   142→    const isActive = formData.get("isActive") === "true";
   143→   143→
   144→   144→    if (!zipCode) return { error: "Zip code is required." };
   145→   145→
   146→   146→    try {
   147→   147→      await db.zipCode.update({
   148→   148→        where: { id },
   149→   149→        data: {
   150→   150→          zipCode,
   151→   151→          label: label || null,
   152→   152→          zone: zone || null,
   153→   153→          message: message || null,
   154→   154→          eta: eta || null,
   155→   155→          type,
   156→   156→          isActive,
   157→   157→        },
   158→   158→      });
   159→   159→      return { success: true, action: "updated", zipCode };
   160→   160→    } catch {
   161→   161→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   162→   162→    }
   163→   163→  }
   164→   164→
   165→   165→  if (intent === "toggle") {
   166→   166→    const id = String(formData.get("id"));
   167→   167→    const isActive = formData.get("isActive") === "true";
   168→   168→    try {
   169→   169→      await db.zipCode.update({
   170→   170→        where: { id },
   171→   171→        data: { isActive: !isActive },
   172→   172→      });
   173→   173→      return { success: true, action: "toggled" };
   174→   174→    } catch {
   175→   175→      return { error: "Failed to toggle zip code status." };
   176→   176→    }
   177→   177→  }
   178→   178→
   179→   179→  if (intent === "check") {
   180→   180→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   181→   181→    try {
   182→   182→      const found = await db.zipCode.findUnique({
   183→   183→        where: { shop_zipCode: { shop, zipCode } },
   184→   184→      });
   185→   185→      if (!found) return { checkResult: { found: false, zipCode } };
   186→   186→      return { checkResult: { found: true, zipCode, record: found } };
   187→   187→    } catch {
   188→   188→      return { error: "Failed to check zip code." };
   189→   189→    }
   190→   190→  }
   191→   191→
   192→   192→  if (intent === "bulk-import") {
   193→   193→    const csvData = String(formData.get("csvData") || "");
   194→   194→    if (!csvData.trim()) return { error: "No CSV data provided." };
   195→   195→
   196→   196→    const subscription = await getShopSubscription(shop);
   197→   197→    const limits = PLAN_LIMITS[subscription.planTier];
   198→   198→
   199→   199→    const lines = csvData
   200→   200→      .split("\n")
   201→   201→      .map((l) => l.trim())
   202→   202→      .filter((l) => l.length > 0);
   203→   203→
   204→   204→    // Skip header row if present
   205→   205→    const startIdx =
   206→   206→      lines.length > 0 &&
   207→   207→      lines[0].toLowerCase().includes("zip")
   208→   208→        ? 1
   209→   209→        : 0;
   210→   210→
   211→   211→    const currentCount = await db.zipCode.count({ where: { shop } });
   212→   212→    let imported = 0;
   213→   213→    let skipped = 0;
   214→   214→    const errors: string[] = [];
   215→   215→
   216→   216→    for (let i = startIdx; i < lines.length; i++) {
   217→   217→      const cols = lines[i].split(",").map((c) => c.trim().replace(/^"|"$/g, ""));
   218→   218→      const zipCode = (cols[0] || "").toUpperCase();
   219→   219→      if (!zipCode) continue;
   220→   220→
   221→   221→      const zone = cols[1] || null;
   222→   222→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   223→   223→      const message = cols[3] || null;
   224→   224→      const eta = cols[4] || null;
   225→   225→
   226→   226→      if (type === "blocked" && !limits.allowBlocked) {
   227→   227→        skipped++;
   228→   228→        continue;
   229→   229→      }
   230→   230→
   231→   231→      if (
   232→   232→        limits.maxZipCodes !== Infinity &&
   233→   233→        currentCount + imported >= limits.maxZipCodes
   234→   234→      ) {
   235→   235→        errors.push(
   236→   236→          `Reached the ${limits.maxZipCodes} zip code limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   237→   237→        );
   238→   238→        break;
   239→   239→      }
   240→   240→
   241→   241→      try {
   242→   242→        await db.zipCode.upsert({
   243→   243→          where: { shop_zipCode: { shop, zipCode } },
   244→   244→          create: {
   245→   245→            shop,
   246→   246→            zipCode,
   247→   247→            zone,
   248→   248→            type,
   249→   249→            message,
   250→   250→            eta,
   251→   251→          },
   252→   252→          update: {
   253→   253→            zone: zone ?? undefined,
   254→   254→            type,
   255→   255→            message: message ?? undefined,
   256→   256→            eta: eta ?? undefined,
   257→   257→          },
   258→   258→        });
   259→   259→        imported++;
   260→   260→      } catch {
   261→   261→        skipped++;
   262→   262→      }
   263→   263→    }
   264→   264→
   265→   265→    return {
   266→   266→      success: true,
   267→   267→      action: "bulk-import",
   268→   268→      imported,
   269→   269→      skipped,
   270→   270→      errors,
   271→   271→    };
   272→   272→  }
   273→   273→
   274→   274→  if (intent === "export") {
   275→   275→    try {
   276→   276→      const zipCodes = await db.zipCode.findMany({
   277→   277→        where: { shop },
   278→   278→        orderBy: { createdAt: "desc" },
   279→   279→      });
   280→   280→
   281→   281→      const header = "Zip Code,Zone,Status,Message,ETA,Active";
   282→   282→      const rows = zipCodes.map(
   283→   283→        (z) =>
   284→   284→          `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}"`,
   285→   285→      );
   286→   286→      const csv = [header, ...rows].join("\n");
   287→   287→
   288→   288→      return { success: true, action: "export", csv };
   289→   289→    } catch {
   290→   290→      return { error: "Failed to export zip codes." };
   291→   291→    }
   292→   292→  }
   293→   293→
   294→   294→  return null;
   295→   295→};

   303→   303→  eta: string | null;
   304→   304→  type: string;
   305→   305→  isActive: boolean;
