    43→    43→// Helper: sync widget config to an App Installation metafield so the
    44→    44→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    45→    45→//
    46→    46→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    47→    47→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    48→    48→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    49→    49→async function syncConfigMetafield(
    50→    50→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    51→    51→  configData: Record<string, unknown>,
    52→    52→) {
    53→    53→  try {
    54→    54→    // Step 1: get the App Installation GID
    55→    55→    const installResponse = await admin.graphql(
    56→    56→      `query { currentAppInstallation { id } }`,
    57→    57→    );
    58→    58→    const installJson = (await installResponse.json()) as {
    59→    59→      data: { currentAppInstallation: { id: string } };
    60→    60→    };
    61→    61→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    62→    62→
    63→    63→    // Step 2: write the metafield to the App Installation
    64→    64→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    65→    65→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    66→    66→    const metaResponse = await admin.graphql(
    67→    67→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    68→    68→        metafieldsSet(metafields: $metafields) {
    69→    69→          metafields { id namespace key }
    70→    70→          userErrors { field message }
    71→    71→        }
    72→    72→      }`,
    73→    73→      {
    74→    74→        variables: {
    75→    75→          metafields: [
    76→    76→            {
    77→    77→              ownerId: appInstallationGid,
    78→    78→              namespace: "zip_checker",
    79→    79→              key: "widget_config",
    80→    80→              type: "json",
    81→    81→              value: JSON.stringify(configData),
    82→    82→            },
    83→    83→          ],
    84→    84→        },
    85→    85→      },
    86→    86→    );
    87→    87→    const metaJson = (await metaResponse.json()) as {
    88→    88→      data: {
    89→    89→        metafieldsSet: {
    90→    90→          metafields: { id: string; namespace: string; key: string }[];
    91→    91→          userErrors: { field: string; message: string }[];
    92→    92→        };
    93→    93→      };
    94→    94→    };
    95→    95→
    96→    96→    if (metaJson.data.metafieldsSet.userErrors.length > 0) {
    97→    97→      console.error(
    98→    98→        "metafieldsSet userErrors:",
    99→    99→        metaJson.data.metafieldsSet.userErrors,
   100→   100→      );
   101→   101→    }
   102→   102→  } catch (e) {
   103→   103→    // Non-fatal — the API fallback still works
   104→   104→    console.error("Failed to sync widget config metafield:", e);
   105→   105→  }
   106→   106→}
