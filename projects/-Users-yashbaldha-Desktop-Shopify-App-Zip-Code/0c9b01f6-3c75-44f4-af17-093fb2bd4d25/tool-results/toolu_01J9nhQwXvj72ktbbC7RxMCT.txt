    50→    50→// Helper: sync widget config to an App Installation metafield so the
    51→    51→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    52→    52→//
    53→    53→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    54→    54→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    55→    55→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    56→    56→async function syncConfigMetafield(
    57→    57→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    58→    58→  configData: Record<string, unknown>,
    59→    59→) {
    60→    60→  try {
    61→    61→    // Step 1: get the App Installation GID
    62→    62→    const installResponse = await admin.graphql(
    63→    63→      `query { currentAppInstallation { id } }`,
    64→    64→    );
    65→    65→    const installJson = (await installResponse.json()) as {
    66→    66→      data: { currentAppInstallation: { id: string } };
    67→    67→    };
    68→    68→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    69→    69→
    70→    70→    // Step 2: write the metafield to the App Installation
    71→    71→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    72→    72→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    73→    73→    const metaResponse = await admin.graphql(
    74→    74→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    75→    75→        metafieldsSet(metafields: $metafields) {
    76→    76→          metafields { id namespace key }
    77→    77→          userErrors { field message }
    78→    78→        }
    79→    79→      }`,
    80→    80→      {
    81→    81→        variables: {
    82→    82→          metafields: [
    83→    83→            {
    84→    84→              ownerId: appInstallationGid,
    85→    85→              namespace: "zip_checker",
    86→    86→              key: "widget_config",
    87→    87→              type: "json",
    88→    88→              value: JSON.stringify(configData),
    89→    89→            },
    90→    90→          ],
    91→    91→        },
    92→    92→      },
    93→    93→    );
    94→    94→    const metaJson = (await metaResponse.json()) as {
    95→    95→      data: {
    96→    96→        metafieldsSet: {
    97→    97→          metafields: { id: string; namespace: string; key: string }[];
    98→    98→          userErrors: { field: string; message: string }[];
    99→    99→        };
   100→   100→      };
   101→   101→    };
   102→   102→
   103→   103→    // userErrors are non-fatal — the API fallback still works
   104→   104→  } catch {
   105→   105→    // Non-fatal — the API fallback still works
   106→   106→  }
   107→   107→}

   262→   262→// ── Scope custom CSS for admin preview — prefix selectors with #wid ─────────
   263→   263→function scopeAdminCss(rawCss: string | null | undefined, wid: string): string {
   264→   264→  if (!rawCss) return "";
   265→   265→  const css = rawCss.replace(/<\/style>/gi, "");
   266→   266→  let result = "";
   267→   267→  let remaining = css;
   268→   268→  while (remaining.length > 0) {
   269→   269→    const openBrace = remaining.indexOf("{");
   270→   270→    if (openBrace === -1) break;
   271→   271→    const closeBrace = remaining.indexOf("}", openBrace);
   272→   272→    if (closeBrace === -1) break;
   273→   273→    const selector = remaining.substring(0, openBrace).trim();
   274→   274→    const body = remaining.substring(openBrace + 1, closeBrace);
   275→   275→    remaining = remaining.substring(closeBrace + 1);
   276→   276→    if (selector.startsWith("@")) {
   277→   277→      result += selector + "{" + body + "}";
   278→   278→    } else if (selector) {
   279→   279→      const prefixed = selector.split(",").map(s => {
   280→   280→        s = s.trim();
   281→   281→        return s ? "#" + wid + " " + s : "";
   282→   282→      }).filter(Boolean).join(", ");
   283→   283→      result += prefixed + " {" + body + "}";
   284→   284→    }
   285→   285→  }
   286→   286→  return result;
   287→   287→}
