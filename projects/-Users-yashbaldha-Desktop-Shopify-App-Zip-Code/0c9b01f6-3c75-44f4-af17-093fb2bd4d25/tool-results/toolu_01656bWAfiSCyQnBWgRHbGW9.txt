    50→// Helper: sync widget config to an App Installation metafield so the
    51→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    52→//
    53→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    54→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    55→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    56→async function syncConfigMetafield(
    57→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    58→  configData: Record<string, unknown>,
    59→) {
    60→  try {
    61→    // Step 1: get the App Installation GID
    62→    const installResponse = await admin.graphql(
    63→      `query { currentAppInstallation { id } }`,
    64→    );
    65→    const installJson = (await installResponse.json()) as {
    66→      data: { currentAppInstallation: { id: string } };
    67→    };
    68→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    69→
    70→    // Step 2: write the metafield to the App Installation
    71→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    72→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    73→    const metaResponse = await admin.graphql(
    74→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    75→        metafieldsSet(metafields: $metafields) {
    76→          metafields { id namespace key }
    77→          userErrors { field message }
    78→        }
    79→      }`,
    80→      {
    81→        variables: {
    82→          metafields: [
    83→            {
    84→              ownerId: appInstallationGid,
    85→              namespace: "zip_checker",
    86→              key: "widget_config",
    87→              type: "json",
    88→              value: JSON.stringify(configData),
    89→            },
    90→          ],
    91→        },
    92→      },
    93→    );
    94→    const metaJson = (await metaResponse.json()) as {
    95→      data: {
    96→        metafieldsSet: {
    97→          metafields: { id: string; namespace: string; key: string }[];
    98→          userErrors: { field: string; message: string }[];
    99→        };
   100→      };
   101→    };
   102→
   103→    // userErrors are non-fatal — the API fallback still works
   104→  } catch {
   105→    // Non-fatal — the API fallback still works
   106→  }
   107→}

   264→// ── Scope custom CSS for admin preview — prefix selectors with #wid ─────────
   265→function scopeAdminCss(rawCss: string | null | undefined, wid: string): string {
   266→  if (!rawCss) return "";
   267→  const css = rawCss.replace(/<\/style>/gi, "");
   268→  let result = "";
   269→  let remaining = css;
   270→  while (remaining.length > 0) {
   271→    const openBrace = remaining.indexOf("{");
   272→    if (openBrace === -1) break;
   273→    const closeBrace = remaining.indexOf("}", openBrace);
   274→    if (closeBrace === -1) break;
   275→    const selector = remaining.substring(0, openBrace).trim();
   276→    const body = remaining.substring(openBrace + 1, closeBrace);
   277→    remaining = remaining.substring(closeBrace + 1);
   278→    if (selector.startsWith("@")) {
   279→      result += selector + "{" + body + "}";
   280→    } else if (selector) {
   281→      const prefixed = selector.split(",").map(s => {
   282→        s = s.trim();
   283→        return s ? "#" + wid + " " + s : "";
   284→      }).filter(Boolean).join(", ");
   285→      result += prefixed + " {" + body + "}";
   286→    }
   287→  }
   288→  return result;
   289→}
