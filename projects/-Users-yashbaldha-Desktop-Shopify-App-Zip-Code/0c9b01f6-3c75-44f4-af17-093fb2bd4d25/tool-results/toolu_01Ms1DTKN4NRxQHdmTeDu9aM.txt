    49→// Helper: sync widget config to an App Installation metafield so the
    50→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    51→//
    52→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    53→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    54→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    55→async function syncConfigMetafield(
    56→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    57→  configData: Record<string, unknown>,
    58→) {
    59→  try {
    60→    // Step 1: get the App Installation GID
    61→    const installResponse = await admin.graphql(
    62→      `query { currentAppInstallation { id } }`,
    63→    );
    64→    const installJson = (await installResponse.json()) as {
    65→      data: { currentAppInstallation: { id: string } };
    66→    };
    67→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    68→
    69→    // Step 2: write the metafield to the App Installation
    70→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    71→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    72→    const metaResponse = await admin.graphql(
    73→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    74→        metafieldsSet(metafields: $metafields) {
    75→          metafields { id namespace key }
    76→          userErrors { field message }
    77→        }
    78→      }`,
    79→      {
    80→        variables: {
    81→          metafields: [
    82→            {
    83→              ownerId: appInstallationGid,
    84→              namespace: "zip_checker",
    85→              key: "widget_config",
    86→              type: "json",
    87→              value: JSON.stringify(configData),
    88→            },
    89→          ],
    90→        },
    91→      },
    92→    );
    93→    const metaJson = (await metaResponse.json()) as {
    94→      data: {
    95→        metafieldsSet: {
    96→          metafields: { id: string; namespace: string; key: string }[];
    97→          userErrors: { field: string; message: string }[];
    98→        };
    99→      };
   100→    };
   101→
   102→    // userErrors are non-fatal — the API fallback still works
   103→  } catch {
   104→    // Non-fatal — the API fallback still works
   105→  }
   106→}

   263→// ── Scope custom CSS for admin preview — prefix selectors with #wid ─────────
   264→function scopeAdminCss(rawCss: string | null | undefined, wid: string): string {
   265→  if (!rawCss) return "";
   266→  const css = rawCss.replace(/<\/style>/gi, "");
   267→  let result = "";
   268→  let remaining = css;
   269→  while (remaining.length > 0) {
   270→    const openBrace = remaining.indexOf("{");
   271→    if (openBrace === -1) break;
   272→    const closeBrace = remaining.indexOf("}", openBrace);
   273→    if (closeBrace === -1) break;
   274→    const selector = remaining.substring(0, openBrace).trim();
   275→    const body = remaining.substring(openBrace + 1, closeBrace);
   276→    remaining = remaining.substring(closeBrace + 1);
   277→    if (selector.startsWith("@")) {
   278→      result += selector + "{" + body + "}";
   279→    } else if (selector) {
   280→      const prefixed = selector.split(",").map(s => {
   281→        s = s.trim();
   282→        return s ? "#" + wid + " " + s : "";
   283→      }).filter(Boolean).join(", ");
   284→      result += prefixed + " {" + body + "}";
   285→    }
   286→  }
   287→  return result;
   288→}
