   260→  // ── Scope custom CSS: auto-prefix selectors with widget container ID ──────
   261→  function scopeCustomCss(rawCss, wid) {
   262→    if (!rawCss) return "";
   263→    // Strip </style> tags to prevent injection
   264→    var css = rawCss.replace(/<\/style>/gi, "");
   265→    var result = "";
   266→    var i = 0;
   267→    while (i < css.length) {
   268→      // Skip whitespace
   269→      while (i < css.length && /\s/.test(css[i])) { result += css[i]; i++; }
   270→      if (i >= css.length) break;
   271→      // Pass through @keyframes and @font-face unchanged
   272→      if (css[i] === '@') {
   273→        var atRule = "";
   274→        var j = i;
   275→        while (j < css.length && css[j] !== '{' && css[j] !== ';') { atRule += css[j]; j++; }
   276→        var atName = atRule.trim().toLowerCase();
   277→        if (atName.indexOf("@keyframes") === 0 || atName.indexOf("@font-face") === 0) {
   278→          result += atRule;
   279→          if (j < css.length && css[j] === '{') {
   280→            var depth = 1; result += css[j]; j++;
   281→            while (j < css.length && depth > 0) {
   282→              if (css[j] === '{') depth++;
   283→              if (css[j] === '}') depth--;
   284→              result += css[j]; j++;
   285→            }
   286→          }
   287→          i = j; continue;
   288→        }
   289→        // @media: scope inner rules
   290→        if (atName.indexOf("@media") === 0) {
   291→          result += atRule;
   292→          if (j < css.length && css[j] === '{') {
   293→            result += '{'; j++;
   294→            // Collect inner content
   295→            var depth = 1; var inner = "";
   296→            while (j < css.length && depth > 0) {
   297→              if (css[j] === '{') depth++;
   298→              if (css[j] === '}') depth--;
   299→              if (depth > 0) inner += css[j];
   300→              j++;
   301→            }
   302→            result += scopeCustomCss(inner, wid);
   303→            result += '}';
   304→          }
   305→          i = j; continue;
   306→        }
   307→      }
   308→      // Regular rule: grab selector, prefix with #wid
   309→      var sel = ""; var body = "";
   310→      while (i < css.length && css[i] !== '{') { sel += css[i]; i++; }
   311→      if (i < css.length) { i++; } // skip {
   312→      var depth = 1;
   313→      while (i < css.length && depth > 0) {
   314→        if (css[i] === '{') depth++;
   315→        if (css[i] === '}') depth--;
   316→        if (depth > 0) body += css[i];
   317→        i++;
   318→      }
   319→      if (sel.trim()) {
   320→        // Split multiple selectors by comma and prefix each
   321→        var selectors = sel.split(',').map(function(s) {
   322→          s = s.trim();
   323→          if (!s) return '';
   324→          return '#' + wid + ' ' + s;
   325→        }).filter(Boolean).join(', ');
   326→        result += selectors + ' {' + body + '}';
   327→      }
   328→    }
   329→    return result;
   330→  }

   483→    var res = document.getElementById(wid + "-res");

   486→    if (!inp || !btn || !res) return;
   487→
   488→    function normalizeZip(raw) {
   489→      var z = raw.trim().toUpperCase();
   490→      z = z.replace(/\s+/g, '');        // collapse all spaces (UK: SW1A 2AA → SW1A2AA)
   491→      z = z.replace(/-\d{4}$/, '');     // strip US ZIP+4 suffix (90210-1234 → 90210)
   492→      return z;
   493→    }
   494→
   495→    function checkZip() {
   496→      var zip = normalizeZip(inp.value);

   498→      if (!zip) { inp.focus(); return; }
   499→
   500→      btn.disabled = true;
   501→      btn.textContent = "\u2026";
   502→      res.innerHTML = "";
   503→
   504→      fetch(
   505→        API_BASE + "/api/zip-check?shop=" + encodeURIComponent(SHOP) +
   506→          "&zip=" + encodeURIComponent(zip)
