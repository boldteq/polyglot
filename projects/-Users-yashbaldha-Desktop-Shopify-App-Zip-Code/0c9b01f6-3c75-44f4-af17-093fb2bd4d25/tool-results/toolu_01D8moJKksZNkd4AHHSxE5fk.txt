    49→    49→// Helper: sync widget config to an App Installation metafield so the
    50→    50→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
    51→    51→//
    52→    52→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
    53→    53→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
    54→    54→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
    55→    55→async function syncConfigMetafield(
    56→    56→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
    57→    57→  configData: Record<string, unknown>,
    58→    58→) {
    59→    59→  try {
    60→    60→    // Step 1: get the App Installation GID
    61→    61→    const installResponse = await admin.graphql(
    62→    62→      `query { currentAppInstallation { id } }`,
    63→    63→    );
    64→    64→    const installJson = (await installResponse.json()) as {
    65→    65→      data: { currentAppInstallation: { id: string } };
    66→    66→    };
    67→    67→    const appInstallationGid = installJson.data.currentAppInstallation.id;
    68→    68→
    69→    69→    // Step 2: write the metafield to the App Installation
    70→    70→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
    71→    71→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
    72→    72→    const metaResponse = await admin.graphql(
    73→    73→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
    74→    74→        metafieldsSet(metafields: $metafields) {
    75→    75→          metafields { id namespace key }
    76→    76→          userErrors { field message }
    77→    77→        }
    78→    78→      }`,
    79→    79→      {
    80→    80→        variables: {
    81→    81→          metafields: [
    82→    82→            {
    83→    83→              ownerId: appInstallationGid,
    84→    84→              namespace: "zip_checker",
    85→    85→              key: "widget_config",
    86→    86→              type: "json",
    87→    87→              value: JSON.stringify(configData),
    88→    88→            },
    89→    89→          ],
    90→    90→        },
    91→    91→      },
    92→    92→    );
    93→    93→    const metaJson = (await metaResponse.json()) as {
    94→    94→      data: {
    95→    95→        metafieldsSet: {
    96→    96→          metafields: { id: string; namespace: string; key: string }[];
    97→    97→          userErrors: { field: string; message: string }[];
    98→    98→        };
    99→    99→      };
   100→   100→    };
   101→   101→
   102→   102→    // userErrors are non-fatal — the API fallback still works
   103→   103→  } catch {
   104→   104→    // Non-fatal — the API fallback still works
   105→   105→  }
   106→   106→}
