    71→    71→export const action = async ({ request }: ActionFunctionArgs) => {
    72→    72→  const { session } = await authenticate.admin(request);
    73→    73→  const shop = session.shop;
    74→    74→  const formData = await request.formData();
    75→    75→  const intent = formData.get("intent");
    76→    76→
    77→    77→  if (intent === "add") {
    78→    78→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
    79→    79→    const label = String(formData.get("label") || "").trim();
    80→    80→    const zone = String(formData.get("zone") || "").trim();
    81→    81→    const message = String(formData.get("message") || "").trim();
    82→    82→    const eta = String(formData.get("eta") || "").trim();
    83→    83→    const type = String(formData.get("type") || "allowed");
    84→    84→
    85→    85→    if (!zipCode) return { error: "Zip code is required." };
    86→    86→
    87→    87→    const subscription = await getShopSubscription(shop);
    88→    88→    const limits = PLAN_LIMITS[subscription.planTier];
    89→    89→
    90→    90→    if (type === "blocked" && !limits.allowBlocked) {
    91→    91→      return {
    92→    92→        error:
    93→    93→          "Blocked zip codes are not available on the Free plan. Upgrade to Pro or Ultimate.",
    94→    94→        upgradeRequired: true,
    95→    95→      };
    96→    96→    }
    97→    97→
    98→    98→    const currentCount = await db.zipCode.count({ where: { shop } });
    99→    99→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   100→   100→      return {
   101→   101→        error: `You have reached the ${limits.maxZipCodes} zip code limit for the Free plan. Upgrade to Pro for unlimited zip codes.`,
   102→   102→        upgradeRequired: true,
   103→   103→      };
   104→   104→    }
   105→   105→
   106→   106→    try {
   107→   107→      await db.zipCode.create({
   108→   108→        data: {
   109→   109→          shop,
   110→   110→          zipCode,
   111→   111→          label: label || null,
   112→   112→          zone: zone || null,
   113→   113→          message: message || null,
   114→   114→          eta: eta || null,
   115→   115→          type,
   116→   116→        },
   117→   117→      });
   118→   118→      return { success: true, action: "added", zipCode };
   119→   119→    } catch {
   120→   120→      return { error: `Zip code "${zipCode}" already exists.` };
   121→   121→    }
   122→   122→  }
   123→   123→
   124→   124→  if (intent === "delete") {
   125→   125→    const id = String(formData.get("id"));
   126→   126→    await db.zipCode.delete({ where: { id } });
   127→   127→    return { success: true, action: "deleted" };
   128→   128→  }
   129→   129→
   130→   130→  if (intent === "update") {
   131→   131→    const id = String(formData.get("id"));
   132→   132→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   133→   133→    const label = String(formData.get("label") || "").trim();
   134→   134→    const zone = String(formData.get("zone") || "").trim();
   135→   135→    const message = String(formData.get("message") || "").trim();
   136→   136→    const eta = String(formData.get("eta") || "").trim();
   137→   137→    const type = String(formData.get("type") || "allowed");
   138→   138→    const isActive = formData.get("isActive") === "true";
   139→   139→
   140→   140→    if (!zipCode) return { error: "Zip code is required." };
   141→   141→
   142→   142→    try {
   143→   143→      await db.zipCode.update({
   144→   144→        where: { id },
   145→   145→        data: {
   146→   146→          zipCode,
   147→   147→          label: label || null,
   148→   148→          zone: zone || null,
   149→   149→          message: message || null,
   150→   150→          eta: eta || null,
   151→   151→          type,
   152→   152→          isActive,
   153→   153→        },
   154→   154→      });
   155→   155→      return { success: true, action: "updated", zipCode };
   156→   156→    } catch {
   157→   157→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   158→   158→    }
   159→   159→  }
   160→   160→
   161→   161→  if (intent === "toggle") {
   162→   162→    const id = String(formData.get("id"));
   163→   163→    const isActive = formData.get("isActive") === "true";
   164→   164→    await db.zipCode.update({
   165→   165→      where: { id },
   166→   166→      data: { isActive: !isActive },
   167→   167→    });
   168→   168→    return { success: true, action: "toggled" };
   169→   169→  }
   170→   170→
   171→   171→  if (intent === "check") {
   172→   172→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   173→   173→    const found = await db.zipCode.findUnique({
   174→   174→      where: { shop_zipCode: { shop, zipCode } },
   175→   175→    });
   176→   176→    if (!found) return { checkResult: { found: false, zipCode } };
   177→   177→    return { checkResult: { found: true, zipCode, record: found } };
   178→   178→  }
   179→   179→
   180→   180→  if (intent === "bulk-import") {
   181→   181→    const csvData = String(formData.get("csvData") || "");
   182→   182→    if (!csvData.trim()) return { error: "No CSV data provided." };
   183→   183→
   184→   184→    const subscription = await getShopSubscription(shop);
   185→   185→    const limits = PLAN_LIMITS[subscription.planTier];
   186→   186→
   187→   187→    const lines = csvData
   188→   188→      .split("\n")
   189→   189→      .map((l) => l.trim())
   190→   190→      .filter((l) => l.length > 0);
   191→   191→
   192→   192→    // Skip header row if present
   193→   193→    const startIdx =
   194→   194→      lines.length > 0 &&
   195→   195→      lines[0].toLowerCase().includes("zip")
   196→   196→        ? 1
   197→   197→        : 0;
   198→   198→
   199→   199→    const currentCount = await db.zipCode.count({ where: { shop } });
   200→   200→    let imported = 0;
   201→   201→    let skipped = 0;
   202→   202→    const errors: string[] = [];
   203→   203→
   204→   204→    for (let i = startIdx; i < lines.length; i++) {
   205→   205→      const cols = lines[i].split(",").map((c) => c.trim().replace(/^"|"$/g, ""));
   206→   206→      const zipCode = (cols[0] || "").toUpperCase();
   207→   207→      if (!zipCode) continue;
   208→   208→
   209→   209→      const zone = cols[1] || null;
   210→   210→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   211→   211→      const message = cols[3] || null;
   212→   212→      const eta = cols[4] || null;
   213→   213→
   214→   214→      if (type === "blocked" && !limits.allowBlocked) {
   215→   215→        skipped++;
   216→   216→        continue;
   217→   217→      }
   218→   218→
   219→   219→      if (
   220→   220→        limits.maxZipCodes !== Infinity &&
   221→   221→        currentCount + imported >= limits.maxZipCodes
   222→   222→      ) {
   223→   223→        errors.push(
   224→   224→          `Reached the ${limits.maxZipCodes} zip code limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   225→   225→        );
   226→   226→        break;
   227→   227→      }
   228→   228→
   229→   229→      try {
   230→   230→        await db.zipCode.upsert({
   231→   231→          where: { shop_zipCode: { shop, zipCode } },
   232→   232→          create: {
   233→   233→            shop,
   234→   234→            zipCode,
   235→   235→            zone,
   236→   236→            type,
   237→   237→            message,
   238→   238→            eta,
   239→   239→          },
   240→   240→          update: {
   241→   241→            zone: zone ?? undefined,
   242→   242→            type,
   243→   243→            message: message ?? undefined,
   244→   244→            eta: eta ?? undefined,
   245→   245→          },
   246→   246→        });
   247→   247→        imported++;
   248→   248→      } catch {
   249→   249→        skipped++;
   250→   250→      }
   251→   251→    }
   252→   252→
   253→   253→    return {
   254→   254→      success: true,
   255→   255→      action: "bulk-import",
   256→   256→      imported,
   257→   257→      skipped,
   258→   258→      errors,
   259→   259→    };
   260→   260→  }
   261→   261→
   262→   262→  if (intent === "export") {
   263→   263→    const zipCodes = await db.zipCode.findMany({
   264→   264→      where: { shop },
   265→   265→      orderBy: { createdAt: "desc" },
   266→   266→    });
   267→   267→
   268→   268→    const header = "Zip Code,Zone,Status,Message,ETA,Active";
   269→   269→    const rows = zipCodes.map(
   270→   270→      (z) =>
   271→   271→        `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}"`,
   272→   272→    );
   273→   273→    const csv = [header, ...rows].join("\n");
   274→   274→
   275→   275→    return { success: true, action: "export", csv };
   276→   276→  }
   277→   277→
   278→   278→  return null;
   279→   279→};
