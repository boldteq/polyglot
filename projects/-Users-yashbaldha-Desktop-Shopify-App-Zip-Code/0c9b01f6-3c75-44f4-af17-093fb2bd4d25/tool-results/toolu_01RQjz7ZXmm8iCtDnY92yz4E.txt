    31→export const loader = async ({ request }: LoaderFunctionArgs) => {
    32→  const { session, admin } = await authenticate.admin(request);
    33→  const shop = session.shop;
    34→
    35→  let config = await db.widgetConfig.findUnique({ where: { shop } });
    36→
    37→  if (!config) {
    38→    config = await db.widgetConfig.create({ data: { shop } });
    39→  }
    40→
    41→  // Fetch the active (MAIN) theme and detect whether our App Embed block
    42→  // is actually enabled in it by reading config/settings_data.json.
    43→  let activeThemeName: string | null = null;
    44→  let themeEditorUrl = `https://${shop}/admin/themes/current/editor`;
    45→  let themeEditorAppEmbedsUrl = `https://${shop}/admin/themes/current/editor?context=apps`;
    46→  let appEmbedEnabled = false;
    47→  let embedDebug: {
    48→    httpStatus: number | null;
    49→    hasContent: boolean;
    50→    blockTypes: string[];
    51→    error: string | null;
    52→    rawBlocksSnippet: string | null;
    53→  } = {
    54→    httpStatus: null,
    55→    hasContent: false,
    56→    blockTypes: [],
    57→    error: null,
    58→    rawBlocksSnippet: null,
    59→  };
    60→
    61→  try {
    62→    // Step 1 — get the published theme via REST API.
    63→    // GraphQL `themes(roles:[MAIN])` is unreliable in 2026-01 and returns
    64→    // empty nodes. REST /themes.json?role=main is the canonical approach.
    65→    const themesRes = await fetch(
    66→      `https://${shop}/admin/api/2026-01/themes.json?role=main`,
    67→      { headers: { "X-Shopify-Access-Token": session.accessToken ?? "" } },
    68→    );
    69→
    70→    if (themesRes.ok) {
    71→      const themesJson = (await themesRes.json()) as {
    72→        themes?: { id: number; name: string }[];
    73→      };
    74→      const mainTheme = themesJson.themes?.[0];
    75→
    76→      if (mainTheme) {
    77→        const numericId = String(mainTheme.id);
    78→        activeThemeName = mainTheme.name;
    79→        themeEditorUrl = `https://${shop}/admin/themes/${numericId}/editor`;
    80→        themeEditorAppEmbedsUrl = `https://${shop}/admin/themes/${numericId}/editor?context=apps`;
    81→
    82→        // Step 2 — read settings_data.json to detect app embed state.
    83→        // Shopify stores app embed blocks in config/settings_data.json under:
    84→        //   current.blocks[uuid] = { type: "shopify://apps/{handle}/blocks/app-embed", disabled: false }
    85→        try {
    86→          const assetUrl = `https://${shop}/admin/api/2026-01/themes/${numericId}/assets.json?asset%5Bkey%5D=config%2Fsettings_data.json`;
    87→          const assetRes = await fetch(assetUrl, {
    88→            headers: { "X-Shopify-Access-Token": session.accessToken ?? "" },
    89→          });
    90→
    91→          embedDebug.httpStatus = assetRes.status;
    92→
    93→          if (assetRes.ok) {
    94→            const assetJson = (await assetRes.json()) as {
    95→              asset?: { value?: string };
    96→            };
    97→            const content = assetJson?.asset?.value;
    98→            embedDebug.hasContent = !!content;
    99→
   100→            if (content) {
   101→              const settingsData = JSON.parse(content) as {
   102→                current?: { blocks?: Record<string, unknown> };
   103→              };
   104→              const blocks = settingsData?.current?.blocks ?? {};
   105→              embedDebug.rawBlocksSnippet = JSON.stringify(blocks).slice(0, 800);
   106→
   107→              embedDebug.blockTypes = Object.values(blocks)
   108→                .map((b) => (b as { type?: string }).type ?? "")
   109→                .filter(Boolean);
   110→
   111→              // Match on "app-embed" OR "app_embed" — covers file-name variants
   112→              appEmbedEnabled = Object.values(blocks).some((block) => {
   113→                const b = block as { type?: string; disabled?: boolean };
   114→                return (
   115→                  typeof b.type === "string" &&
   116→                  (b.type.includes("app-embed") || b.type.includes("app_embed")) &&
   117→                  b.disabled !== true
   118→                );
   119→              });
   120→            }
   121→          } else {
   122→            const errText = await assetRes.text().catch(() => "");
   123→            embedDebug.error = `HTTP ${assetRes.status}: ${errText.slice(0, 200)}`;
   124→          }
   125→        } catch (e) {
   126→          embedDebug.error = String(e);
   127→        }
   128→      } else {
   129→        embedDebug.error = "No main theme found in REST response";
   130→      }
   131→    } else {
   132→      embedDebug.error = `Themes REST HTTP ${themesRes.status}`;
   133→    }
   134→  } catch (e) {
   135→    embedDebug.error = `Outer catch: ${String(e)}`;
   136→  }
   137→
   138→  return {
   139→    config,
   140→    themeEditorUrl,
   141→    themeEditorAppEmbedsUrl,
   142→    activeThemeName,
   143→    appEmbedEnabled,
   144→    embedDebug,
   145→  };
   146→};

   148→// Helper: sync widget config to an App Installation metafield so the
   149→// storefront Liquid block can read it via app.metafields.zip_checker.widget_config.value
   150→//
   151→// IMPORTANT: ownerId MUST be the App Installation GID (not the Shop GID).
   152→// app.metafields in Liquid only surfaces metafields owned by the App Installation.
   153→// Using the Shop GID writes to shop metafields which are invisible to app.metafields.
   154→async function syncConfigMetafield(
   155→  admin: Awaited<ReturnType<typeof authenticate.admin>>["admin"],
   156→  configData: Record<string, unknown>,
   157→) {
   158→  try {
   159→    // Step 1: get the App Installation GID
   160→    const installResponse = await admin.graphql(
   161→      `query { currentAppInstallation { id } }`,
   162→    );
   163→    const installJson = (await installResponse.json()) as {
   164→      data: { currentAppInstallation: { id: string } };
   165→    };
   166→    const appInstallationGid = installJson.data.currentAppInstallation.id;
   167→
   168→    // Step 2: write the metafield to the App Installation
   169→    // Namespace must be plain "zip_checker" (no "$app:" prefix) for App Installation metafields.
   170→    // The Liquid block reads: app.metafields.zip_checker.widget_config.value
   171→    const metaResponse = await admin.graphql(
   172→      `mutation MetafieldsSet($metafields: [MetafieldsSetInput!]!) {
   173→        metafieldsSet(metafields: $metafields) {
   174→          metafields { id namespace key }
   175→          userErrors { field message }
   176→        }
   177→      }`,
   178→      {
   179→        variables: {
   180→          metafields: [
   181→            {
   182→              ownerId: appInstallationGid,
   183→              namespace: "zip_checker",
   184→              key: "widget_config",
   185→              type: "json",
   186→              value: JSON.stringify(configData),
   187→            },
   188→          ],
   189→        },
   190→      },
   191→    );
   192→    const metaJson = (await metaResponse.json()) as {
   193→      data: {
   194→        metafieldsSet: {
   195→          metafields: { id: string; namespace: string; key: string }[];
   196→          userErrors: { field: string; message: string }[];
   197→        };
   198→      };
   199→    };
   200→
   201→    if (metaJson.data.metafieldsSet.userErrors.length > 0) {
   202→      console.error(
   203→        "metafieldsSet userErrors:",
   204→        metaJson.data.metafieldsSet.userErrors,
   205→      );
   206→    }
   207→  } catch (e) {
   208→    // Non-fatal — the API fallback still works
   209→    console.error("Failed to sync widget config metafield:", e);
   210→  }
   211→}
