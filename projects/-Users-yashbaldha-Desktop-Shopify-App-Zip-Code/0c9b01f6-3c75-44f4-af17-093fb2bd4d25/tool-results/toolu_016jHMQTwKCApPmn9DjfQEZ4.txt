    49→export const loader = async ({ request }: LoaderFunctionArgs) => {
    50→  const { session } = await authenticate.admin(request);
    51→  const shop = session.shop;
    52→
    53→  const [zipCodes, subscription] = await Promise.all([
    54→    db.zipCode.findMany({
    55→      where: { shop },
    56→      orderBy: { createdAt: "desc" },
    57→    }),
    58→    getShopSubscription(shop),
    59→  ]);
    60→
    61→  const stats = {
    62→    total: zipCodes.length,
    63→    allowed: zipCodes.filter((z) => z.type === "allowed").length,
    64→    blocked: zipCodes.filter((z) => z.type === "blocked").length,
    65→    active: zipCodes.filter((z) => z.isActive).length,
    66→  };
    67→
    68→  return { zipCodes, stats, subscription };
    69→};

    71→export const action = async ({ request }: ActionFunctionArgs) => {
    72→  const { session } = await authenticate.admin(request);
    73→  const shop = session.shop;
    74→  const formData = await request.formData();
    75→  const intent = formData.get("intent");
    76→
    77→  if (intent === "add") {
    78→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
    79→    const label = String(formData.get("label") || "").trim();
    80→    const zone = String(formData.get("zone") || "").trim();
    81→    const message = String(formData.get("message") || "").trim();
    82→    const eta = String(formData.get("eta") || "").trim();
    83→    const type = String(formData.get("type") || "allowed");
    84→
    85→    if (!zipCode) return { error: "Zip code is required." };
    86→
    87→    const subscription = await getShopSubscription(shop);
    88→    const limits = PLAN_LIMITS[subscription.planTier];
    89→
    90→    if (type === "blocked" && !limits.allowBlocked) {
    91→      return {
    92→        error:
    93→          "Blocked zip codes are not available on the Free plan. Upgrade to Pro or Ultimate.",
    94→        upgradeRequired: true,
    95→      };
    96→    }
    97→
    98→    const currentCount = await db.zipCode.count({ where: { shop } });
    99→    if (limits.maxZipCodes !== Infinity && currentCount >= limits.maxZipCodes) {
   100→      return {
   101→        error: `You have reached the ${limits.maxZipCodes} zip code limit for the Free plan. Upgrade to Pro for unlimited zip codes.`,
   102→        upgradeRequired: true,
   103→      };
   104→    }
   105→
   106→    try {
   107→      await db.zipCode.create({
   108→        data: {
   109→          shop,
   110→          zipCode,
   111→          label: label || null,
   112→          zone: zone || null,
   113→          message: message || null,
   114→          eta: eta || null,
   115→          type,
   116→        },
   117→      });
   118→      return { success: true, action: "added", zipCode };
   119→    } catch {
   120→      return { error: `Zip code "${zipCode}" already exists.` };
   121→    }
   122→  }
   123→
   124→  if (intent === "delete") {
   125→    const id = String(formData.get("id"));
   126→    await db.zipCode.delete({ where: { id } });
   127→    return { success: true, action: "deleted" };
   128→  }
   129→
   130→  if (intent === "update") {
   131→    const id = String(formData.get("id"));
   132→    const zipCode = String(formData.get("zipCode")).trim().toUpperCase();
   133→    const label = String(formData.get("label") || "").trim();
   134→    const zone = String(formData.get("zone") || "").trim();
   135→    const message = String(formData.get("message") || "").trim();
   136→    const eta = String(formData.get("eta") || "").trim();
   137→    const type = String(formData.get("type") || "allowed");
   138→    const isActive = formData.get("isActive") === "true";
   139→
   140→    if (!zipCode) return { error: "Zip code is required." };
   141→
   142→    try {
   143→      await db.zipCode.update({
   144→        where: { id },
   145→        data: {
   146→          zipCode,
   147→          label: label || null,
   148→          zone: zone || null,
   149→          message: message || null,
   150→          eta: eta || null,
   151→          type,
   152→          isActive,
   153→        },
   154→      });
   155→      return { success: true, action: "updated", zipCode };
   156→    } catch {
   157→      return { error: `Failed to update zip code "${zipCode}". It may already exist.` };
   158→    }
   159→  }
   160→
   161→  if (intent === "toggle") {
   162→    const id = String(formData.get("id"));
   163→    const isActive = formData.get("isActive") === "true";
   164→    await db.zipCode.update({
   165→      where: { id },
   166→      data: { isActive: !isActive },
   167→    });
   168→    return { success: true, action: "toggled" };
   169→  }
   170→
   171→  if (intent === "check") {
   172→    const zipCode = String(formData.get("checkZip")).trim().toUpperCase();
   173→    const found = await db.zipCode.findUnique({
   174→      where: { shop_zipCode: { shop, zipCode } },
   175→    });
   176→    if (!found) return { checkResult: { found: false, zipCode } };
   177→    return { checkResult: { found: true, zipCode, record: found } };
   178→  }
   179→
   180→  if (intent === "bulk-import") {
   181→    const csvData = String(formData.get("csvData") || "");
   182→    if (!csvData.trim()) return { error: "No CSV data provided." };
   183→
   184→    const subscription = await getShopSubscription(shop);
   185→    const limits = PLAN_LIMITS[subscription.planTier];
   186→
   187→    const lines = csvData
   188→      .split("\n")
   189→      .map((l) => l.trim())
   190→      .filter((l) => l.length > 0);
   191→
   192→    // Skip header row if present
   193→    const startIdx =
   194→      lines.length > 0 &&
   195→      lines[0].toLowerCase().includes("zip")
   196→        ? 1
   197→        : 0;
   198→
   199→    const currentCount = await db.zipCode.count({ where: { shop } });
   200→    let imported = 0;
   201→    let skipped = 0;
   202→    const errors: string[] = [];
   203→
   204→    for (let i = startIdx; i < lines.length; i++) {
   205→      const cols = lines[i].split(",").map((c) => c.trim().replace(/^"|"$/g, ""));
   206→      const zipCode = (cols[0] || "").toUpperCase();
   207→      if (!zipCode) continue;
   208→
   209→      const zone = cols[1] || null;
   210→      const type = (cols[2] || "allowed").toLowerCase() === "blocked" ? "blocked" : "allowed";
   211→      const message = cols[3] || null;
   212→      const eta = cols[4] || null;
   213→
   214→      if (type === "blocked" && !limits.allowBlocked) {
   215→        skipped++;
   216→        continue;
   217→      }
   218→
   219→      if (
   220→        limits.maxZipCodes !== Infinity &&
   221→        currentCount + imported >= limits.maxZipCodes
   222→      ) {
   223→        errors.push(
   224→          `Reached the ${limits.maxZipCodes} zip code limit. ${lines.length - startIdx - imported - skipped} zip codes were not imported.`,
   225→        );
   226→        break;
   227→      }
   228→
   229→      try {
   230→        await db.zipCode.upsert({
   231→          where: { shop_zipCode: { shop, zipCode } },
   232→          create: {
   233→            shop,
   234→            zipCode,
   235→            zone,
   236→            type,
   237→            message,
   238→            eta,
   239→          },
   240→          update: {
   241→            zone: zone ?? undefined,
   242→            type,
   243→            message: message ?? undefined,
   244→            eta: eta ?? undefined,
   245→          },
   246→        });
   247→        imported++;
   248→      } catch {
   249→        skipped++;
   250→      }
   251→    }
   252→
   253→    return {
   254→      success: true,
   255→      action: "bulk-import",
   256→      imported,
   257→      skipped,
   258→      errors,
   259→    };
   260→  }
   261→
   262→  if (intent === "export") {
   263→    const zipCodes = await db.zipCode.findMany({
   264→      where: { shop },
   265→      orderBy: { createdAt: "desc" },
   266→    });
   267→
   268→    const header = "Zip Code,Zone,Status,Message,ETA,Active";
   269→    const rows = zipCodes.map(
   270→      (z) =>
   271→        `"${z.zipCode}","${z.zone || ""}","${z.type}","${z.message || ""}","${z.eta || ""}","${z.isActive ? "Yes" : "No"}"`,
   272→    );
   273→    const csv = [header, ...rows].join("\n");
   274→
   275→    return { success: true, action: "export", csv };
   276→  }
   277→
   278→  return null;
   279→};
