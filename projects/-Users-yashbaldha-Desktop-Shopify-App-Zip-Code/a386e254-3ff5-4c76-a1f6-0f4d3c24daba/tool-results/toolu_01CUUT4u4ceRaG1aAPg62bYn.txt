   259→   259→  // ── Scope custom CSS: auto-prefix selectors with widget container ID ──────
   260→   260→  function scopeCustomCss(rawCss, wid) {
   261→   261→    if (!rawCss) return "";
   262→   262→    // Strip </style> tags to prevent injection
   263→   263→    var css = rawCss.replace(/<\/style>/gi, "");
   264→   264→    var result = "";
   265→   265→    var i = 0;
   266→   266→    while (i < css.length) {
   267→   267→      // Skip whitespace
   268→   268→      while (i < css.length && /\s/.test(css[i])) { result += css[i]; i++; }
   269→   269→      if (i >= css.length) break;
   270→   270→      // Pass through @keyframes and @font-face unchanged
   271→   271→      if (css[i] === '@') {
   272→   272→        var atRule = "";
   273→   273→        var j = i;
   274→   274→        while (j < css.length && css[j] !== '{' && css[j] !== ';') { atRule += css[j]; j++; }
   275→   275→        var atName = atRule.trim().toLowerCase();
   276→   276→        if (atName.indexOf("@keyframes") === 0 || atName.indexOf("@font-face") === 0) {
   277→   277→          result += atRule;
   278→   278→          if (j < css.length && css[j] === '{') {
   279→   279→            var depth = 1; result += css[j]; j++;
   280→   280→            while (j < css.length && depth > 0) {
   281→   281→              if (css[j] === '{') depth++;
   282→   282→              if (css[j] === '}') depth--;
   283→   283→              result += css[j]; j++;
   284→   284→            }
   285→   285→          }
   286→   286→          i = j; continue;
   287→   287→        }
   288→   288→        // @media: scope inner rules
   289→   289→        if (atName.indexOf("@media") === 0) {
   290→   290→          result += atRule;
   291→   291→          if (j < css.length && css[j] === '{') {
   292→   292→            result += '{'; j++;
   293→   293→            // Collect inner content
   294→   294→            var depth = 1; var inner = "";
   295→   295→            while (j < css.length && depth > 0) {
   296→   296→              if (css[j] === '{') depth++;
   297→   297→              if (css[j] === '}') depth--;
   298→   298→              if (depth > 0) inner += css[j];
   299→   299→              j++;
   300→   300→            }
   301→   301→            result += scopeCustomCss(inner, wid);
   302→   302→            result += '}';
   303→   303→          }
   304→   304→          i = j; continue;
   305→   305→        }
   306→   306→      }
   307→   307→      // Regular rule: grab selector, prefix with #wid
   308→   308→      var sel = ""; var body = "";
   309→   309→      while (i < css.length && css[i] !== '{') { sel += css[i]; i++; }
   310→   310→      if (i < css.length) { i++; } // skip {
   311→   311→      var depth = 1;
   312→   312→      while (i < css.length && depth > 0) {
   313→   313→        if (css[i] === '{') depth++;
   314→   314→        if (css[i] === '}') depth--;
   315→   315→        if (depth > 0) body += css[i];
   316→   316→        i++;
   317→   317→      }
   318→   318→      if (sel.trim()) {
   319→   319→        // Split multiple selectors by comma and prefix each
   320→   320→        var selectors = sel.split(',').map(function(s) {
   321→   321→          s = s.trim();
   322→   322→          if (!s) return '';
   323→   323→          return '#' + wid + ' ' + s;
   324→   324→        }).filter(Boolean).join(', ');
   325→   325→        result += selectors + ' {' + body + '}';
   326→   326→      }
   327→   327→    }
   328→   328→    return result;
   329→   329→  }
