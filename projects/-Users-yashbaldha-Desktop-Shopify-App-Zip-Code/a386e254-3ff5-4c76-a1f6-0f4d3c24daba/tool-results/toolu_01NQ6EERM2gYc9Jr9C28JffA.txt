   259→  // ── Scope custom CSS: auto-prefix selectors with widget container ID ──────
   260→  function scopeCustomCss(rawCss, wid) {
   261→    if (!rawCss) return "";
   262→    // Strip </style> tags to prevent injection
   263→    var css = rawCss.replace(/<\/style>/gi, "");
   264→    var result = "";
   265→    var i = 0;
   266→    while (i < css.length) {
   267→      // Skip whitespace
   268→      while (i < css.length && /\s/.test(css[i])) { result += css[i]; i++; }
   269→      if (i >= css.length) break;
   270→      // Pass through @keyframes and @font-face unchanged
   271→      if (css[i] === '@') {
   272→        var atRule = "";
   273→        var j = i;
   274→        while (j < css.length && css[j] !== '{' && css[j] !== ';') { atRule += css[j]; j++; }
   275→        var atName = atRule.trim().toLowerCase();
   276→        if (atName.indexOf("@keyframes") === 0 || atName.indexOf("@font-face") === 0) {
   277→          result += atRule;
   278→          if (j < css.length && css[j] === '{') {
   279→            var depth = 1; result += css[j]; j++;
   280→            while (j < css.length && depth > 0) {
   281→              if (css[j] === '{') depth++;
   282→              if (css[j] === '}') depth--;
   283→              result += css[j]; j++;
   284→            }
   285→          }
   286→          i = j; continue;
   287→        }
   288→        // @media: scope inner rules
   289→        if (atName.indexOf("@media") === 0) {
   290→          result += atRule;
   291→          if (j < css.length && css[j] === '{') {
   292→            result += '{'; j++;
   293→            // Collect inner content
   294→            var depth = 1; var inner = "";
   295→            while (j < css.length && depth > 0) {
   296→              if (css[j] === '{') depth++;
   297→              if (css[j] === '}') depth--;
   298→              if (depth > 0) inner += css[j];
   299→              j++;
   300→            }
   301→            result += scopeCustomCss(inner, wid);
   302→            result += '}';
   303→          }
   304→          i = j; continue;
   305→        }
   306→      }
   307→      // Regular rule: grab selector, prefix with #wid
   308→      var sel = ""; var body = "";
   309→      while (i < css.length && css[i] !== '{') { sel += css[i]; i++; }
   310→      if (i < css.length) { i++; } // skip {
   311→      var depth = 1;
   312→      while (i < css.length && depth > 0) {
   313→        if (css[i] === '{') depth++;
   314→        if (css[i] === '}') depth--;
   315→        if (depth > 0) body += css[i];
   316→        i++;
   317→      }
   318→      if (sel.trim()) {
   319→        // Split multiple selectors by comma and prefix each
   320→        var selectors = sel.split(',').map(function(s) {
   321→          s = s.trim();
   322→          if (!s) return '';
   323→          return '#' + wid + ' ' + s;
   324→        }).filter(Boolean).join(', ');
   325→        result += selectors + ' {' + body + '}';
   326→      }
   327→    }
   328→    return result;
   329→  }
