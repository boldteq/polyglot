    49→    49→/** RFC-4180 compliant CSV row parser — handles quoted fields containing commas. */
    50→    50→function parseCsvRow(line: string): string[] {
    51→    51→  const cols: string[] = [];
    52→    52→  let cur = "";
    53→    53→  let inQuotes = false;
    54→    54→  for (let i = 0; i < line.length; i++) {
    55→    55→    const ch = line[i];
    56→    56→    if (ch === '"') {
    57→    57→      if (inQuotes && line[i + 1] === '"') {
    58→    58→        cur += '"';
    59→    59→        i++;
    60→    60→      } else {
    61→    61→        inQuotes = !inQuotes;
    62→    62→      }
    63→    63→    } else if (ch === "," && !inQuotes) {
    64→    64→      cols.push(cur.trim());
    65→    65→      cur = "";
    66→    66→    } else {
    67→    67→      cur += ch;
    68→    68→    }
    69→    69→  }
    70→    70→  cols.push(cur.trim());
    71→    71→  return cols;
    72→    72→}

   454→   454→          update: { zone: zone ?? undefined, type, message: message ?? undefined, eta: eta ?? undefined },
   455→   455→        });
   456→   456→        imported++;
   457→   457→      } catch {
   458→   458→        // skip individual errors
   459→   459→      }
   460→   460→    }
   461→   461→
   462→   462→    return { success: true, action: "range-import", imported, total: rangeSize };
   463→   463→  }
   464→   464→
   465→   465→  return null;
